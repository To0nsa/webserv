32/// 1 MiB default for request body limit.
33constexpr std::size_t DEFAULT_CLIENT_MAX_BODY_SIZE = 1 * 1024 * 1024;
34/// Default error page path used for common error codes.
35const std::string DEFAULT_ERROR_PAGE_PATH = "/error.html";
36/// Default filesystem root for locations missing an explicit root.
37const std::string DEFAULT_ROOT = "/var/www";
38/// Default index file for the root location if none provided.
39const std::string DEFAULT_INDEX = "index.html";
40/// Default allowed HTTP methods when none are specified.
41const std::vector<std::string> DEFAULT_METHODS = {"GET", "POST", "DELETE"};
42
43/**
44 * @brief Normalizes a single server and its locations.
45 *
46 * @details
47 * - Ensures a default `client_max_body_size` if unset (1 MiB).
48 * - Injects a default set of `error_page` mappings if none were provided.
49 * - For each location:
50 * - Sets a default `root` when missing.
51 * - Adds a default `index` when the location is `/` and index list is empty.
52 * - Ensures a default set of allowed HTTP methods when unspecified.
53 *
54 * @param server Server to normalize (modified in place).
55 *
56 * @ingroup config_normalizing
57 */
58static void normalizeServer(Server& server) {
59 // Fill default body size if not specified
60 if (server.getClientMaxBodySize() == 0) {
61 server.setClientMaxBodySize(DEFAULT_CLIENT_MAX_BODY_SIZE);
62 }
63
64 // Install a basic set of error pages if none provided
65 if (server.getErrorPages().empty()) {
66 server.setErrorPage(500, DEFAULT_ERROR_PAGE_PATH);
67 server.setErrorPage(404, DEFAULT_ERROR_PAGE_PATH);
68 server.setErrorPage(403, DEFAULT_ERROR_PAGE_PATH);
69 server.setErrorPage(502, DEFAULT_ERROR_PAGE_PATH);
70 }
71
72 // Normalize each location
73 for (Location& loc : server.getLocations()) {
74 // Root path default
75 if (loc.getRoot().empty()) {
76 loc.setRoot(DEFAULT_ROOT);
77 }
78
79 // Provide an index for "/" if none given
80 if (loc.getIndexFiles().empty() && loc.getPath() == "/") {
81 loc.addIndexFile(DEFAULT_INDEX);
82 }
83
84 // Ensure at least a default method set
85 if (!loc.hasAllowedMethods()) {
86 loc.setAllowedMethods(DEFAULT_METHODS);
87 }
88 }
89}

91/**
92 * @brief Applies normalization to all servers/locations in the config.
93 *
94 * @param config Parsed configuration to normalize (modified in place).
95 *
96 * @ingroup config_normalizing
97 */
98void normalizeConfig(Config& config) {
99 for (Server& s : config.getServers()) {
100 normalizeServer(s);
101 }
