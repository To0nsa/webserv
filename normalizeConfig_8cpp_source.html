27static void normalizeServer(Server& server) {
28 if (server.getClientMaxBodySize() == 0) {
29 server.setClientMaxBodySize(DEFAULT_CLIENT_MAX_BODY_SIZE);
30 }
31
32 if (server.getErrorPages().empty()) {
33 server.setErrorPage(500, DEFAULT_ERROR_PAGE_PATH);
34 server.setErrorPage(404, DEFAULT_ERROR_PAGE_PATH);
35 server.setErrorPage(403, DEFAULT_ERROR_PAGE_PATH);
36 server.setErrorPage(502, DEFAULT_ERROR_PAGE_PATH);
37 }
38
39 for (Location& loc : server.getLocations()) {
40 if (loc.getRoot().empty()) {
41 loc.setRoot(DEFAULT_ROOT);
42 }
43
44 if (loc.getIndexFiles().empty() && loc.getPath() == "/") {
45 loc.addIndexFile(DEFAULT_INDEX);
46 }
47
48 if (!loc.hasAllowedMethods()) {
49 loc.setAllowedMethods(DEFAULT_METHODS);
50 }
51 }
52}
