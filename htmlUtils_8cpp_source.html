15// HTML-escape &, <, >, ", '
16// so it's safe to embed in an HTML context.
17std::string htmlEscape(const std::string& in) {
18 std::string out;
19 out.reserve(in.size());
20 for (unsigned char c : in) {
21 switch (c) {
22 case '&':
23 out += "&amp;";
24 break;
25 case '<':
26 out += "&lt;";
27 break;
28 case '>':
29 out += "&gt;";
30 break;
31 case '"':
32 out += "&quot;";
33 break;
34 case '\'':
35 out += "&#39;";
36 break;
37 default:
38 out += c;

40 }
41 }
42 return out;
