128std::string Location::getCgiInterpreter(const std::string& ext) const {
129 std::string key = toLower(ext);
130 if (!key.empty() && key[0] != '.')
131 key = "." + key;
132
133 std::map<std::string, std::string>::const_iterator it = _cgi_interpreters.find(key);
134 return (it != _cgi_interpreters.end()) ? it->second : "";
135}

156bool Location::matchesPath(const std::string& uri) const {
157 std::string cleanUri = normalizePath(uri);
158 std::string locPath = normalizePath(_path);
159
160 return cleanUri.rfind(locPath, 0) == 0;
161}

163std::string Location::resolveAbsolutePath(const std::string& uri) const {
164 std::string cleanUri = normalizePath(uri);
165 if (!matchesPath(cleanUri) || _path.length() > cleanUri.length())
166 return "";
167 return joinPath(_root, cleanUri.substr(_path.length()));
168}
