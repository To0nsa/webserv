300/**
301 * @brief Returns the interpreter for a CGI extension.
302 *
303 * @param ext Extension (case-insensitive, optional dot).
304 * @return Interpreter path or empty string.
305 *
306 * @ingroup location_component
307 */
308std::string Location::getCgiInterpreter(const std::string& ext) const {
309 std::string key = toLower(ext);
310 if (!key.empty() && key[0] != '.')
311 key = "." + key;
312
313 std::map<std::string, std::string>::const_iterator it = _cgi_interpreters.find(key);
314 return (it != _cgi_interpreters.end()) ? it->second : "";
315}

353/**
354 * @brief Checks if a URI matches this location's path.
355 *
356 * @param uri Request URI.
357 * @return `true` if URI starts with location path.
358 *
359 * @ingroup location_component
360 */
361bool Location::matchesPath(const std::string& uri) const {
362 std::string cleanUri = normalizePath(uri);
363 std::string locPath = normalizePath(_path);
364 return cleanUri.rfind(locPath, 0) == 0;
365}
366
367/**
368 * @brief Resolves an absolute filesystem path from a URI.
369 *
370 * @param uri Request URI.
371 * @return Joined filesystem path, or empty on mismatch.
372 *
373 * @ingroup location_component
374 */
375std::string Location::resolveAbsolutePath(const std::string& uri) const {
376 std::string cleanUri = normalizePath(uri);
377 if (!matchesPath(cleanUri) || _path.length() > cleanUri.length())
378 return "";
379 return joinPath(_root, cleanUri.substr(_path.length()));
380}
