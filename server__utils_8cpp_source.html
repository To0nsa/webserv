16/**
17 * @brief Selects the best matching Server for a given port and Host header.
18 *
19 * @details This function implements name-based virtual host resolution.
20 * It searches for a server block that matches both the listening port
21 * and the requested `Host:` header. If no exact match is found, it falls
22 * back to the first server configured on the target port.
23 *
24 * @param servers All parsed Server instances.
25 * @param port The port the connection was accepted on.
26 * @param host_name The Host header from the HTTP request (e.g. "example.com").
27 * @return A reference to the matching Server instance.
28 *
29 * @throws std::runtime_error If no Server listens on the given port.
30 *
31 * @warning When using this function in unit tests, avoid passing string literals
32 * like `"example.com"` directly to `host_name` — this may trigger
33 * `-Werror=dangling-reference` with strict compilers. Use a `std::string` variable
34 * instead. This is not an issue in production code, where the Host header is
35 * always stored in a string variable.
36 */
37const Server& findMatchingServer(const std::vector<Server>& servers, int port,
38 const std::string& host_name) {
39 // Pass 1: Look for a server that matches both port and Host name
40 for (const Server& server : servers) {
41 if (server.getPort() == port && server.hasServerName(host_name)) {
42 return server; // Exact match
43 }
44 }
45
46 // Pass 2: No name match — fall back to the first server that listens on the same port
47 for (const Server& server : servers) {
48 if (server.getPort() == port) {
49 return server; // Fallback default server
50 }
51 }
52
53 // No server found on the target port — likely config error
54 throw std::runtime_error("No matching server found for port " + std::to_string(port));
55}
