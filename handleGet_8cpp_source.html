13/**
14 * @file handleGet.cpp
15 * @brief Implements the HTTP GET request handler.
16 *
17 * @details This file provides the logic for serving resources in response
18 * to HTTP GET requests. It includes:
19 * - **MIME type detection** (@ref detectMimeType) for correct
20 * `Content-Type` headers.
21 * - **File serving** (@ref serveFile) with small-file in-memory
22 * responses and large-file streaming responses.
23 * - **Directory handling** (@ref processDirectory):
24 * - Redirects URIs missing a trailing slash.
25 * - Serves configured `index` files if present.
26 * - Generates autoindex listings if enabled.
27 * - Returns 403 when directory listing is disabled.
28 * - **Main dispatcher** (@ref handleGet) which ties everything
29 * together, enforcing nginx-like behavior:
30 * - Rejects symlinks for security.
31 * - Differentiates between directories and regular files.
32 * - Returns appropriate errors (403/404) when access is denied
33 * or resources are missing.
34 *
35 * @ingroup request_handler
36 */

58/**
59 * @brief Detects the MIME type of a file based on its extension.
60 *
61 * @details Uses a static lookup table mapping common file extensions
62 * (e.g., `.html`, `.png`, `.json`) to their corresponding
63 * MIME types. The file extension is normalized to lowercase
64 * before lookup. If the extension is not recognized, the
65 * generic `application/octet-stream` type is returned.
66 *
67 * @param file_path Path to the file whose MIME type should be determined.
68 *
69 * @return A MIME type string suitable for the `Content-Type` header.
70 *
71 * @ingroup request_handler
72 */
73static std::string detectMimeType(const std::string& file_path) {
74 // Static lookup table of common extensions → MIME types
75 static const std::map<std::string, std::string> mime_types = {
76 {".html", "text/html"}, {".htm", "text/html"},
77 {".css", "text/css"}, {".js", "application/javascript"},
78 {".json", "application/json"}, {".txt", "text/plain"},
79 {".jpg", "image/jpeg"}, {".jpeg", "image/jpeg"},
80 {".png", "image/png"}, {".gif", "image/gif"},
81 {".svg", "image/svg+xml"}, {".ico", "image/x-icon"},
82 {".pdf", "application/pdf"}, {".zip", "application/zip"},
83 {".tar", "application/x-tar"}, {".xml", "application/xml"},
84 {".mp3", "audio/mpeg"}, {".mp4", "video/mp4"},
85 {".wasm", "application/wasm"}};
86
87 // Extract file extension from path
88 std::filesystem::path fsPath(file_path);
89 std::string ext = fsPath.extension().string();
90
91 // Normalize extension to lowercase for lookup
92 std::transform(ext.begin(), ext.end(), ext.begin(),
93 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
94
95 // Return known type if found, else default to binary
96 auto it = mime_types.find(ext);
97 if (it != mime_types.end())
98 return it->second;
99
100 return "application/octet-stream";
101}

103/**
104 * @brief Serves a static file in response to an HTTP GET request.
105 *
106 * @details This helper reads a file from disk and builds the
107 * appropriate @ref HttpResponse:
108 * - Verifies that the target is a valid file.
109 * - Opens the file in binary mode, rejects if inaccessible.
110 * - Determines the `Content-Type` header (uses
111 * @ref detectMimeType if none is provided).
112 * - If the file size is small (≤ 100 KiB), reads the
113 * entire file into memory and returns it inline.
114 * - If the file size is larger, streams it back using
115 * @ref ResponseBuilder::generateSuccessFile.
116 *
117 * Logs each decision (missing file, forbidden, in-memory,
118 * or streaming).
119 *
120 * @param file_path Filesystem path to the target file.
121 * @param request Incoming HTTP request.
122 * @param server Active server context (for error generation).
123 * @param content_type Optional MIME type override; if empty, auto-detected.
124 *
125 * @return A valid @ref HttpResponse:
126 * - 200 with body or stream on success.
127 * - 404 if file not found.
128 * - 403 if access is denied.
129 *
130 * @ingroup request_handler
131 */
132static HttpResponse serveFile(const std::string& file_path, const HttpRequest& request,
133 const Server& server, std::string content_type) {
134 // 1) Check if the path is a file
135 if (!isFile(file_path)) {
136 Logger::logFrom(LogLevel::WARN, "Get Handler", "File not found: " + file_path);
137 return ResponseBuilder::generateError(404, server, request);
138 }
139
140 // 2) Attempt to open the file
141 std::ifstream file(file_path, std::ios::binary | std::ios::ate);
142 if (!file.is_open()) {
143 Logger::logFrom(LogLevel::WARN, "Get Handler", "Cannot open file: " + file_path);
144 return ResponseBuilder::generateError(403, server, request);
145 }
146
147 // 3) Determine file size
148 std::streamsize size = file.tellg();
149 file.seekg(0, std::ios::beg);
150
151 // 4) Detect MIME type if not provided
152 if (content_type.empty())
153 content_type = detectMimeType(file_path);
154
155 // 5) Small files (≤100 KiB) → load fully into memory
156 constexpr std::streamsize MEMORY_LIMIT = 100 * 1024;
157 if (size <= MEMORY_LIMIT) {
158 std::ostringstream buffer;
159 buffer << file.rdbuf();
160 std::string body = buffer.str();
161
162 Logger::logFrom(LogLevel::INFO, "Get Handler",
163 "Serving small file in-memory (size: " + std::to_string(size) +
164 " bytes): " + file_path);
165 return ResponseBuilder::generateSuccess(200, body, content_type, request);
166 } else {
167 // 6) Large files → return as streaming response
168 Logger::logFrom(LogLevel::INFO, "Get Handler",
169 "Serving large file via streaming (size: " + std::to_string(size) +
170 " bytes): " + file_path);
171 return ResponseBuilder::generateSuccessFile(200, file_path, content_type, request, size);
172 }
173}

175/**
176 * @brief Handles HTTP GET requests targeting a directory.
177 *
178 * @details This function enforces nginx-like directory handling rules:
179 * 1. Normalize the URI and reject empty results (403).
180 * 2. If the URI does not end with a slash, issue a `301 Moved Permanently`
181 * redirect to the slash-terminated URI.
182 * 3. If an `index` file is configured:
183 * - Serve it if it exists.
184 * - Otherwise, if autoindex is enabled, generate a directory listing.
185 * - Otherwise, return 403 (listing disabled).
186 * 4. If no index is configured:
187 * - Generate autoindex if enabled.
188 * - Otherwise, return 403 (directory listing forbidden).
189 *
190 * @param dirPath Physical filesystem path to the requested directory.
191 * @param uri Original request URI.
192 * @param request Incoming HTTP request.
193 * @param server Active server context.
194 * @param loc The matched location block for this URI.
195 *
196 * @return A fully built @ref HttpResponse:
197 * - 301 Redirect if missing slash.
198 * - 200 OK with index file or autoindex if allowed.
199 * - 403 Forbidden if neither index nor autoindex is permitted.
200 *
201 * @ingroup request_handler
202 */
203static HttpResponse processDirectory(const std::string& dirPath, const std::string& uri,
204 const HttpRequest& request, const Server& server,
205 const Location& loc) {
206 // 1) Normalize URI and reject empty (security check)
207 std::string normalized = normalizePath(uri);
208 if (normalized.empty()) {
209 Logger::logFrom(LogLevel::WARN, "Get Handler",
210 "Normalized URI is empty (possible escape) for URI: " + uri);
211 return ResponseBuilder::generateError(403, server, request);
212 }
213
214 // 2) Redirect if URI does not end with '/'
215 if (normalized.back() != '/') {
216 std::string target = normalized + "/";
217 Logger::logFrom(LogLevel::INFO, "Get Handler", "Redirecting to: " + target);
218 return ResponseBuilder::generateRedirect(301, target, request);
219 }
220
221 // 3) If an index file is configured in the location
222 const std::string& indexName = loc.getIndex();
223 if (!indexName.empty()) {
224 std::string indexFullPath = joinPath(dirPath, indexName);
225 if (isFile(indexFullPath)) {
226 return serveFile(indexFullPath, request, server,
227 ""); // serve index.html (or configured name)
228 }
229
230 if (loc.isAutoindexEnabled()) {
231 return generateAutoindex(dirPath, uri, request, server); // fallback to autoindex
232 }
233
234 Logger::logFrom(LogLevel::WARN, "Get Handler",
235 "Index file not found and autoindex disabled for: " + dirPath);
236 return ResponseBuilder::generateError(403, server, request);
237 }
238
239 // 4) If no index: allow autoindex or forbid
240 if (loc.isAutoindexEnabled()) {
241 return generateAutoindex(dirPath, uri, request, server);
242 }
243
244 Logger::logFrom(LogLevel::WARN, "Get Handler",
245 "Directory listing not allowed (autoindex disabled) for: " + dirPath);
246 return ResponseBuilder::generateError(403, server, request);
247}

251/**
252 * @brief Handles an HTTP GET request for a resource.
253 *
254 * @details This function implements nginx-like GET semantics:
255 * 1. Resolve the request URI to a physical filesystem path.
256 * 2. If the URI ends with a `/` but points to a file (e.g., `/file/`):
257 * - Reject with 404 (trailing slash on file).
258 * 3. Reject symlinks for security (403).
259 * 4. If the path exists:
260 * - If it’s a directory, delegate to @ref processDirectory
261 * (redirect, index, or autoindex handling).
262 * - If it’s a regular file, delegate to @ref serveFile
263 * (in-memory or streaming response).
264 * 5. If nothing matches, return 404 (not found).
265 *
266 * @param request Incoming HTTP request object.
267 * @param server Active server context.
268 * @param loc The matched location block for this URI.
269 *
270 * @return A fully constructed @ref HttpResponse:
271 * - 200 OK with file contents or directory listing.
272 * - 301 Redirect for missing slash.
273 * - 403 Forbidden for symlinks or disallowed directories.
274 * - 404 Not Found when resource doesn’t exist.
275 *
276 * @ingroup request_handler
277 */
278HttpResponse handleGet(const HttpRequest& request, const Server& server, const Location& loc) {
279 // 1) Resolve request path to a filesystem path
280 std::string filepath = resolvePhysicalPath(request, loc);
281
282 // 2) Special nginx rule: if URI ends with '/' but points to a file → 404
283 const std::string& uri = request.getPath();
284 if (!uri.empty() && uri.back() == '/') {
285 std::string fileNoSlash = filepath;
286 if (!fileNoSlash.empty() && fileNoSlash.back() == '/')
287 fileNoSlash.pop_back();
288 if (isFile(fileNoSlash)) {
289 Logger::logFrom(LogLevel::WARN, "Get Handler",
290 "Trailing slash on file → returning 404 for URI: " + uri);
291 return ResponseBuilder::generateError(404, server, request);
292 }
293 }
294
295 // 3) Reject symlinks for security
296 if (isSymlink(filepath)) {
297 Logger::logFrom(LogLevel::WARN, "Get Handler",
298 "Symlink detected, rejecting request for: " + filepath);
299 return ResponseBuilder::generateError(403, server, request);
300 }
301
302 // 4) Check file type
303 struct stat fileStat;
304 if (stat(filepath.c_str(), &fileStat) == 0) {
305 const std::string& uri = request.getPath();
306
307 if (S_ISDIR(fileStat.st_mode)) {
308 // Directory → delegate to processDirectory
309 return processDirectory(filepath, uri, request, server, loc);
310 }
311
312 if (S_ISREG(fileStat.st_mode)) {
313 // Regular file → delegate to serveFile
314 return serveFile(filepath, request, server, "");
315 }
316 }
317
318 // 5) Not found or inaccessible
319 Logger::logFrom(LogLevel::WARN, "Get Handler", "File not found or inaccessible: " + filepath);
320 return ResponseBuilder::generateError(404, server, request);
321}
