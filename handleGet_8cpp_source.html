33static std::string detectMimeType(const std::string& file_path) {
34 static const std::map<std::string, std::string> mime_types = {
35 {".html", "text/html"}, {".htm", "text/html"},
36 {".css", "text/css"}, {".js", "application/javascript"},
37 {".json", "application/json"}, {".txt", "text/plain"},
38 {".jpg", "image/jpeg"}, {".jpeg", "image/jpeg"},
39 {".png", "image/png"}, {".gif", "image/gif"},
40 {".svg", "image/svg+xml"}, {".ico", "image/x-icon"},
41 {".pdf", "application/pdf"}, {".zip", "application/zip"},
42 {".tar", "application/x-tar"}, {".xml", "application/xml"},
43 {".mp3", "audio/mpeg"}, {".mp4", "video/mp4"},
44 {".wasm", "application/wasm"}};
45
46 std::filesystem::path fsPath(file_path);
47 std::string ext = fsPath.extension().string();
48
49 std::transform(ext.begin(), ext.end(), ext.begin(),
50 [](unsigned char c) { return static_cast<char>(std::tolower(c)); });
51
52 auto it = mime_types.find(ext);
53 if (it != mime_types.end())
54 return it->second;
55
56 return "application/octet-stream";
57}

59static HttpResponse serveFile(const std::string& file_path, const HttpRequest& request,
60 const Server& server, std::string content_type) {
61 if (!isFile(file_path)) {
62 Logger::logFrom(LogLevel::WARN, "Get Handler", "File not found: " + file_path);
63 return ResponseBuilder::generateError(404, server, request);
64 }
65
66 std::ifstream file(file_path, std::ios::binary | std::ios::ate);
67 if (!file.is_open()) {
68 Logger::logFrom(LogLevel::WARN, "Get Handler", "Cannot open file: " + file_path);
69 return ResponseBuilder::generateError(403, server, request);
70 }
71
72 std::streamsize size = file.tellg();
73 file.seekg(0, std::ios::beg);
74
75 if (content_type.empty())
76 content_type = detectMimeType(file_path);
77
78 constexpr std::streamsize MEMORY_LIMIT = 100 * 1024;
79 if (size <= MEMORY_LIMIT) {
80 std::ostringstream buffer;
81 buffer << file.rdbuf();
82 std::string body = buffer.str();
83
84 Logger::logFrom(LogLevel::INFO, "Get Handler",
85 "Serving small file in-memory (size: " + std::to_string(size) +
86 " bytes): " + file_path);
87 return ResponseBuilder::generateSuccess(200, body, content_type, request);
88 } else {
89 Logger::logFrom(LogLevel::INFO, "Get Handler",
90 "Serving large file via streaming (size: " + std::to_string(size) +
91 " bytes): " + file_path);
92 return ResponseBuilder::generateSuccessFile(200, file_path, content_type, request, size);
93 }
94}

96static HttpResponse processDirectory(const std::string& dirPath, const std::string& uri,
97 const HttpRequest& request, const Server& server,
98 const Location& loc) {
99 std::string normalized = normalizePath(uri);
100 if (normalized.empty()) {
101 Logger::logFrom(LogLevel::WARN, "Get Handler",
102 "Normalized URI is empty (possible escape) for URI: " + uri);
103 return ResponseBuilder::generateError(403, server, request);
104 }
105
106 if (normalized.back() != '/') {
107 std::string target = normalized + "/";
108 Logger::logFrom(LogLevel::INFO, "Get Handler", "Redirecting to: " + target);
109 return ResponseBuilder::generateRedirect(301, target, request);
110 }
111
112 const std::string& indexName = loc.getIndex();
113 if (!indexName.empty()) {
114 std::string indexFullPath = joinPath(dirPath, indexName);
115 if (isFile(indexFullPath)) {
116 return serveFile(indexFullPath, request, server, "");
117 }
118
119 if (loc.isAutoindexEnabled()) {
120 return generateAutoindex(dirPath, uri, request, server);
121 }
122
123 Logger::logFrom(LogLevel::WARN, "Get Handler",
124 "Index file not found and autoindex disabled for: " + dirPath);
125 return ResponseBuilder::generateError(403, server, request);
126 }
127
128 if (loc.isAutoindexEnabled()) {
129 return generateAutoindex(dirPath, uri, request, server);
130 }
131
132 Logger::logFrom(LogLevel::WARN, "Get Handler",
133 "Directory listing not allowed (autoindex disabled) for: " + dirPath);
134 return ResponseBuilder::generateError(403, server, request);
135}

139HttpResponse handleGet(const HttpRequest& request, const Server& server, const Location& loc) {
140 std::string filepath = resolvePhysicalPath(request, loc);
141
142 // NGINX: any GET ending in '/' is a directory lookup.
143 // If stripping the slash yields an existing file, return 404.
144 const std::string& uri = request.getPath();
145 if (!uri.empty() && uri.back() == '/') {
146 std::string fileNoSlash = filepath;
147 if (!fileNoSlash.empty() && fileNoSlash.back() == '/')
148 fileNoSlash.pop_back();
149 if (isFile(fileNoSlash)) {
150 Logger::logFrom(LogLevel::WARN, "Get Handler",
151 "Trailing slash on file → returning 404 for URI: " + uri);
152 return ResponseBuilder::generateError(404, server, request);
153 }
154 }
155
156 if (isSymlink(filepath)) {
157 Logger::logFrom(LogLevel::WARN, "Get Handler",
158 "Symlink detected, rejecting request for: " + filepath);
159 return ResponseBuilder::generateError(403, server, request);
160 }
161
162 struct stat fileStat;
163 if (stat(filepath.c_str(), &fileStat) == 0) {
164 const std::string& uri = request.getPath();
165
166 if (S_ISDIR(fileStat.st_mode)) {
167 return processDirectory(filepath, uri, request, server, loc);
168 }
169
170 if (S_ISREG(fileStat.st_mode)) {
171 return serveFile(filepath, request, server, "");
172 }
173 }
174
175 Logger::logFrom(LogLevel::WARN, "Get Handler", "File not found or inaccessible: " + filepath);
176 return ResponseBuilder::generateError(404, server, request);
177}
