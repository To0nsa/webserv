25std::string getDefaultMessage(int status_code) {
26 // Define a local alias for better readability and maintainability
27 using StatusMessageMap = std::map<int, std::string>;
28 // Static map ensures it's initialized once and reused on every call
29 static const StatusMessageMap status_messages = {{200, "OK"},
30 {201, "Created"},
31 {301, "Moved Permanently"},
32 {302, "Found"},
33 {307, "Temporary Redirect"},
34 {308, "Permanent Redirect"},
35 {400, "Bad Request"},
36 {403, "Forbidden"},
37 {404, "Not Found"},
38 {405, "Method Not Allowed"},
39 {408, "Request Timeout"},
40 {411, "Length Required"},
41 {413, "Payload Too Large"},
42 {414, "Request-URI Too Long"},
43 {415, "Unsupported Media Type"},
44 {417, "Expectation Failed"},
45 {431, "Request Header Fields Too Large"},
46 {500, "Internal Server Error"},
47 {501, "Not Implemented"},
48 {502, "Bad Gateway"},
49 {503, "Service Unavailable"},
50 {504, "Gateway Timeout"},
51 {505, "HTTP Version Not Supported"}};
52
53 // Attempt to find the status code in the map
54 StatusMessageMap::const_iterator it = status_messages.find(status_code);
55 // Return the associated message or a fallback if unknown
56 return (it != status_messages.end()) ? it->second : "Unknown Error";

62// Determines keep-alive status from HTTP version and connection header.
63bool shouldKeepAlive(const std::string& version, const std::string& conn) {
64 // HTTP/1.1 defaults to keep-alive unless explicitly closed
65 if (version == "HTTP/1.1")
66 return conn != "close";
67 // HTTP/1.0 defaults to close unless explicitly kept alive
68 if (version == "HTTP/1.0")
69 return conn == "keep-alive";
70 // Unknown or unsupported version → safest behavior: close
71 return false;
72}

74// Common initialization for any response.
75void initializeResponse(HttpResponse& response, int status_code, const std::string& message,
76 const HttpRequest& request) {
77 // Set HTTP status code and message (e.g., 200 OK)
78 response.setStatus(status_code, message);
79 // Store request version and connection header to guide connection persistence
80 response.setRequestMeta(request.getVersion(), request.getHeader("Connection"));
81
82 // Extract relevant headers from the request for keep-alive logic
83 const std::string& conn = request.getHeader("Connection");
84 const std::string& version = request.getVersion();
85
86 // These status codes indicate the server should always close the connection
87 static const std::set<int> force_close_codes = {400, 408, 413, 500};
88
89 // Determine if we should keep the connection alive based on protocol rules and status
90 bool keep_alive = shouldKeepAlive(version, conn) && !force_close_codes.count(status_code);
91
92 // Set the appropriate Connection header in the response
93 response.setHeader("Connection", keep_alive ? "keep-alive" : "close");
94}

133HttpResponse generateError(int status_code, const Server& server, const HttpRequest& request) {
134 HttpResponse response;
135 std::string message = MessageHandler::getDefaultMessage(status_code);
136 std::string body;
137
138 // Try custom error page
139 const auto& error_pages = server.getErrorPages();
140 auto it = error_pages.find(status_code);
141 if (it != error_pages.end()) {
142 std::string uri = it->second;
143 // Strip leading slash
144 if (!uri.empty() && uri.front() == '/')
145 uri.erase(0, 1);
146 // Determine default root (location "/")
147 std::string default_root;
148 for (const auto& loc : server.getLocations()) {
149 if (loc.getPath() == "/") {
150 default_root = loc.getRoot();
151 break;
152 }
153 }
154 std::string full_path = joinPath(default_root, uri);
155 std::ifstream file(full_path.c_str());
156 if (file)
157 body.assign((std::istreambuf_iterator<char>(file)), std::istreambuf_iterator<char>());
158 }
159
160 // Fallback default page
161 if (body.empty()) {
162 std::ostringstream ss;
163 ss << "<html><body><h1>" << status_code << " " << htmlEscape(message)
164 << "</h1></body></html>";
165 body = ss.str();
166 }
167
168 initializeResponse(response, status_code, message, request);
169 response.setHeader("Content-Type", "text/html");
170 // Attach the generated or loaded error page body
171 if (!body.empty()) {
172 response.setBody(body);
173 }
174 /* response.setBody(body); */
175
176 return response;
177}
