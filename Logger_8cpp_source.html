13/**
14 * @file Logger.cpp
15 * @brief Simple color-coded logging utility.
16 *
17 * @details Implements the @ref Logger class methods and related helper
18 * functions for mapping log levels to human-readable strings
19 * and ANSI terminal colors. Supports standard log levels
20 * (`DEBUG`, `INFO`, `WARN`, `ERROR`) and can optionally
21 * prefix messages with a source identifier.
22 *
23 * Output is written to `stdout` for non-error messages and
24 * `stderr` for `ERROR` level messages, ensuring separation
25 * of normal and error output streams. ANSI escape codes are
26 * used for color, which may not render correctly in all
27 * terminals.
28 *
29 * @ingroup utils
30 *
31 * @note This logger is intended for human-readable output during
32 * development and runtime diagnostics, not structured logging.
33 * For production environments or log parsing, consider extending
34 * it to support formats like JSON or syslog.
35 */

48std::string levelToString(LogLevel level) {

78void Logger::log(LogLevel level, const std::string& message) {
79 if (level == LogLevel::ERROR) {
80 std::cerr << levelColor(level) << "[" << levelToString(level) << "]" << COLOR_RESET << " "
81 << message << std::endl;
82 return;
83 }
84 std::cout << levelColor(level) << "[" << levelToString(level) << "]" << COLOR_RESET << " "
85 << message << std::endl;
86}

88void Logger::logFrom(LogLevel level, const std::string& from, const std::string& message) {
