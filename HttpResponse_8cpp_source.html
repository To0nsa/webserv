43 * @brief Stores HTTP version and Connection header from the request.

45 * @details This metadata is required to determine whether the connection should be
46 * kept alive or closed after the response is sent. The HTTP version and the client's
47 * `Connection` header together define the default persistence behavior according to
48 * RFC 7230 §6.3. This method must be called before sending the response to ensure
49 * proper behavior in `isConnectionClose()`.
50 *
51 * @param version The HTTP version from the client's request (e.g., "HTTP/1.1").
52 * @param conn The value of the Connection header from the client's request.
53 */

59/**
60 * @brief Determines whether the connection should be closed after the response.
61 *
62 * @details This logic properly accounts for the HTTP version and the client's
63 * Connection header to implement persistent connections correctly.
64 *
65 * Unlike the previous version, which only checked if `Connection: close` was present,
66 * this method enforces the default semantics of each protocol version:
67 * - HTTP/1.1 assumes keep-alive unless explicitly closed.
68 * - HTTP/1.0 assumes close unless explicitly kept alive.
69 *
70 * This behavior is compliant with RFC 7230 §6.3 and avoids premature connection
71 * termination when clients do not send a `Connection` header.
72 *
73 * @return `true` if the connection should be closed, `false` to keep it alive.
74 */
75bool HttpResponse::isConnectionClose() const {
76 // 1) If the response explicitly sets Connection, honor it.
77 std::map<std::string, std::string>::const_iterator hit = _headers.find("Connection");
78 if (hit != _headers.end()) {
79 std::string v = hit->second;
80 std::transform(v.begin(), v.end(), v.begin(), ::tolower);
81 return v == "close";
82 }
83
84 // 2) Certain status codes must close (RFC-friendly behavior).
85 static const std::set<int> force_close_codes = {400, 408, 413, 500};
86 if (force_close_codes.count(_status_code))
87 return true;
88
89 // 3) Fall back to protocol semantics using the request metadata.
90 std::string conn = _connection_header;
91 std::transform(conn.begin(), conn.end(), conn.begin(), ::tolower);
92
93 if (_http_version == "HTTP/1.1") {
94 // Keep-alive by default unless client asked to close
95 return conn == "close";
96 }
97 if (_http_version == "HTTP/1.0") {
98 // Close by default unless client asked to keep-alive
99 return conn != "keep-alive";
100 }
101
102 // Unknown version: safest is to close.
103 return true;
104}

106std::string HttpResponse ::toHttpString(void) const {
107 std::stringstream ss;
108
109 ss << "HTTP/1.1 " << _status_code << " " << _status_message << "\r\n";
110
111 for (std::map<std::string, std::string>::const_iterator it = _headers.begin();
112 it != _headers.end(); ++it)
113 ss << it->first << ": " << it->second << "\r\n";
114
115 ss << "Content-Length: " << _body.length() << "\r\n";
116 ss << "\r\n";
117 ss << _body;
118
119 return ss.str();
120}
