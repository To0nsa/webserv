140/**
141 * @brief Serializes the response into HTTP wire format.
142 *
143 * @details Produces a string containing the status line, headers,
144 * `Content-Length`, and body, ready for transmission.
145 *
146 * @return HTTP response string.
147 * @ingroup http
148 */
149std::string HttpResponse::toHttpString(void) const {
150 std::stringstream ss;
151
152 ss << "HTTP/1.1 " << _status_code << " " << _status_message << "\r\n";
153
154 for (std::map<std::string, std::string>::const_iterator it = _headers.begin();
155 it != _headers.end(); ++it)
156 ss << it->first << ": " << it->second << "\r\n";
157
158 ss << "Content-Length: " << _body.length() << "\r\n";
159 ss << "\r\n";
160 ss << _body;
161
162 return ss.str();
163}

215/**
216 * @brief Determines whether the server should close the connection
217 * after sending this response.
218 *
219 * @details The decision follows HTTP semantics and some defensive RFC-inspired
220 * rules:
221 * 1. **Explicit Connection header in the response**:
222 * If present, its value takes precedence (case-insensitive).
223 * - `"Connection: close"` → close the connection.
224 * - Any other value → keep open, unless other rules apply.
225 *
226 * 2. **Force-close status codes**:
227 * Certain error codes (400, 408, 413, 500) always require closing
228 * the connection to remain protocol-compliant and prevent reuse
229 * of a potentially invalid connection.
230 *
231 * 3. **Fallback to request metadata**:
232 * - For HTTP/1.1: keep-alive by default, unless request said `"close"`.
233 * - For HTTP/1.0: close by default, unless request said `"keep-alive"`.
234 *
235 * 4. **Unknown HTTP version**:
236 * Defaults to closing the connection for safety.
237 *
238 * @return `true` if the server must close the TCP connection,
239 * `false` if it can be kept alive.
240 *
241 * @ingroup http
242 */
243bool HttpResponse::isConnectionClose() const {
244 // 1) If the response explicitly sets Connection, honor it.
245 std::map<std::string, std::string>::const_iterator hit = _headers.find("Connection");
246 if (hit != _headers.end()) {
247 std::string v = hit->second;
248 std::transform(v.begin(), v.end(), v.begin(), ::tolower);
249 return v == "close";
250 }
251
252 // 2) Certain status codes must close (RFC-friendly behavior).
253 static const std::set<int> force_close_codes = {400, 408, 413, 500};
254 if (force_close_codes.count(_status_code))
255 return true;
256
257 // 3) Fall back to protocol semantics using the request metadata.
258 std::string conn = _connection_header;
259 std::transform(conn.begin(), conn.end(), conn.begin(), ::tolower);
260
261 if (_http_version == "HTTP/1.1") {
262 // Keep-alive by default unless client asked to close
263 return conn == "close";
264 }
265 if (_http_version == "HTTP/1.0") {
266 // Close by default unless client asked to keep-alive
267 return conn != "keep-alive";
268 }
269
270 // Unknown version: safest is to close.
271 return true;
272}
