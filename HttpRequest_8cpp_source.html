28void HttpRequest::printRequest() const {
29 // Header
30 std::cout << "\n===== HTTP REQUEST =====\n"
31 << "Method: " << _method << "\n"
32 << "Path: " << _path << "\n"
33 << "Query: " << (_query.empty() ? "(none)" : _query) << "\n"
34 << "Version: " << _version << "\n\n";
35
36 // Headers
37 std::cout << "----- Headers -----\n";
38 if (_headers.empty()) {
39 std::cout << " (none)\n";
40 } else {
41 for (const auto& [key, value] : _headers) {
42 std::cout << " " << std::left << std::setw(20) << key << ": " << value << "\n";
43 }
44 }
45
46 // Body
47 std::cout << "\n------ Body ------\n";
48 if (_body.empty()) {
49 std::cout << " (empty)\n";
50 } else {
51 // Check if all chars are printable
52 bool allPrint = std::all_of(_body.begin(), _body.end(), [](char c) {
53 return std::isprint(static_cast<unsigned char>(c));
54 });
55
56 if (!allPrint) {
57 std::cout << " (binary data)\n";
58 } else {
59 // Truncate to 100 chars
60 std::size_t showLen = std::min<std::size_t>(_body.size(), 100);
61 std::string snippet = _body.substr(0, showLen);
62 std::cout << " " << snippet;
63 if (_body.size() > showLen) {
64 std::cout << "... (+" << (_body.size() - showLen) << " more)";
65 }
66 std::cout << "\n";
67 }
68 }
69
70 // Footer
71 std::cout << "=======================\n";
72}
