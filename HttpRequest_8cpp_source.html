99/**
100 * @brief Returns the value of a header.
101 *
102 * @details Header lookup is case-insensitive. If the header is not present,
103 * returns a reference to a static empty string.
104 *
105 * @param key Header name to look up.
106 * @return Reference to the header value, or an empty string if not found.
107 * @ingroup http
108 */
109const std::string& HttpRequest::getHeader(const std::string& key) const {
110 const std::string upperKey = toUpper(key);
111 static const std::string empty = "";
112 std::map<std::string, std::string>::const_iterator it = _headers.find(upperKey);
113 if (it != _headers.end())
114 return it->second;
115 return empty;
116}
