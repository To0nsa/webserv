30void HttpRequest::printRequest() const {
31 // Header
32 std::cout << "\n===== HTTP REQUEST =====\n"
33 << "Method: " << _method << "\n"
34 << "Path: " << _path << "\n"
35 << "Query: " << (_query.empty() ? "(none)" : _query) << "\n"
36 << "Version: " << _version << "\n\n";
37
38 // Headers
39 std::cout << "----- Headers -----\n";
40 if (_headers.empty()) {
41 std::cout << " (none)\n";
42 } else {
43 for (const auto& [key, value] : _headers) {
44 std::cout << " " << std::left << std::setw(20) << key << ": " << value << "\n";
45 }
46 }
47
48 // Body
49 std::cout << "\n------ Body ------\n";
50 if (_body.empty()) {
51 std::cout << " (empty)\n";
52 } else {
53 // Check if all chars are printable
54 bool allPrint = std::all_of(_body.begin(), _body.end(), [](char c) {
55 return std::isprint(static_cast<unsigned char>(c));
56 });
57
58 if (!allPrint) {
59 std::cout << " (binary data)\n";
60 } else {
61 // Truncate to 100 chars
62 std::size_t showLen = std::min<std::size_t>(_body.size(), 100);
63 std::string snippet = _body.substr(0, showLen);
64 std::cout << " " << snippet;
65 if (_body.size() > showLen) {
66 std::cout << "... (+" << (_body.size() - showLen) << " more)";
67 }
68 std::cout << "\n";
69 }
70 }
71
72 // Footer
73 std::cout << "=======================\n";
74}
