37// 1) Scan directory entries: separate names into dirs/files, keep “..” first if present

39 std::string name;
40 std::uintmax_t size;
41 std::time_t mtime;
42 bool isDir;
43};

47std::vector<DirEntry> listDirectoryEntries(const std::string& path, bool& ok) {
48 std::vector<DirEntry> entries;
49 ok = true;
50
51 fs::path dirPath(path);
52 if (!fs::exists(dirPath) || !fs::is_directory(dirPath)) {
53 ok = false;
54 return {};
55 }
56
57 // Try to construct a directory_iterator without throwing:
58 std::error_code dirEc;
59 fs::directory_iterator it(dirPath, dirEc);
60 if (dirEc) {
61 ok = false;
62 return {};
63 }
64
65 // Now loop in a non‐throwing way:
66 std::error_code entryEc;
67 for (; it != fs::directory_iterator{}; it.increment(entryEc)) {
68 if (entryEc) {
69 entryEc.clear();
70 continue;
71 }
72
73 fs::path p = it->path();
74 std::error_code statEc;
75
76 fs::file_time_type ftime = fs::last_write_time(p, statEc);
77 if (statEc)
78 continue;
79
80 DirEntry entry;
81 entry.name = p.filename().string();
82
83 std::error_code dirCheckEc;
84 entry.isDir = fs::is_directory(p, dirCheckEc) && !dirCheckEc;
85
86 if (entry.isDir) {
87 entry.size = 0ULL;
88 } else {
89 entry.size = fs::file_size(p, statEc);
90 if (statEc)
91 entry.size = 0ULL;
92 }
93
94 // Convert ftime → time_t
95 auto sctp = std::chrono::time_point_cast<std::chrono::system_clock::duration>(
96 ftime - fs::file_time_type::clock::now() + std::chrono::system_clock::now());
97 entry.mtime = std::chrono::system_clock::to_time_t(sctp);
98
99 entries.push_back(entry);
100 }
101
102 // Insert “..” if parent exists:
103 fs::path parent = dirPath.parent_path();
104 if (!parent.empty() && fs::exists(parent)) {
105 DirEntry up;
106 up.name = "..";
107 up.isDir = true;
108 up.size = 0;
109 up.mtime = 0;
110 entries.insert(entries.begin(), up);
111 }
112
113 // Sort “..” at front, then other dirs, then files:
114 std::vector<DirEntry> dirs, files;
115 for (auto const& e : entries) {
116 if (e.name == "..")
117 continue;
118 if (e.isDir)
119 dirs.push_back(e);
120 else
121 files.push_back(e);
122 }
123 auto cmp = [](auto const& a, auto const& b) { return a.name < b.name; };
124 std::sort(dirs.begin(), dirs.end(), cmp);
125 std::sort(files.begin(), files.end(), cmp);
126
127 std::vector<DirEntry> sorted;
128 if (!entries.empty() && entries[0].name == "..")
129 sorted.push_back(entries[0]);
130 sorted.insert(sorted.end(), dirs.begin(), dirs.end());
131 sorted.insert(sorted.end(), files.begin(), files.end());
132 return sorted;
133}

135std::string escapeUriComponent(const std::string& s) {
136 std::ostringstream oss;
137 oss << std::uppercase << std::hex;
138 for (unsigned char c : s) {
139 if (std::isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
140 oss << c;
141 } else {
142 oss << '%' << std::setw(2) << std::setfill('0') << static_cast<int>(c);
143 }
144 }
145 return oss.str();
146}

148std::string formatMTimeUTC(std::time_t t) {
149 std::tm* gmPtr = std::gmtime(&t);
150 if (!gmPtr) {
151 return {};
152 }
153 std::ostringstream ss;
154 ss << std::put_time(gmPtr, "%d-%b-%Y %H:%M");
155 return ss.str();
156}

158void renderRow(std::ostringstream& body, const DirEntry& entry, const std::string& baseUri) {
159 bool isParent = (entry.name == "..");
160 bool isDir = entry.isDir;
161
162 std::string href;
163 std::string disp = entry.name;
164 if (isParent) {
165 href = baseUri + "../";
166 disp = "../";
167 } else {
168 href = baseUri + escapeUriComponent(entry.name);
169 if (isDir) {
170 disp += "/";
171 href += "/";
172 }
173 }
174
175 std::string icon = isDir ? "📁" : "📄";
176 std::string mtime = isParent ? "-" : formatMTimeUTC(entry.mtime);
177 std::string size = (isDir || isParent) ? "-" : std::to_string(entry.size);
178
179 body << " <tr>" << "<td><a href=\"" << htmlEscape(href) << "\">" << icon << " "
180 << htmlEscape(disp) << "</a></td>" << "<td>" << mtime << "</td>" << "<td>" << size
181 << "</td>" << "</tr>\n";
182}

208HttpResponse generateAutoindex(const std::string& filepath, const std::string& uri,
209 const HttpRequest& request, const Server& server) {
210 bool dirOk;
211 auto entries = listDirectoryEntries(filepath, dirOk);
212 if (!dirOk) {
213 Logger::logFrom(LogLevel::WARN, "Autoindex",
214 "Cannot open directory: " + filepath +
215 " → rejecting autoindex for URI: " + request.getPath());
216 return ResponseBuilder::generateError(403, server, request);
217 }
218
219 // Ensure baseUri ends with “/”
220 std::string baseUri = uri;
221 if (baseUri.empty() || baseUri.back() != '/')
222 baseUri += '/';
223
224 // Build the HTML body
225 std::ostringstream body;
226 body << htmlHeader(uri);
227 for (auto const& e : entries) {
228 renderRow(body, e, baseUri);
229 }
230 body << htmlFooter();
231
232 Logger::logFrom(LogLevel::INFO, "Autoindex",
233 "Generated autoindex for URI: " + request.getPath() +
234 " (directory: " + filepath + ")");
235 return ResponseBuilder::generateSuccess(200, body.str(), "text/html", request);
236}
