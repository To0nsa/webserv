34// 1) Scan directory entries: separate names into dirs/files, keep “..” first if present
35struct DirEntry {
36 std::string name;
37 std::uintmax_t size;
38 std::time_t mtime;
39 bool isDir;
40};

44std::vector<DirEntry> listDirectoryEntries(const std::string& path, bool& ok) {
45 std::vector<DirEntry> entries;
46 ok = true;
47
48 fs::path dirPath(path);
49 if (!fs::exists(dirPath) || !fs::is_directory(dirPath)) {
50 ok = false;
51 return {};
52 }
53
54 // Try to construct a directory_iterator without throwing:
55 std::error_code dirEc;
56 fs::directory_iterator it(dirPath, dirEc);
57 if (dirEc) {
58 ok = false;
59 return {};
60 }
61
62 // Now loop in a non‐throwing way:
63 std::error_code entryEc;
64 for (; it != fs::directory_iterator{}; it.increment(entryEc)) {
65 if (entryEc) {
66 entryEc.clear();
67 continue;
68 }
69
70 fs::path p = it->path();
71 std::error_code statEc;
72
73 fs::file_time_type ftime = fs::last_write_time(p, statEc);
74 if (statEc)
75 continue;
76
77 DirEntry entry;
78 entry.name = p.filename().string();
79
80 std::error_code dirCheckEc;
81 entry.isDir = fs::is_directory(p, dirCheckEc) && !dirCheckEc;
82
83 if (entry.isDir) {
84 entry.size = 0ULL;
85 } else {
86 entry.size = fs::file_size(p, statEc);
87 if (statEc)
88 entry.size = 0ULL;
89 }
90
91 // Convert ftime → time_t
92 auto sctp = std::chrono::time_point_cast<std::chrono::system_clock::duration>(
93 ftime - fs::file_time_type::clock::now() + std::chrono::system_clock::now());
94 entry.mtime = std::chrono::system_clock::to_time_t(sctp);
95
96 entries.push_back(entry);
97 }
98
99 // Insert “..” if parent exists:
100 fs::path parent = dirPath.parent_path();
101 if (!parent.empty() && fs::exists(parent)) {
102 DirEntry up;
103 up.name = "..";
104 up.isDir = true;
105 up.size = 0;
106 up.mtime = 0;
107 entries.insert(entries.begin(), up);
108 }
109
110 // Sort “..” at front, then other dirs, then files:
111 std::vector<DirEntry> dirs, files;
112 for (auto const& e : entries) {
113 if (e.name == "..")
114 continue;
115 if (e.isDir)
116 dirs.push_back(e);
117 else
118 files.push_back(e);
119 }
120 auto cmp = [](auto const& a, auto const& b) { return a.name < b.name; };
121 std::sort(dirs.begin(), dirs.end(), cmp);
122 std::sort(files.begin(), files.end(), cmp);
123
124 std::vector<DirEntry> sorted;
125 if (!entries.empty() && entries[0].name == "..")
126 sorted.push_back(entries[0]);
127 sorted.insert(sorted.end(), dirs.begin(), dirs.end());
128 sorted.insert(sorted.end(), files.begin(), files.end());
129 return sorted;
130}

132std::string escapeUriComponent(const std::string& s) {
133 std::ostringstream oss;
134 oss << std::uppercase << std::hex;
135 for (unsigned char c : s) {
136 if (std::isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
137 oss << c;
138 } else {
139 oss << '%' << std::setw(2) << std::setfill('0') << static_cast<int>(c);
140 }
141 }
142 return oss.str();
143}

145std::string formatMTimeUTC(std::time_t t) {
146 // Convert time_t → chrono::sys_time
147 std::chrono::sys_time<std::chrono::seconds> tp{std::chrono::seconds{t}};
148
149 // Round to minutes to match "%d-%b-%Y %H:%M"
150 auto tp_minutes = std::chrono::floor<std::chrono::minutes>(tp);
151
152 return std::format("{:%d-%b-%Y %H:%M}", tp_minutes);
153}

155void renderRow(std::ostringstream& body, const DirEntry& entry, const std::string& baseUri) {
156 bool isParent = (entry.name == "..");
157 bool isDir = entry.isDir;
158
159 std::string href;
160 std::string disp = entry.name;
161 if (isParent) {
162 href = baseUri + "../";
163 disp = "../";
164 } else {
165 href = baseUri + escapeUriComponent(entry.name);
166 if (isDir) {
167 disp += "/";
168 href += "/";
169 }
170 }
171
172 std::string icon = isDir ? "📁" : "📄";
173 std::string mtime = isParent ? "-" : formatMTimeUTC(entry.mtime);
174 std::string size = (isDir || isParent) ? "-" : std::to_string(entry.size);
175
176 body << " <tr>" << "<td><a href=\"" << htmlEscape(href) << "\">" << icon << " "
177 << htmlEscape(disp) << "</a></td>" << "<td>" << mtime << "</td>" << "<td>" << size
178 << "</td>" << "</tr>\n";
179}

205HttpResponse generateAutoindex(const std::string& filepath, const std::string& uri,
206 const HttpRequest& request, const Server& server) {
207 bool dirOk;
208 auto entries = listDirectoryEntries(filepath, dirOk);
209 if (!dirOk) {
210 Logger::logFrom(LogLevel::WARN, "Autoindex",
211 "Cannot open directory: " + filepath +
212 " → rejecting autoindex for URI: " + request.getPath());
213 return ResponseBuilder::generateError(403, server, request);
214 }
215
216 // Ensure baseUri ends with “/”
217 std::string baseUri = uri;
218 if (baseUri.empty() || baseUri.back() != '/')
219 baseUri += '/';
220
221 // Build the HTML body
222 std::ostringstream body;
223 body << htmlHeader(uri);
224 for (auto const& e : entries) {
225 renderRow(body, e, baseUri);
226 }
227 body << htmlFooter();
228
229 Logger::logFrom(LogLevel::INFO, "Autoindex",
230 "Generated autoindex for URI: " + request.getPath() +
231 " (directory: " + filepath + ")");
232 return ResponseBuilder::generateSuccess(200, body.str(), "text/html", request);
233}
