13/**
14 * @file generateAutoindex.cpp
15 * @brief Implements directory autoindex (HTML listing) generation.
16 *
17 * @details This file provides nginx-like autoindex functionality for HTTP GET
18 * requests on directories when no index file is found and autoindexing
19 * is enabled in the location block.
20 *
21 * Main responsibilities:
22 * - **Directory scanning** (@ref listDirectoryEntries):
23 * Collects entries, separates directories from files, preserves `..`.
24 * - **Escaping utilities**:
25 * - @ref escapeUriComponent for safe URIs.
26 * - @ref htmlEscape (from utils/htmlUtils) for HTML-safe output.
27 * - **Formatting helpers**:
28 * - @ref formatMTimeUTC for human-readable UTC timestamps.
29 * - @ref renderRow for HTML table rows with icons and metadata.
30 * - **HTML layout**:
31 * - @ref htmlHeader and @ref htmlFooter wrap the index table.
32 * - **Entrypoint**: @ref generateAutoindex assembles the full HTML
33 * directory listing and returns a @ref HttpResponse.
34 *
35 * Errors (nonexistent or inaccessible directories) result in 403
36 * responses. Valid directories produce a simple HTML table with:
37 * - File/directory name (with 📄/📁 icons).
38 * - Last modified timestamp (UTC).
39 * - File size (bytes).
40 *
41 * @ingroup request_handler
42 */

99/**
100 * @brief Lists and sorts directory entries for autoindex generation.
101 *
102 * @details Scans the given filesystem path and collects metadata into
103 * @ref DirEntry objects. Handles errors in a non-throwing way
104 * using `std::error_code`. Results are separated into:
105 * - `".."` (if parent exists, inserted at the front).
106 * - Subdirectories (sorted alphabetically).
107 * - Files (sorted alphabetically).
108 *
109 * Each entry records:
110 * - `name`: filename or `".."`.
111 * - `isDir`: directory vs file.
112 * - `size`: file size in bytes (0 for directories).
113 * - `mtime`: last modification time (UTC).
114 *
115 * @param path Filesystem path to the directory.
116 * @param ok Reference flag set to `false` if the directory
117 * cannot be opened or scanned safely.
118 *
119 * @return A vector of @ref DirEntry objects sorted for autoindex rendering.
120 *
121 * @ingroup request_handler
122 */
123std::vector<DirEntry> listDirectoryEntries(const std::string& path, bool& ok) {
124 std::vector<DirEntry> entries;
125 ok = true;
126
127 fs::path dirPath(path);
128 if (!fs::exists(dirPath) || !fs::is_directory(dirPath)) {
129 ok = false;
130 return {};
131 }
132
133 // Try non-throwing directory iterator
134 std::error_code dirEc;
135 fs::directory_iterator it(dirPath, dirEc);
136 if (dirEc) {
137 ok = false;
138 return {};
139 }
140
141 // Walk entries safely (clear errors on failure)
142 std::error_code entryEc;
143 for (; it != fs::directory_iterator{}; it.increment(entryEc)) {
144 if (entryEc) { // skip unreadable entries
145 entryEc.clear();
146 continue;
147 }
148
149 fs::path p = it->path();
150 std::error_code statEc;
151
152 // Get last modification time
153 fs::file_time_type ftime = fs::last_write_time(p, statEc);
154 if (statEc)
155 continue;
156
157 DirEntry entry;
158 entry.name = p.filename().string();
159
160 // Detect directory
161 std::error_code dirCheckEc;
162 entry.isDir = fs::is_directory(p, dirCheckEc) && !dirCheckEc;
163
164 // Get size (only for files)
165 if (entry.isDir) {
166 entry.size = 0ULL;
167 } else {
168 entry.size = fs::file_size(p, statEc);
169 if (statEc)
170 entry.size = 0ULL;
171 }
172
173 // Convert ftime → time_t for portable rendering
174 auto sctp = std::chrono::time_point_cast<std::chrono::system_clock::duration>(
175 ftime - fs::file_time_type::clock::now() + std::chrono::system_clock::now());
176 entry.mtime = std::chrono::system_clock::to_time_t(sctp);
177
178 entries.push_back(entry);
179 }
180
181 // Insert “..” if parent directory exists
182 fs::path parent = dirPath.parent_path();
183 if (!parent.empty() && fs::exists(parent)) {
184 DirEntry up;
185 up.name = "..";
186 up.isDir = true;
187 up.size = 0;
188 up.mtime = 0;
189 entries.insert(entries.begin(), up);
190 }
191
192 // Sort: “..” first, then directories, then files (alphabetically)
193 std::vector<DirEntry> dirs, files;
194 for (auto const& e : entries) {
195 if (e.name == "..")
196 continue;
197 if (e.isDir)
198 dirs.push_back(e);
199 else
200 files.push_back(e);
201 }
202 auto cmp = [](auto const& a, auto const& b) { return a.name < b.name; };
203 std::sort(dirs.begin(), dirs.end(), cmp);
204 std::sort(files.begin(), files.end(), cmp);
205
206 // Build sorted result
207 std::vector<DirEntry> sorted;
208 if (!entries.empty() && entries[0].name == "..")
209 sorted.push_back(entries[0]); // keep parent first
210 sorted.insert(sorted.end(), dirs.begin(), dirs.end());
211 sorted.insert(sorted.end(), files.begin(), files.end());
212 return sorted;
213}

215/**
216 * @brief Percent-encodes a string for safe inclusion in a URI component.
217 *
218 * @details Encodes all characters except the unreserved set defined by RFC 3986:
219 * `ALPHA / DIGIT / "-" / "_" / "." / "~"`.
220 * Other bytes are converted into `%HH` format using uppercase hex digits.
221 *
222 * @param s Input string (raw filename or path segment).
223 *
224 * @return A URI-safe string with reserved characters percent-encoded.
225 *
226 * @note This is used when generating autoindex links to ensure that spaces,
227 * control characters, and unsafe symbols don’t break the resulting URL.
228 *
229 * @ingroup request_handler
230 */
231std::string escapeUriComponent(const std::string& s) {
232 std::ostringstream oss;
233 oss << std::uppercase << std::hex;
234
235 for (unsigned char c : s) {
236 // Allowed characters remain unchanged
237 if (std::isalnum(c) || c == '-' || c == '_' || c == '.' || c == '~') {
238 oss << c;
239 } else {
240 // Encode everything else as %HH
241 oss << '%' << std::setw(2) << std::setfill('0') << static_cast<int>(c);
242 }
243 }
244 return oss.str();
245}

247/**
248 * @brief Formats a file modification time into a human-readable UTC string.
249 *
250 * @details Converts a `time_t` (last modification time) into a textual
251 * representation using `gmtime()` and `std::put_time`.
252 * The format is: `DD-Mon-YYYY HH:MM` (UTC).
253 *
254 * @param t Last modification time as `std::time_t`.
255 *
256 * @return A formatted string (e.g., `"19-Aug-2025 09:42"`) or an empty
257 * string if conversion fails.
258 *
259 * @note This is used in autoindex HTML tables for the **Last Modified** column.
260 *
261 * @ingroup request_handler
262 */
263std::string formatMTimeUTC(std::time_t t) {
264 std::tm* gmPtr = std::gmtime(&t); // Convert to UTC (thread-unsafe)
265 if (!gmPtr) {
266 return {};
267 }
268 std::ostringstream ss;
269 ss << std::put_time(gmPtr, "%d-%b-%Y %H:%M"); // Format: "19-Aug-2025 09:42"
270 return ss.str();
271}

273/**
274 * @brief Renders a single `<tr>` row in the autoindex HTML table.
275 *
276 * @details Converts a directory entry into a clickable HTML link with
277 * metadata (last modified, size).
278 * - The `".."` entry is treated as a parent directory link.
279 * - Directories get a trailing slash (`/`) in both display and href.
280 * - Files show their size in bytes, while directories/`".."` show `"-"`.
281 * - Icons: 📁 for directories, 📄 for files.
282 *
283 * @param body Output stringstream accumulating the HTML response.
284 * @param entry Directory entry metadata (name, size, mtime, type).
285 * @param baseUri Base URI (must end with `/`) to prepend to href links.
286 *
287 * @note This function applies both @ref escapeUriComponent (for href safety)
288 * and @ref htmlEscape (for display safety), preventing XSS or broken links.
289 *
290 * @ingroup request_handler
291 */
292void renderRow(std::ostringstream& body, const DirEntry& entry, const std::string& baseUri) {
293 bool isParent = (entry.name == "..");
294 bool isDir = entry.isDir;
295
296 // Construct href and display name
297 std::string href;
298 std::string disp = entry.name;
299 if (isParent) {
300 href = baseUri + "../"; // parent directory link
301 disp = "../";
302 } else {
303 href = baseUri + escapeUriComponent(entry.name); // percent-encode special chars
304 if (isDir) {
305 disp += "/";
306 href += "/";
307 }
308 }
309
310 // Metadata columns
311 std::string icon = isDir ? "📁" : "📄";
312 std::string mtime = isParent ? "-" : formatMTimeUTC(entry.mtime);
313 std::string size = (isDir || isParent) ? "-" : std::to_string(entry.size);
314
315 // Append HTML table row
316 body << " <tr>" << "<td><a href=\"" << htmlEscape(href) << "\">" << icon << " "
317 << htmlEscape(disp) << "</a></td>" << "<td>" << mtime << "</td>" << "<td>" << size
318 << "</td>" << "</tr>\n";
319}

321/**
322 * @brief Generates the opening HTML markup for an autoindex page.
323 *
324 * @details Builds the document `<head>`, page title, and initial table
325 * structure for directory listing.
326 * The provided @p uri is escaped with @ref htmlEscape to prevent
327 * XSS and is displayed in both the `<title>` and `<h1>` headers.
328 *
329 * @param uri The request URI being indexed (e.g. "/images/").
330 * @return HTML string containing the document header and table headers.
331 *
332 * @ingroup request_handler
333 */
334std::string htmlHeader(const std::string& uri) {
335 std::ostringstream ss;
336 ss << R"(<!DOCTYPE html>
337<html lang="en">
338<head>
339 <meta charset="UTF-8">
340 <title>Index of )"
341 << htmlEscape(uri) << R"(</title>
342</head>
343<body>
344 <h1>Index of )"
345 << htmlEscape(uri) << R"(</h1>
346 <table>
347 <tr><th>Name</th><th>Last Modified (UTC)</th><th>Size</th></tr>
348)";
349 return ss.str();
350}

368/**
369 * @brief Generates an HTML autoindex page for a directory listing.
370 *
371 * @details This function inspects the given filesystem path, enumerates
372 * its directory entries via @ref listDirectoryEntries, and
373 * produces an HTML table showing subdirectories and files with
374 * name, last modification time, and size.
375 *
376 * Workflow:
377 * 1. **Directory validation**: Ensure the target path exists and is
378 * accessible. If not, return a `403 Forbidden` error.
379 * 2. **Base URI normalization**: Guarantee that the URI ends with
380 * a trailing slash (`/`) for proper link resolution.
381 * 3. **HTML generation**: Build the autoindex page by composing
382 * @ref htmlHeader, calling @ref renderRow for each entry, and
383 * closing with @ref htmlFooter.
384 * 4. **Response assembly**: Return a `200 OK` response with the
385 * generated HTML body and `text/html` MIME type.
386 *
387 * @param filepath Filesystem path to the directory being indexed.
388 * @param uri Request URI corresponding to the directory.
389 * @param request Incoming HTTP request context.
390 * @param server Server configuration (used for error responses).
391 *
392 * @return HttpResponse with a full HTML autoindex page on success,
393 * or an error response (403) if the directory cannot be accessed.
394 *
395 * @ingroup request_handler
396 */
397HttpResponse generateAutoindex(const std::string& filepath, const std::string& uri,
398 const HttpRequest& request, const Server& server) {
399 bool dirOk;
400 auto entries = listDirectoryEntries(filepath, dirOk);
401 if (!dirOk) {
402 // Directory not accessible → return 403 Forbidden
403 Logger::logFrom(LogLevel::WARN, "Autoindex",
404 "Cannot open directory: " + filepath +
405 " → rejecting autoindex for URI: " + request.getPath());
406 return ResponseBuilder::generateError(403, server, request);
407 }
408
409 // Ensure base URI ends with “/” for proper relative link building
410 std::string baseUri = uri;
411 if (baseUri.empty() || baseUri.back() != '/')
412 baseUri += '/';
413
414 // Build the HTML body: header + rows + footer
415 std::ostringstream body;
416 body << htmlHeader(uri);
417 for (auto const& e : entries) {
418 renderRow(body, e, baseUri); // Add a row for each directory/file entry
419 }
420 body << htmlFooter();
421
422 // Log and return success response with generated HTML
423 Logger::logFrom(LogLevel::INFO, "Autoindex",
424 "Generated autoindex for URI: " + request.getPath() +
425 " (directory: " + filepath + ")");
426 return ResponseBuilder::generateSuccess(200, body.str(), "text/html", request);
427}
