70// Set up sockets for each server (host:port)
71void SocketManager::setupSockets(const std::vector<Server>& servers) {
72 std::set<std::pair<std::string, int>> bound;
73 for (size_t i = 0; i < servers.size(); ++i) {
74 const std::string& host = servers[i].getHost();
75 int port = servers[i].getPort();
76
77 std::pair<std::string, int> key = std::make_pair(host, port);
78 if (bound.count(key))
79 continue; // Already bound, skip
80 bound.insert(key);
81
82 int fd = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK, 0);
83 if (fd < 0)
84 throw SocketError("socket() failed: " + std::string(std::strerror(errno)));
85
86 int opt = 1; // Reuse same address/port if the server is restarted quickly
87 if (setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)) < 0) {
88 close(fd);
89 throw SocketError("setsockopt() failed: " + std::string(std::strerror(errno)));
90 }
91
92 sockaddr_in addr;
93 addr.sin_family = AF_INET;
94 addr.sin_port = htons(port); // Convert port to network byte order
95 // Convert hostname to IP address
96 if (host == "localhost")
97 addr.sin_addr.s_addr = inet_addr("127.0.0.1");
98 else
99 addr.sin_addr.s_addr = inet_addr(host.c_str());
100
101 if (bind(fd, (sockaddr*) &addr, sizeof(addr)) < 0) { // Bind socket to IP:port
102 close(fd);
103 throw SocketError("bind() failed on " + host + ":" + std::to_string(port) + ": " +
104 strerror(errno));
105 }
106
107 if (listen(fd, SOMAXCONN) < 0) { // Start listening for incoming connections
108 close(fd);
109 throw SocketError("listen() failed: " + std::string(std::strerror(errno)));
110 }
111
112 // Register fd in poll list
113 _poll_fds.push_back((pollfd){fd, POLLIN, 0});
114 // Collect all servers for this host:port
115 std::vector<Server> vhosts;
116 for (size_t j = 0; j < servers.size(); ++j) {
117 if (servers[j].getHost() == host && servers[j].getPort() == port)
118 vhosts.push_back(servers[j]);
119 }
120
121 _listen_map[fd] = vhosts;
122 Logger::logFrom(LogLevel::INFO, "SocketManager",
123 "Listening on " + host + ":" + std::to_string(port));
124 }
125}

127void SocketManager::run() {
128 while (running) {
129 int n = poll(&_poll_fds[0], _poll_fds.size(), 1000);
130 if (n < 0) {
131 if (errno == EINTR) {
132 running = 0;
133 continue;
134 }
135 throw SocketError("poll() failed: " + std::string(std::strerror(errno)));
136 }
137
138 handleCgiPollEvents();
139
140 // Then handle sockets — but skip *all* CGI FDs before doing error/HUP checks
141 for (size_t i = _poll_fds.size(); i-- > 0;) {
142 try {
143 short revents = _poll_fds[i].revents;
144 int current_fd = _poll_fds[i].fd;
145 if (checkClientTimeouts(current_fd, i)) {
146 _poll_fds[i].events |=
147 POLLOUT; // If connection keep-alive but client idle we close
148 }
149
150 // **FIX**: skip CGI pipe FDs entirely
151 if (_fd_to_cgi.contains(current_fd)) {
152 continue;
153 }
154
155 // Now error/hangup on *client* sockets
156 if (revents & POLLERR || revents & POLLHUP || revents & POLLNVAL) {
157 handlePollError(current_fd, i, revents);
158 continue;
159 }
160
161 if (revents & POLLIN) {
162 if (_listen_map.count(current_fd)) {
163 handleNewConnection(current_fd);
164 } else {
165 if (!handleClientData(current_fd, i))
166 continue;
167 // we have a response queued, request poll‐out
168 _poll_fds[i].events |= POLLOUT;
169 }
170 }
171
172 if ((revents & POLLOUT) && !_client_info[current_fd].responses.empty()) {
173 sendResponse(current_fd, i);
174 }
175 } catch (const std::exception& e) {
176 Logger::logFrom(LogLevel::ERROR, "SocketManager",
177 "Exception in poll loop for fd: " +
178 std::to_string(_poll_fds[i].fd) + ": " + e.what());
179 cleanupClientConnectionClose(_poll_fds[i].fd, i);
180 } catch (...) {
181 Logger::logFrom(LogLevel::ERROR, "SocketManager",
182 "Unknown exception in poll loop for fd: " +
183 std::to_string(_poll_fds[i].fd));
184 cleanupClientConnectionClose(_poll_fds[i].fd, i);
185 }
186 }
187 }
188
189 Logger::logFrom(LogLevel::INFO, "SocketManager", "Shutting down server");
190}

205// Accept new client and add to poll list
206void SocketManager::handleNewConnection(int listen_fd) {
207 int client_fd = accept(listen_fd, NULL, NULL);
208 if (client_fd < 0) {
209 if (errno == EMFILE || errno == ENFILE) {
210 // We’ve hit the per‐process or system FD limit.
211 Logger::logFrom(
212 LogLevel::ERROR, "SocketManager",
213 "Out of file descriptors (accept failed: " + std::string(strerror(errno)) + ")");
214 return;
215 }
216 Logger::logFrom(LogLevel::ERROR, "SocketManager",
217 "accept() failed: " + std::string(std::strerror(errno)));
218 return;
219 }
220
221 if (_poll_fds.size() >= MAX_CLIENTS) {
222 Logger::logFrom(LogLevel::WARN, "SocketManager",
223 "Maximum client limit reached. Rejecting connection.");
224 close(client_fd); // Optionally send HTTP error before closing (nice but optional) 503
225 return;
226 }
227
228 initializeClientInfo(client_fd, listen_fd);
229 _poll_fds.push_back((pollfd){client_fd, POLLIN, 0});
230 Logger::logFrom(LogLevel::kDEBUG, "SocketManager",
231 "Accept returned fd: " + std::to_string(client_fd) +
232 " | current open clients: " + std::to_string(_client_info.size()));
233}

235bool SocketManager::handleClientData(int client_fd, size_t index) {
236 try {
237 if (!receiveFromClient(client_fd, index))
238 return false;
239 if (!parseAndQueueRequests(client_fd))
240 return false;
241 processPendingRequests(client_fd);
242 return (true);
243 } catch (const std::bad_alloc& e) {
244 Logger::logFrom(LogLevel::ERROR, "SocketManager",
245 "Memory allocation failed while handling client " +
246 std::to_string(client_fd) + ": " + e.what());
247 } catch (const std::exception& e) {
248 Logger::logFrom(LogLevel::ERROR, "SocketManager",
249 "Exception while handling client " + std::to_string(client_fd) + ": " +
250 e.what());
251 } catch (...) {
252 Logger::logFrom(LogLevel::ERROR, "SocketManager",
253 "Unknown exception while handling client " + std::to_string(client_fd));
254 }
255 _poll_fds[index].events &= ~POLLIN;
256 respondError(client_fd, 500);
257 return (true);
258}
