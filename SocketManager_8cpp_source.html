96 *
97 * @details
98 * Interprets the poll revents mask for a given client file descriptor

100 * - `POLLNVAL`: Invalid poll event on this fd.
101 * - `POLLERR`: Generic socket error.
102 * - `POLLHUP`: Client disconnected.
103 *
104 * After logging, the client connection is closed and cleaned up
105 * via @ref cleanupClientConnectionClose.

111void SocketManager::handlePollError(int fd, size_t index, short revents) {
112 if (revents & POLLNVAL) {
113 Logger::logFrom(LogLevel::ERROR, "SocketManager",
114 "Invalid poll event on fd: " + std::to_string(fd));
115 } else if (revents & POLLERR) {
116 Logger::logFrom(LogLevel::ERROR, "SocketManager",
117 "Socket error on fd: " + std::to_string(fd));
118 } else if (revents & POLLHUP) {
119 Logger::logFrom(LogLevel::INFO, "SocketManager",
120 "Client disconnected (POLLHUP) on fd: " + std::to_string(fd));
121 }
122 cleanupClientConnectionClose(fd, index);

134/**
135 * @brief Initializes and binds listening sockets for all configured servers.
136 *
137 * @details
138 * Iterates over the provided @ref Server instances and:
139 * - Ensures each (host, port) pair is only bound once.
140 * - Creates a non-blocking TCP socket (`SOCK_STREAM | SOCK_NONBLOCK`).
141 * - Enables `SO_REUSEADDR` to allow quick restarts on the same port.
142 * - Binds the socket to the host:port.
143 * - Puts the socket into listening mode with `listen()`.
144 * - Registers the socket in the internal poll list for event monitoring.
145 * - Groups servers sharing the same host:port into a virtual host list.
146 *
147 * @param servers List of servers to initialize listening sockets for.
148 *
149 * @throws SocketError If socket creation, setsockopt, bind, or listen fails.
150 *
151 * @note
152 * - "localhost" is explicitly mapped to `127.0.0.1`.
153 * - Duplicate host:port entries are skipped to avoid double-binding.
154 */
155void SocketManager::setupSockets(const std::vector<Server>& servers) {
156 std::set<std::pair<std::string, int>> bound; // Track already-bound host:port pairs
157
158 for (size_t i = 0; i < servers.size(); ++i) {
159 const std::string& host = servers[i].getHost();
160 int port = servers[i].getPort();
161
162 // Skip if this host:port has already been bound
163 std::pair<std::string, int> key = std::make_pair(host, port);
164 if (bound.count(key))
165 continue;
166 bound.insert(key);
167
168 // Create a non-blocking TCP socket
169 int fd = socket(AF_INET, SOCK_STREAM | SOCK_NONBLOCK, 0);
170 if (fd < 0)
171 throw SocketError("socket() failed: " + std::string(std::strerror(errno)));
172
173 // Allow socket address reuse to restart quickly after shutdown
174 int opt = 1;
175 if (setsockopt(fd, SOL_SOCKET, SO_REUSEADDR, &opt, sizeof(opt)) < 0) {
176 close(fd);
177 throw SocketError("setsockopt() failed: " + std::string(std::strerror(errno)));
178 }
179
180 // Prepare socket address structure
181 sockaddr_in addr;
182 addr.sin_family = AF_INET;
183 addr.sin_port = htons(port); // Convert port to network byte order
184
185 // Map "localhost" to loopback explicitly, otherwise use given host string
186 if (host == "localhost")
187 addr.sin_addr.s_addr = inet_addr("127.0.0.1");
188 else
189 addr.sin_addr.s_addr = inet_addr(host.c_str());
190
191 // Bind socket to the host:port
192 if (bind(fd, (sockaddr*) &addr, sizeof(addr)) < 0) {
193 close(fd);
194 throw SocketError("bind() failed on " + host + ":" + std::to_string(port) + ": " +
195 strerror(errno));
196 }
197
198 // Put socket into listening mode
199 if (listen(fd, SOMAXCONN) < 0) {
200 close(fd);
201 throw SocketError("listen() failed: " + std::string(std::strerror(errno)));
202 }
203
204 // Register fd in poll list for event monitoring
205 _poll_fds.push_back((pollfd){fd, POLLIN, 0});
206
207 // Collect all servers configured on this host:port (virtual hosts)
208 std::vector<Server> vhosts;
209 for (size_t j = 0; j < servers.size(); ++j) {
210 if (servers[j].getHost() == host && servers[j].getPort() == port)
211 vhosts.push_back(servers[j]);
212 }
213
214 // Associate this listening fd with its vhost set
215 _listen_map[fd] = vhosts;
216
217 Logger::logFrom(LogLevel::INFO, "SocketManager",
218 "Listening on " + host + ":" + std::to_string(port));
219 }
220}

222/**
223 * @brief Runs the main event loop of the server.
224 *
225 * @details
226 * This method drives the **non-blocking, poll()-based event loop** of Webserv.
227 * - Uses a single `poll()` call to multiplex all I/O (listening sockets, client FDs, CGI pipes).
228 * - Periodically checks for CGI activity and client timeouts.
229 * - Handles:
230 * - New connections (on listening sockets),
231 * - Read events (incoming client data),
232 * - Write events (pending responses),
233 * - Error/hangup events (POLLERR, POLLHUP, POLLNVAL).
234 * - Ensures resilience with try/catch around per-FD handling.
235 *
236 * The loop continues until the global `running` flag is cleared (SIGINT).
237 *
238 * @throws SocketError If `poll()` fails unexpectedly (other than EINTR).
239 */
240void SocketManager::run() {
241 while (running) {
242 // Wait up to 1s for activity on any registered FD
243 int n = poll(&_poll_fds[0], _poll_fds.size(), 1000);
244 if (n < 0) {
245 if (errno == EINTR) {
246 // Interrupted by signal → trigger graceful shutdown
247 running = 0;
248 continue;
249 }
250 throw SocketError("poll() failed: " + std::string(std::strerror(errno)));
251 }
252
253 // First, handle any active CGI processes (timeout, completion, cleanup)
254 handleCgiPollEvents();
255
256 // Process poll events for each file descriptor in reverse order
257 for (size_t i = _poll_fds.size(); i-- > 0;) {
258 try {
259 short revents = _poll_fds[i].revents; // events that fired
260 int current_fd = _poll_fds[i].fd;
261
262 // Check if this client has hit a timeout (idle/header/body/send)
263 if (checkClientTimeouts(current_fd, i)) {
264 // Mark for POLLOUT → we will close after sending error/timeout response
265 _poll_fds[i].events |= POLLOUT;
266 }
267
268 // Skip CGI pipe descriptors entirely — they’re handled in handleCgiPollEvents()
269 if (_fd_to_cgi.contains(current_fd)) {
270 continue;
271 }
272
273 // Handle socket-level errors or hangups
274 if (revents & POLLERR || revents & POLLHUP || revents & POLLNVAL) {
275 handlePollError(current_fd, i, revents);
276 continue;
277 }
278
279 // Incoming data available
280 if (revents & POLLIN) {
281 if (_listen_map.count(current_fd)) {
282 // This is a listening socket → accept new connection
283 handleNewConnection(current_fd);
284 } else {
285 // Existing client → read request
286 if (!handleClientData(current_fd, i))
287 continue; // client closed or error, skip further handling
288 // At least one response is now queued → request POLLOUT
289 _poll_fds[i].events |= POLLOUT;
290 }
291 }
292
293 // Ready to write response
294 if ((revents & POLLOUT) && !_client_info[current_fd].responses.empty()) {
295 sendResponse(current_fd, i);
296 }
297 } catch (const std::exception& e) {
298 // Per-FD error resilience: log and cleanup
299 Logger::logFrom(LogLevel::ERROR, "SocketManager",
300 "Exception in poll loop for fd: " +
301 std::to_string(_poll_fds[i].fd) + ": " + e.what());
302 cleanupClientConnectionClose(_poll_fds[i].fd, i);
303 } catch (...) {
304 Logger::logFrom(LogLevel::ERROR, "SocketManager",
305 "Unknown exception in poll loop for fd: " +
306 std::to_string(_poll_fds[i].fd));
307 cleanupClientConnectionClose(_poll_fds[i].fd, i);
308 }
309 }
310 }
311
312 Logger::logFrom(LogLevel::INFO, "SocketManager", "Shutting down server");
313}

352/**
353 * @brief Accepts and registers a new client connection.
354 *
355 * @details
356 * Called when a listening socket signals readiness (`POLLIN`):
357 * - Calls `accept()` to retrieve a new client socket FD.
358 * - Handles resource exhaustion gracefully:
359 * - `EMFILE` / `ENFILE`: process/system FD limit reached → log and drop.
360 * - `MAX_CLIENTS` limit: too many clients → reject and close FD.
361 * - Initializes the new client state via @ref initializeClientInfo.
362 * - Adds the client FD to the poll set, monitoring for read events.
363 *
364 * @param listen_fd The listening socket FD on which the connection was accepted.
365 */
366void SocketManager::handleNewConnection(int listen_fd) {
367 // Accept a pending connection on the listening socket
368 int client_fd = accept(listen_fd, NULL, NULL);
369 if (client_fd < 0) {
370 if (errno == EMFILE || errno == ENFILE) {
371 // System or process has run out of file descriptors
372 Logger::logFrom(
373 LogLevel::ERROR, "SocketManager",
374 "Out of file descriptors (accept failed: " + std::string(strerror(errno)) + ")");
375 return;
376 }
377 // Generic accept() failure
378 Logger::logFrom(LogLevel::ERROR, "SocketManager",
379 "accept() failed: " + std::string(std::strerror(errno)));
380 return;
381 }
382
383 // Enforce application-level connection limit
384 if (_poll_fds.size() >= MAX_CLIENTS) {
385 Logger::logFrom(LogLevel::WARN, "SocketManager",
386 "Maximum client limit reached. Rejecting connection.");
387 close(client_fd); // Could optionally send an HTTP 503 before closing
388 return;
389 }
390
391 // Initialize per-client state tracking
392 initializeClientInfo(client_fd, listen_fd);
393
394 // Register client FD into poll list, listening for incoming data
395 _poll_fds.push_back((pollfd){client_fd, POLLIN, 0});
396
397 // Debug log: report accepted client and total open connections
398 Logger::logFrom(LogLevel::kDEBUG, "SocketManager",
399 "Accept returned fd: " + std::to_string(client_fd) +
400 " | current open clients: " + std::to_string(_client_info.size()));
401}

403/**
404 * @brief Processes incoming data for a client connection.
405 *
406 * @details
407 * Called when a client socket is readable (`POLLIN`):
408 * 1. Reads available data from the client using @ref receiveFromClient.
409 * - If the client disconnects or a read error occurs → return `false` (FD will be closed).
410 * 2. Attempts to parse one or more complete HTTP requests from the buffer
411 * via @ref parseAndQueueRequests.
412 * - If parsing is incomplete, waits for more data.
413 * - If a parse error occurs, queues an error response.
414 * 3. Processes all queued requests (except those blocked by CGI execution)
415 * via @ref processPendingRequests.
416 *
417 * Exception safety:
418 * - Catches `std::bad_alloc` (memory exhaustion), `std::exception`, and unknown errors.
419 * - On error: disables further POLLIN events for this client and queues a `500 Internal Server
420 * Error`.
421 *
422 * @param client_fd File descriptor of the client socket.
423 * @param index Index of the client FD in the internal poll list.
424 *
425 * @return `true` if processing completed (even with errors queued),
426 * `false` if the connection should be closed.
427 */
428bool SocketManager::handleClientData(int client_fd, size_t index) {
429 try {
430 // 1) Receive raw bytes from the client socket
431 if (!receiveFromClient(client_fd, index))
432 return false; // Client disconnected or fatal recv() error
433
434 // 2) Parse requests and queue them for processing
435 if (!parseAndQueueRequests(client_fd))
436 return false; // Incomplete request → wait for more data
437
438 // 3) Execute queued requests (routing, CGI, response building)
439 processPendingRequests(client_fd);
440
441 return true;
442 } catch (const std::bad_alloc& e) {
443 // Out-of-memory while processing this client
444 Logger::logFrom(LogLevel::ERROR, "SocketManager",
445 "Memory allocation failed while handling client " +
446 std::to_string(client_fd) + ": " + e.what());
447 } catch (const std::exception& e) {
448 // Generic runtime error
449 Logger::logFrom(LogLevel::ERROR, "SocketManager",
450 "Exception while handling client " + std::to_string(client_fd) + ": " +
451 e.what());
452 } catch (...) {
453 // Any other unknown error
454 Logger::logFrom(LogLevel::ERROR, "SocketManager",
455 "Unknown exception while handling client " + std::to_string(client_fd));
456 }
457
458 // Disable further reads from this client (avoid repeated errors)
459 _poll_fds[index].events &= ~POLLIN;
460
461 // Queue an internal server error response
462 respondError(client_fd, 500);
463
464 return true; // Keep connection alive long enough to send error response
465}
