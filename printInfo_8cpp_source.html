46void printConfig(Config& config) {
47 const std::vector<Server>& servers = config.getServers();
48 for (size_t i = 0; i < servers.size(); ++i) {
49 const Server& server = servers[i];
50 printServerHeader(i, server);
51
52 std::cout << " ▸ client_max_body_size : " << formatBytes(server.getClientMaxBodySize())
53 << "\n";
54
55 const auto& errors = server.getErrorPages();
56 if (!errors.empty()) {
57 std::cout << " ▸ error_pages:\n";
58 for (std::map<int, std::string>::const_iterator it = errors.begin(); it != errors.end();
59 ++it) {
60 std::cout << " - " << it->first << " → " << it->second << "\n";
61 }
62 }
63
64 const std::vector<Location>& locations = server.getLocations();
65 for (size_t k = 0; k < locations.size(); ++k) {
66 const Location& loc = locations[k];
67
68 std::cout << " ▸ location " << loc.getPath() << ":\n";
69 std::cout << " - root : " << loc.getRoot() << "\n";
70 std::cout << " - index : " << loc.getIndex() << "\n";
71 std::cout << " - autoindex : " << (loc.isAutoindexEnabled() ? "on" : "off")
72 << "\n";
73
74 const std::set<std::string>& methods = loc.getMethods();
75 std::cout << " - methods : ";
76 if (methods.empty())
77 std::cout << "(none)";
78 else {
79 std::set<std::string>::const_iterator it = methods.begin();
80 std::cout << *it;
81 for (++it; it != methods.end(); ++it)
82 std::cout << ", " << *it;
83 }
84 std::cout << "\n";
85
86 if (loc.hasRedirect()) {
87 std::cout << " - redirect : " << loc.getRedirect();
88 if (loc.getReturnCode())
89 std::cout << " (code " << loc.getReturnCode() << ")";
90 std::cout << "\n";
91 }
92
93 if (loc.isUploadEnabled()) {
94 std::cout << " - upload_store: " << loc.getUploadStore() << "\n";
95 }
96
97 const std::vector<std::string>& cgi = loc.getCgiExtensions();
98 if (!cgi.empty()) {
99 std::cout << " - cgi_pass : ";
100 for (size_t ci = 0; ci < cgi.size(); ++ci) {
101 std::cout << cgi[ci];
102 if (ci + 1 < cgi.size())
103 std::cout << ", ";
104 }
105 std::cout << "\n";
106
107 std::cout << " - interpreter :\n";
108 bool found = false;
109 for (size_t ci = 0; ci < cgi.size(); ++ci) {
110 std::string interp = loc.getCgiInterpreter(cgi[ci]);
111 if (!interp.empty()) {
112 std::cout << " - " << cgi[ci] << " → " << interp << "\n";
113 found = true;
114 }
115 }
116 if (!found)
117 std::cout << " (none)\n";
118 }
119 }
120 }
