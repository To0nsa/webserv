13/**
14 * @file handleMultipartForm.cpp
15 * @brief Handles `multipart/form-data` POST uploads and persists the first file part to disk.
16 *
17 * @details
18 * Parses the request body using the `boundary` from the `Content-Type` header,
19 * scans parts until it finds a file part (`filename="..."`), trims the trailing
20 * CRLF from the part payload, enforces `client_max_body_size`, and writes the
21 * bytes to `<upload_dir>/<sanitized-filename>` (or a time‑based fallback).
22 *
23 * ### Flow
24 * 1. Extract `boundary` from `Content-Type`; `400` if missing.
25 * 2. `parseMultipartForm`: iterate parts, split headers/data, detect `filename=`,
26 * sanitize file name, trim payload CRLF, return name + bytes.
27 * 3. Enforce per‑part size ≤ `Server::getClientMaxBodySize()`; `413` otherwise.
28 * 4. Open destination file (binary) and write bytes atomically; `500` on I/O error.
29 * 5. Log outcome and return `201 Created` with a minimal HTML confirmation.
30 *
31 * ### Security & Robustness
32 * - **Filename sanitization**: strips unsafe characters to prevent path traversal.
33 * - **Size limits**: rejects oversized payloads to mitigate DoS.
34 * - **Strict delimiter matching**: avoids bleed between parts.
35 *
36 * ### Limitations
37 * - Parses the **first** file part only (ignores subsequent files/fields).
38 * - Operates in‑memory (no streaming/chunking); large files may increase RAM use.
39 *
40 * @note Consider extending to stream to a temp file, support multiple files/fields,
41 * and return JSON for API clients.
42 *
43 * @see HttpRequest, HttpResponse, ResponseBuilder, Logger, filesystemUtils
44 * @ingroup request_handler
45 */

62/**
63 * @brief Construct the opening delimiter for a multipart part.
64 *
65 * @param boundary Boundary token from the Content-Type header.
66 * @return A delimiter string in the form `"--<boundary>\r\n"`.
67 */
68inline std::string makePartDelimiter(const std::string& boundary) {
69 return "--" + boundary + "\r\n";
70}

72/**
73 * @brief Construct the closing delimiter for a multipart body.
74 *
75 * @param boundary Boundary token from the Content-Type header.
76 * @return A delimiter string in the form `"--<boundary>--"`.

78inline std::string makeCloseDelimiter(const std::string& boundary) {

82/**
83 * @brief Find the next occurrence of a delimiter in a multipart body.
84 *
85 * @param body Full request body.
86 * @param delimiter Delimiter string to search for.
87 * @param startPos Starting offset in the body.
88 * @param outPos Output parameter; set to the position of the delimiter if found.
89 * @return true if the delimiter was found, false otherwise.
90 */
91bool findNextDelimiter(const std::string& body, const std::string& delimiter, size_t startPos,
92 size_t& outPos) {
93 outPos = body.find(delimiter, startPos);
94 return outPos != std::string::npos;
95}

97/**
98 * @brief Check if the body contains the closing delimiter at the given position.
99 *
100 * @param body Full request body.
101 * @param pos Position to check.
102 * @param closeDelimiter Expected closing delimiter string.
103 * @return true if the body matches the closing delimiter at `pos`.
104 */
105bool isCloseDelimiter(const std::string& body, size_t pos, const std::string& closeDelimiter) {
106 return body.compare(pos, closeDelimiter.size(), closeDelimiter) == 0;

109/**
110 * @brief Split a raw part into headers and data sections.
111 *
112 * @param part Raw part (between delimiters).
113 * @param headersBlock Output; substring up to the first double CRLF.
114 * @param dataBlock Output; substring after the first double CRLF.
115 * @return true if headers and data were successfully split, false otherwise.
116 */
117bool parseHeadersAndData(const std::string& part, std::string& headersBlock,
118 std::string& dataBlock) {
119 size_t headerEnd = part.find("\r\n\r\n");
120 if (headerEnd == std::string::npos)
121 return false;
122
123 headersBlock = part.substr(0, headerEnd);
124 dataBlock = part.substr(headerEnd + 4); // skip past "\r\n\r\n"
125 return true;
126}

128/**
129 * @brief Extract and sanitize the filename from a part's headers.
130 *
131 * @param headersBlock Headers of the multipart part.
132 * @param outFilename Output; sanitized filename if found.
133 * @return true if a `filename="..."` token was found, false otherwise.
134 */
135bool extractFilename(const std::string& headersBlock, std::string& outFilename) {
136 const std::string token = "filename=\"";
137 auto pos = headersBlock.find(token);
138 if (pos == std::string::npos)
139 return false;
140
141 pos += token.size();
142 auto endQuote = headersBlock.find('"', pos);
143 if (endQuote == std::string::npos)
144 return false;
145
146 // 1. Extract the raw filename
147 std::string rawName = headersBlock.substr(pos, endQuote - pos);
148
149 // 2. Sanitize it before returning
150 outFilename = sanitizeFilename(rawName);
151 return true;
152}

154/**
155 * @brief Strip a trailing CRLF sequence from data if present.
156 *
157 * @param data Raw part data block.
158 * @return A copy of `data` without a trailing `"\r\n"`, if it existed.
159 */
160std::string trimEndingCRLF(const std::string& data) {
161 if (data.size() >= 2 && data[data.size() - 2] == '\r' && data[data.size() - 1] == '\n') {
162 return data.substr(0, data.size() - 2);
163 }
164 return data;
165}

167/**
168 * @brief Parse a multipart/form-data body and extract the first file part.
169 *
170 * @details
171 * Iterates over all parts in the request body, delimited by the given boundary.
172 * For each part:
173 * - Splits headers and data at the first double CRLF.
174 * - Checks for a `filename="..."` header.
175 * - If found, sanitizes the filename, trims trailing CRLF from the data block,
176 * and outputs both.
177 *
178 * Stops at the first file part encountered; ignores non-file parts.
179 *
180 * @param body Full HTTP request body (including all parts).
181 * @param boundary Boundary token from the Content-Type header.
182 * @param filename Output; sanitized filename from the file part.
183 * @param fileContent Output; raw file content (as bytes in a std::string).
184 *
185 * @return true if a file part was found and parsed successfully, false otherwise.
186 *
187 * @note Only the first file part is extracted; additional files are ignored.
188 * @warning Operates in memory; large uploads may impact RAM usage.
189 */
190bool parseMultipartForm(const std::string& body, const std::string& boundary, std::string& filename,
191 std::string& fileContent) {
192
193 const std::string partDelimiter = makePartDelimiter(boundary);
194 const std::string closeDelimiter = makeCloseDelimiter(boundary);
195
196 size_t curPos = 0;
197 while (true) {
198 // 1) Locate the start of the next part:
199 size_t partStart;
200 if (!findNextDelimiter(body, partDelimiter, curPos, partStart)) {
201 // No more "--boundary\r\n", so nothing to parse
202 return false;
203 }
204
205 // 2) Check if this is actually the closing delimiter:
206 if (isCloseDelimiter(body, partStart, closeDelimiter)) {
207 return false;
208 }
209
210 // Move past the delimiter header line:
211 size_t contentStart = partStart + partDelimiter.size();
212
213 // 3) Find end of this part (either next partDelimiter or closeDelimiter):
214 size_t nextPartPos = std::string::npos;
215 findNextDelimiter(body, partDelimiter, contentStart, nextPartPos);
216
217 size_t closePos = std::string::npos;
218 findNextDelimiter(body, closeDelimiter, contentStart, closePos);
219
220 size_t partEnd;
221 if (nextPartPos == std::string::npos && closePos == std::string::npos) {
222 // Malformed: no closing boundary for this part
223 return false;
224 } else if (nextPartPos == std::string::npos) {
225 partEnd = closePos;
226 } else if (closePos == std::string::npos) {
227 partEnd = nextPartPos;
228 } else {
229 partEnd = std::min(nextPartPos, closePos);
230 }
231
232 // 4) Extract the raw bytes of this part (excluding the delimiter itself):
233 std::string part = body.substr(contentStart, partEnd - contentStart);
234
235 // 5) Split into headers vs. data:
236 std::string headersBlock, dataBlock;
237 if (!parseHeadersAndData(part, headersBlock, dataBlock)) {
238 return false;
239 }
240
241 // 6) Look for a filename="…". If not found, skip to next part:
242 std::string extractedName;
243 if (!extractFilename(headersBlock, extractedName)) {
244 curPos = partEnd;
245 continue; // keep scanning subsequent parts
246 }
247
248 // 7) We found a file‐upload part. Trim trailing CRLF from data:
249 filename = std::move(extractedName);
250 fileContent = trimEndingCRLF(dataBlock);
251 return true;
252 }
253
254 // (unreachable)
255 return false;
256}

260/**
261 * @brief Handle a multipart/form-data POST request and persist the uploaded file.
262 *
263 * @details
264 * - Extracts the boundary from the `Content-Type` header.
265 * - Parses the multipart body to find the first file part.
266 * - Enforces `client_max_body_size` to prevent DoS.
267 * - Generates a fallback filename if none was provided.
268 * - Writes the file content to disk under the given upload directory.
269 * - Returns an appropriate `HttpResponse`:
270 * - 400 if malformed,
271 * - 413 if too large,
272 * - 500 if I/O fails,
273 * - 201 with a confirmation HTML page on success.
274 *
275 * @param request Incoming HTTP request containing the multipart body.
276 * @param server Server configuration (used for limits and error pages).
277 * @param fullDirPath Absolute path of the directory where uploads are stored.
278 *
279 * @return HttpResponse representing the outcome (success or error).
280 *
281 * @ingroup request_handler
282 */
283HttpResponse handleMultipartForm(const HttpRequest& request, const Server& server,
284 const std::string& fullDirPath) {
285 // 1) Extract boundary from Content-Type header
286 std::string contentType = request.getHeader("Content-Type");
287 size_t bpos = contentType.find("boundary=");
288 if (bpos == std::string::npos) {
289 Logger::logFrom(LogLevel::WARN, "handleMultipartForm", "Missing boundary in Content-Type");
290 return ResponseBuilder::generateError(400, server, request);
291 }
292 std::string boundary = contentType.substr(bpos + 9);
293
294 // 2) Parse body to extract filename + file content
295 std::string extractedFilename;
296 std::string fileContent;
297 if (!parseMultipartForm(request.getBody(), boundary, extractedFilename, fileContent)) {
298 Logger::logFrom(LogLevel::WARN, "handleMultipartForm", "Failed to parse multipart form");
299 return ResponseBuilder::generateError(400, server, request);
300 }
301
302 // 3) Enforce per-part size limit
303 size_t maxBody = server.getClientMaxBodySize();
304 if (fileContent.size() > maxBody) {
305 Logger::logFrom(LogLevel::WARN, "handleMultipartForm",
306 "Uploaded part size " + std::to_string(fileContent.size()) +
307 " exceeds max client body size " + std::to_string(maxBody));
308 return ResponseBuilder::generateError(413, server, request);
309 }
310
311 // 4) Fallback filename if client didn’t provide one
312 if (extractedFilename.empty()) {
313 extractedFilename = "upload_" + std::to_string(getCurrentTime());
314 }
315
316 // 5) Compute destination path and open file
317 std::string fullpath = joinPath(fullDirPath, extractedFilename);
318 std::ofstream out(fullpath, std::ios::binary);
319 if (!out.is_open()) {
320 Logger::logFrom(LogLevel::ERROR, "handleMultipartForm",
321 "Failed to open file for writing: " + fullpath);
322 return ResponseBuilder::generateError(500, server, request);
323 }
324
325 // 6) Write content and verify
326 out << fileContent;
327 out.close();
328 if (out.fail()) {
329 Logger::logFrom(LogLevel::ERROR, "handleMultipartForm",
330 "Failed to write file content to: " + fullpath);
331 return ResponseBuilder::generateError(500, server, request);
332 }
333
334 // 7) Success → return confirmation page
335 Logger::logFrom(LogLevel::INFO, "handleMultipartForm", "Successfully uploaded: " + fullpath);
336 return ResponseBuilder::generateSuccess(
337 201, "<html><body><h1>Uploaded: " + htmlEscape(extractedFilename) + "</h1></body></html>",
338 "text/html", request);
339}
