40/**
41 * Find the next occurrence of `delimiter` in `body` at or after `startPos`.
42 * Returns true if found (and sets outPos), false otherwise.

44bool findNextDelimiter(const std::string& body, const std::string& delimiter, size_t startPos,
45 size_t& outPos) {
46 outPos = body.find(delimiter, startPos);

50/**
51 * Check if at position `pos` in `body` there is a closing boundary.

53bool isCloseDelimiter(const std::string& body, size_t pos, const std::string& closeDelimiter) {

57/**
58 * Given a single “part” (everything between delimiters), split it into
59 * headersBlock (before the first "\r\n\r\n") and dataBlock (after it).
60 * Returns false if no double‐CRLF is found.
61 */
62bool parseHeadersAndData(const std::string& part, std::string& headersBlock,
63 std::string& dataBlock) {
64 size_t headerEnd = part.find("\r\n\r\n");
65 if (headerEnd == std::string::npos)
66 return false;
67
68 headersBlock = part.substr(0, headerEnd);
69 dataBlock = part.substr(headerEnd + 4); // skip past "\r\n\r\n"
70 return true;
71}

73/**
74 * Look for filename="..." inside headersBlock. If found, extract the filename
75 * (without the quotes) into `outFilename` and return true. Otherwise return false.
76 */
77bool extractFilename(const std::string& headersBlock, std::string& outFilename) {
78 const std::string token = "filename=\"";
79 auto pos = headersBlock.find(token);
80 if (pos == std::string::npos)
81 return false;
82
83 pos += token.size();
84 auto endQuote = headersBlock.find('"', pos);
85 if (endQuote == std::string::npos)
86 return false;
87
88 // 1. Extract the raw filename
89 std::string rawName = headersBlock.substr(pos, endQuote - pos);
90
91 // 2. Sanitize it before returning
92 outFilename = sanitizeFilename(rawName);
93 return true;
94}

96/**
97 * If `data` ends with "\r\n", strip those two characters off. Otherwise return as-is.
98 */
99std::string trimEndingCRLF(const std::string& data) {
100 if (data.size() >= 2 && data[data.size() - 2] == '\r' && data[data.size() - 1] == '\n') {
101 return data.substr(0, data.size() - 2);
102 }
103 return data;
104}

106/**
107 * Parse a multipart/form-data body and extract the first file‐part’s
108 * filename and its raw content. Returns true on success, false otherwise.
109 *
110 * - body: full HTTP request body (including all parts)
111 * - boundary: the “boundary” token from Content-Type, e.g. "----WebKitFormBoundary…"
112 * - filename: output parameter; set to the file’s original name
113 * - fileContent: output parameter; set to the file’s bytes (as a std::string)
114 *
115 * This will scan part by part until it finds a “filename=” header. If none
116 * is found, it ultimately returns false.
117 */
118bool parseMultipartForm(const std::string& body, const std::string& boundary, std::string& filename,
119 std::string& fileContent) {
120
121 const std::string partDelimiter = makePartDelimiter(boundary);
122 const std::string closeDelimiter = makeCloseDelimiter(boundary);
123
124 size_t curPos = 0;
125 while (true) {
126 // 1) Locate the start of the next part:
127 size_t partStart;
128 if (!findNextDelimiter(body, partDelimiter, curPos, partStart)) {
129 // No more "--boundary\r\n", so nothing to parse
130 return false;
131 }
132
133 // 2) Check if this is actually the closing delimiter:
134 if (isCloseDelimiter(body, partStart, closeDelimiter)) {
135 return false;
136 }
137
138 // Move past the delimiter header line:
139 size_t contentStart = partStart + partDelimiter.size();
140
141 // 3) Find end of this part (either next partDelimiter or closeDelimiter):
142 size_t nextPartPos = std::string::npos;
143 findNextDelimiter(body, partDelimiter, contentStart, nextPartPos);
144
145 size_t closePos = std::string::npos;
146 findNextDelimiter(body, closeDelimiter, contentStart, closePos);
147
148 size_t partEnd;
149 if (nextPartPos == std::string::npos && closePos == std::string::npos) {
150 // Malformed: no closing boundary for this part
151 return false;
152 } else if (nextPartPos == std::string::npos) {
153 partEnd = closePos;
154 } else if (closePos == std::string::npos) {
155 partEnd = nextPartPos;
156 } else {
157 partEnd = std::min(nextPartPos, closePos);
158 }
159
160 // 4) Extract the raw bytes of this part (excluding the delimiter itself):
161 std::string part = body.substr(contentStart, partEnd - contentStart);
162
163 // 5) Split into headers vs. data:
164 std::string headersBlock, dataBlock;
165 if (!parseHeadersAndData(part, headersBlock, dataBlock)) {
166 return false;
167 }
168
169 // 6) Look for a filename="…". If not found, skip to next part:
170 std::string extractedName;
171 if (!extractFilename(headersBlock, extractedName)) {
172 curPos = partEnd;
173 continue; // keep scanning subsequent parts
174 }
175
176 // 7) We found a file‐upload part. Trim trailing CRLF from data:
177 filename = std::move(extractedName);
178 fileContent = trimEndingCRLF(dataBlock);
179 return true;
180 }
181
182 // (unreachable)
183 return false;
184}

188HttpResponse handleMultipartForm(const HttpRequest& request, const Server& server,
189 const std::string& fullDirPath) {
190 std::string contentType = request.getHeader("Content-Type");
191 size_t bpos = contentType.find("boundary=");
192 if (bpos == std::string::npos) {
193 Logger::logFrom(LogLevel::WARN, "handleMultipartForm", "Missing boundary in Content-Type");
194 return ResponseBuilder::generateError(400, server, request);
195 }
196 std::string boundary = contentType.substr(bpos + 9);
197
198 std::string extractedFilename;
199 std::string fileContent;
200 if (!parseMultipartForm(request.getBody(), boundary, extractedFilename, fileContent)) {
201 Logger::logFrom(LogLevel::WARN, "handleMultipartForm", "Failed to parse multipart form");
202 return ResponseBuilder::generateError(400, server, request);
203 }
204
205 // Enforce per-part size limit to avoid DoS
206 size_t maxBody = server.getClientMaxBodySize();
207 if (fileContent.size() > maxBody) {
208 Logger::logFrom(LogLevel::WARN, "handleMultipartForm",
209 "Uploaded part size " + std::to_string(fileContent.size()) +
210 " exceeds max client body size " + std::to_string(maxBody));
211 return ResponseBuilder::generateError(413, server, request);
212 }
213
214 if (extractedFilename.empty()) {
215 extractedFilename = "upload_" + std::to_string(getCurrentTime());
216 }
217
218 std::string fullpath = joinPath(fullDirPath, extractedFilename);
219
220 std::ofstream out(fullpath, std::ios::binary);
221 if (!out.is_open()) {
222 Logger::logFrom(LogLevel::ERROR, "handleMultipartForm",
223 "Failed to open file for writing: " + fullpath);
224 return ResponseBuilder::generateError(500, server, request);
225 }
226
227 out << fileContent;
228 out.close();
229 if (out.fail()) {
230 Logger::logFrom(LogLevel::ERROR, "handleMultipartForm",
231 "Failed to write file content to: " + fullpath);
232 return ResponseBuilder::generateError(500, server, request);
233 }
234
235 Logger::logFrom(LogLevel::INFO, "handleMultipartForm", "Successfully uploaded: " + fullpath);
236 return ResponseBuilder::generateSuccess(
237 201, "<html><body><h1>Uploaded: " + htmlEscape(extractedFilename) + "</h1></body></html>",
238 "text/html", request);
239}
