29std::optional<HttpResponse> redirectOnDirectorySlash(const HttpRequest& req, const Server& server,
30 const std::string& uri) {
31 // If there's already a trailing slash in the request path, nothing to do.
32 if (!req.getPath().empty() && req.getPath().back() == '/')
33 return std::nullopt;
34
35 // Try each configured Location block
36 for (const Location& loc : server.getLocations()) {
37 std::string locPath = normalizePath(loc.getPath());
38 // We only care about locations defined with a trailing slash
39 if (locPath.size() > 1 && locPath.back() == '/') {
40 // Does the request URI match this location prefix (sans slash)?
41 std::string prefix = locPath.substr(0, locPath.size() - 1);
42 if (uri.rfind(prefix, 0) != 0)
43 continue;
44
45 // Resolve the physical FS path under this Location
46 std::string fsPath = resolvePhysicalPath(req, loc);
47 if (fsPath.empty())
48 continue;
49
50 struct stat st;
51 if (stat(fsPath.c_str(), &st) == 0 && S_ISDIR(st.st_mode)) {
52 // Found a real directory (or symlink-to-dir) — redirect any method
53 std::string target = req.getPath() + "/";
54 Logger::logFrom(LogLevel::INFO, "Router",
55 "Directory-slash redirect: \"" + uri + "\" → \"" + target + "\"");
56 return ResponseBuilder::generateRedirect(301, target, req);
57 }
58 }
59 }
60
61 // No matching directory to slash-redirect
62 return std::nullopt;
63}

65// 2) Find exact‐match or longest‐prefix location
66const Location* findLocation(const std::string& uri, const Server& server) {
67 // Exact match
68 for (const Location& loc : server.getLocations()) {
69 std::string locPath = normalizePath(loc.getPath());
70 if (uri == locPath) {
71 return &loc;
72 }
73 }
74 // Longest‐prefix
75 const Location* best = nullptr;
76 std::size_t bestLen = 0;
77 for (const Location& loc : server.getLocations()) {
78 std::string locPath = normalizePath(loc.getPath());
79 if (locPath.empty()) {
80 continue;
81 }
82 if (uri.rfind(locPath, 0) == 0 && locPath.size() > bestLen) {
83 best = &loc;
84 bestLen = locPath.size();
85 }
86 }
87 return best;
88}

101// 4) Check for un-implemented / not-allowed methods
102std::optional<HttpResponse> validateRequestMethod(const HttpRequest& request, const Location& loc,
103 const Server& server) {
104 static const std::set<std::string> supported = {"GET", "POST", "DELETE"};
105 const std::string& method = request.getMethod();
106
107 if (supported.count(method) == 0) {
108 Logger::logFrom(LogLevel::WARN, "Router",
109 "Method \"" + method + "\" not implemented for URI \"" + request.getPath() +
110 "\"");
111 return ResponseBuilder::generateError(501, server, request);
112 }
113
114 if (!loc.isMethodAllowed(method)) {
115 Logger::logFrom(LogLevel::WARN, "Router",
116 "Method \"" + method + "\" not allowed on location \"" + loc.getPath() +
117 "\"");
118 return ResponseBuilder::generateError(405, server, request);
119 }
120
121 return std::nullopt;
122}

139HttpResponse handleRequest(const HttpRequest& request, const Server& server) {
140 std::string uri = normalizePath(request.getPath());
141
142 std::optional<HttpResponse> redirectResponse = redirectOnDirectorySlash(request, server, uri);
143 if (redirectResponse.has_value()) {
144 return redirectResponse.value();
145 }
146
147 const Location* loc = findLocation(uri, server);
148 if (loc == nullptr) {
149 Logger::logFrom(LogLevel::WARN, "Router",
150 "No matching location for URI \"" + uri + "\" → 404");
151 return ResponseBuilder::generateError(404, server, request);
152 }
153
154 std::optional<HttpResponse> configuredRedirect = redirectOnConfigured(request, *loc);
155 if (configuredRedirect.has_value()) {
156 return configuredRedirect.value();
157 }
158
159 std::optional<HttpResponse> validationError = validateRequestMethod(request, *loc, server);
160 if (validationError.has_value()) {
161 return validationError.value();
162 }
163
164 return dispatchByMethod(request, server, *loc);
165}
