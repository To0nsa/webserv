13/**
14 * @file requestRouter.cpp
15 * @brief Routes parsed HTTP requests to the appropriate handler.
16 *
17 * @details Implements the routing pipeline that maps an incoming
18 * @ref HttpRequest to a @ref HttpResponse using the active
19 * @ref Server configuration and its @ref Location blocks.
20 *
21 * Workflow:
22 * 1) **Directory slash redirect:** If a request targets a physical
23 * directory without a trailing slash, issue a 301 to the
24 * slash-terminated URI (see @ref redirectOnDirectorySlash).
25 * 2) **Location resolution:** Find the best matching @ref Location by
26 * exact match or longest-prefix (see @ref findLocation).
27 * 3) **Configured redirects:** Apply `return`-based redirects if present
28 * in the matched location (see @ref redirectOnConfigured).
29 * 4) **Method validation:** Check supported/allowed methods, returning
30 * 501 or 405 as appropriate (see @ref validateRequestMethod).
31 * 5) **Dispatch:** Call the concrete method handler
32 * (GET/POST/DELETE) (see @ref dispatchByMethod).
33 *
34 * Errors are rendered via @ref ResponseBuilder helpers. Utilities from
35 * `filesystemUtils` are used to normalize URIs and resolve physical paths.
36 *
37 * @ingroup request_handler
38 */

56/**
57 * @brief Redirects to a slash-terminated URI when the target is a real directory.
58 *
59 * @details This helper enforces nginx-like behavior for directory URIs:
60 * if the request path maps to a directory but lacks a trailing slash,
61 * issue a 301 redirect to the slash-appended path.
62 *
63 * Rules:
64 * - Only considers @ref Location blocks whose configured path ends with '/'.
65 * - Matches the request URI by location prefix (sans trailing slash).
66 * - Resolves a physical filesystem path with `resolvePhysicalPath` and
67 * checks it with `stat`; symlink-to-dir is treated as a directory.
68 * - Applies to any HTTP method (GET/POST/DELETE), not only GET.
69 *
70 * @param req Parsed HTTP request (used for path, method, headers).
71 * @param server Active server context (provides location list).
72 * @param uri Normalized request URI (usually `normalizePath(req.getPath())`).
73 *
74 * @return std::nullopt if no redirect is needed; otherwise a ready 301 response
75 * with `Location: <path>/`.
76 *
77 * @note If `req.getPath()` already ends with '/', no redirect occurs.
78 * @warning Only locations defined with a trailing slash are inspected; locations
79 * configured without a trailing '/' are intentionally ignored here.
80 */
81std::optional<HttpResponse> redirectOnDirectorySlash(const HttpRequest& req, const Server& server,
82 const std::string& uri) {
83 // If there's already a trailing slash in the request path, nothing to do.
84 if (!req.getPath().empty() && req.getPath().back() == '/')
85 return std::nullopt;
86
87 // Try each configured Location block
88 for (const Location& loc : server.getLocations()) {
89 std::string locPath = normalizePath(loc.getPath());
90 // We only care about locations defined with a trailing slash
91 if (locPath.size() > 1 && locPath.back() == '/') {
92 // Does the request URI match this location prefix (sans slash)?
93 std::string prefix = locPath.substr(0, locPath.size() - 1);
94 if (uri.rfind(prefix, 0) != 0)
95 continue;
96
97 // Resolve the physical FS path under this Location
98 std::string fsPath = resolvePhysicalPath(req, loc);
99 if (fsPath.empty())
100 continue;
101
102 struct stat st;
103 if (stat(fsPath.c_str(), &st) == 0 && S_ISDIR(st.st_mode)) {
104 // Found a real directory (or symlink-to-dir) — redirect any method
105 std::string target = req.getPath() + "/";
106 Logger::logFrom(LogLevel::INFO, "Router",
107 "Directory-slash redirect: \"" + uri + "\" → \"" + target + "\"");
108 return ResponseBuilder::generateRedirect(301, target, req);
109 }
110 }
111 }
112
113 // No matching directory to slash-redirect
114 return std::nullopt;
115}

117/**
118 * @brief Selects the best-matching Location for a normalized request URI.
119 *
120 * @details Resolution is two-phase:
121 * 1) **Exact match**: return the first location whose normalized path
122 * equals `uri`.
123 * 2) **Longest-prefix match**: otherwise, return the location with the
124 * longest normalized path that is a prefix of `uri`.
125 *
126 * Each candidate path is normalized with `normalizePath`. Empty
127 * location paths are ignored for prefix matching.
128 *
129 * @param uri Normalized request URI (e.g., from `normalizePath(req.getPath())`).
130 * @param server Active server whose location set is consulted.
131 *
132 * @return Pointer to the selected @ref Location, or `nullptr` if none match.
133 *
134 * @note Matching is purely lexical (prefix test), not filesystem-based.
135 * @warning If multiple locations normalize to the same path, the first one
136 * encountered wins (implementation-defined by iteration order).
137 * @complexity O(N · C) where N = number of locations and C = normalization + prefix check cost.
138 */
139const Location* findLocation(const std::string& uri, const Server& server) {
140 // Exact match
141 for (const Location& loc : server.getLocations()) {
142 std::string locPath = normalizePath(loc.getPath());
143 if (uri == locPath) {
144 return &loc;
145 }
146 }
147 // Longest‐prefix
148 const Location* best = nullptr;
149 std::size_t bestLen = 0;
150 for (const Location& loc : server.getLocations()) {
151 std::string locPath = normalizePath(loc.getPath());
152 if (locPath.empty()) {
153 continue;
154 }
155 if (uri.rfind(locPath, 0) == 0 && locPath.size() > bestLen) {
156 best = &loc;
157 bestLen = locPath.size();
158 }
159 }
160 return best;
161}

163/**
164 * @brief Apply a location-scoped redirect, if configured.
165 *
166 * @details If the matched @ref Location declares a redirect (i.e. a `return` rule),
167 * this function builds and returns a redirect response using
168 * @ref ResponseBuilder::generateRedirect. The redirect is applied
169 * unconditionally for any HTTP method and happens before method
170 * validation/dispatch in the routing pipeline.
171 *
172 * @param request Incoming HTTP request (used to propagate version/connection metadata).
173 * @param loc The matched location to inspect for a configured redirect.
174 *
175 * @return A populated @ref HttpResponse when a redirect rule is present;
176 * `std::nullopt` otherwise.
177 *
178 * @note The target URL is taken verbatim from configuration; no normalization is
179 * performed here. Ensure the configured target is either an absolute URI or a
180 * valid absolute-path per your configuration semantics.
181 *
182 * @warning Choose an appropriate status code (e.g., 301/302/307/308) in the
183 * configuration to control method preservation and cacheability semantics.
184 */
185std::optional<HttpResponse> redirectOnConfigured(const HttpRequest& request, const Location& loc) {
186 if (loc.hasRedirect()) {
187 Logger::logFrom(LogLevel::INFO, "Router",
188 "Configured redirect for URI \"" + request.getPath() + "\" to \"" +
189 loc.getRedirect() + "\"");
190 return ResponseBuilder::generateRedirect(loc.getReturnCode(), loc.getRedirect(), request);
191 }
192 return std::nullopt;
193}

195/**
196 * @brief Validates the HTTP method against server support and location policy.
197 *
198 * @details Two checks in order:
199 * 1) **Server support**: rejects methods not implemented by this server
200 * (currently {"GET","POST","DELETE"}) with **501 Not Implemented**.
201 * 2) **Location policy**: rejects methods not allowed by the matched
202 * @ref Location with **405 Method Not Allowed**.
203 *
204 * Both cases are logged and rendered via @ref ResponseBuilder::generateError.
205 *
206 * @param request Incoming HTTP request (used for method/path and response metadata).
207 * @param loc Matched configuration block to query method allowance.
208 * @param server Active server (used by error rendering).
209 *
210 * @return `std::nullopt` when the method is valid for this location; otherwise a
211 * ready error @ref HttpResponse (501 or 405).
212 *
213 * @note Per RFC 9110, servers **SHOULD** include an `Allow` header on 405
214 * listing permitted methods for the target resource.
215 * @warning Method matching here is case-sensitive; normalize upstream if you
216 * want to accept non‑uppercase tokens.
217 */
218std::optional<HttpResponse> validateRequestMethod(const HttpRequest& request, const Location& loc,
219 const Server& server) {
220 static const std::set<std::string> supported = {"GET", "POST", "DELETE"};
221 const std::string& method = request.getMethod();
222
223 if (supported.count(method) == 0) {
224 Logger::logFrom(LogLevel::WARN, "Router",
225 "Method \"" + method + "\" not implemented for URI \"" + request.getPath() +
226 "\"");
227 return ResponseBuilder::generateError(501, server, request);
228 }
229
230 if (!loc.isMethodAllowed(method)) {
231 Logger::logFrom(LogLevel::WARN, "Router",
232 "Method \"" + method + "\" not allowed on location \"" + loc.getPath() +
233 "\"");
234 return ResponseBuilder::generateError(405, server, request);
235 }
236
237 return std::nullopt;
238}

240/**
241 * @brief Dispatches a validated request to the concrete method handler.
242 *
243 * @details Assumes @ref validateRequestMethod has already run, so only
244 * implemented-and-allowed methods reach this point. Dispatch is a
245 * simple chain on the request method:
246 * - "GET" → @ref handleGet
247 * - "POST" → @ref handlePost
248 * - "DELETE" → @ref handleDelete
249 *
250 * @param request Parsed HTTP request.
251 * @param server Active server context.
252 * @param loc Matched @ref Location.
253 *
254 * @return The handler-produced @ref HttpResponse.
255 *
256 * @note If additional methods are added in the future, extend this dispatcher
257 * (or switch to a table-driven approach).
258 */
259HttpResponse dispatchByMethod(const HttpRequest& request, const Server& server,
260 const Location& loc) {
261 const std::string& method = request.getMethod();
262 if (method == "GET") {
263 return handleGet(request, server, loc);
264 }
265 if (method == "POST") {
266 return handlePost(request, server, loc);
267 }
268 // Only DELETE remains
269 return handleDelete(request, server, loc);
270}

273/**
274 * @brief Top-level router entrypoint for processing an HTTP request.
275 *
276 * @details This function implements the main request handling pipeline:
277 * 1. **Normalize URI** — sanitize the request path.
278 * 2. **Directory-slash redirect** — if a directory is requested without a trailing
279 * slash, generate a `301` redirect (see @ref redirectOnDirectorySlash).
280 * 3. **Location resolution** — find the best-matching @ref Location in the
281 * server configuration (exact or longest-prefix match).
282 * 4. **Configured redirect** — apply `return` directive–based redirects if present
283 * in the matched location (see @ref redirectOnConfigured).
284 * 5. **Method validation** — reject unsupported (501) or disallowed (405) methods
285 * (see @ref validateRequestMethod).
286 * 6. **Dispatch** — call the concrete handler for GET/POST/DELETE
287 * (see @ref dispatchByMethod).
288 *
289 * Errors at any stage are logged and turned into responses via @ref ResponseBuilder.
290 *
291 * @param request Parsed @ref HttpRequest.
292 * @param server Active @ref Server context.
293 * @return Fully constructed @ref HttpResponse to send back to the client.
294 *
295 * @ingroup request_handler
296 */
297HttpResponse handleRequest(const HttpRequest& request, const Server& server) {
298 // 1) Normalize the request path
299 std::string uri = normalizePath(request.getPath());
300
301 // 2) Check if this is a directory missing a trailing slash → 301 redirect
302 std::optional<HttpResponse> redirectResponse = redirectOnDirectorySlash(request, server, uri);
303 if (redirectResponse.has_value()) {
304 return redirectResponse.value();
305 }
306
307 // 3) Find best matching Location block
308 const Location* loc = findLocation(uri, server);
309 if (loc == nullptr) {
310 Logger::logFrom(LogLevel::WARN, "Router",
311 "No matching location for URI \"" + uri + "\" → 404");
312 return ResponseBuilder::generateError(404, server, request);
313 }
314
315 // 4) Check for a configured `return` redirect on this location
316 std::optional<HttpResponse> configuredRedirect = redirectOnConfigured(request, *loc);
317 if (configuredRedirect.has_value()) {
318 return configuredRedirect.value();
319 }
320
321 // 5) Validate that the method is implemented and allowed in this location
322 std::optional<HttpResponse> validationError = validateRequestMethod(request, *loc, server);
323 if (validationError.has_value()) {
324 return validationError.value();
325 }
326
327 // 6) Finally, dispatch to the actual handler (GET/POST/DELETE)
328 return dispatchByMethod(request, server, *loc);
