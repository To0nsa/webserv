13/**
14 * @file SocketManagerTimeouts.cpp
15 * @brief Implements timeout and limit enforcement for SocketManager.
16 *
17 * @details
18 * This file contains helper routines used by the poll-driven event loop to
19 * detect stalled or abusive connections and enforce request limits:
20 * - @ref isHeaderTimeout: header phase exceeded @c HEADER_TIMEOUT_SECONDS.
21 * - @ref isBodyTimeout: body phase exceeded @c TIMEOUT.
22 * - @ref isSendTimeout: no progress while sending for longer than @c TIMEOUT.
23 * - @ref isIdleTimeout: idle connection (no header bytes) exceeded @c TIMEOUT.
24 * - @ref checkClientTimeouts: orchestrates per-FD timeout checks and applies
25 * cleanup / event-mask adjustments.
26 * - @ref checkRequestLimits: enforces header length cap (@c HEADER_MAX_LENGTH)
27 * and queues a 431 if exceeded.
28 *
29 * Responsibilities:
30 * - Decide when to close connections vs. temporarily disable POLLIN.
31 * - Queue appropriate HTTP error responses (e.g., 408, 431).
32 * - Avoid interfering with active CGI executions.
33 *
34 * These functions are invoked from the main event loop to keep the server
35 * responsive, fair, and resilient under slowloris-style behavior or network stalls.
36 *
37 * @ingroup socket_manager
38 */

53/**
54 * @brief Detects header read timeout for a client connection.
55 *
56 * @details
57 * Triggers when:
58 * - No response is currently being sent (`responses` and `current_raw_response` empty),
59 * - The HTTP headers are not yet complete (`headerComplete == false`) but at least
60 * one header byte has arrived (`headerBytesReceived > 0`),
61 * - And the elapsed time since `connectionStartTime` exceeds
62 * `HEADER_TIMEOUT_SECONDS`.
63 *
64 * On timeout, queues a `408 Request Timeout` via @ref respondError and returns `true`
65 * so the caller can disable further reads (POLLIN) and flush the error response.
66 *
67 * @param fd Client socket file descriptor.
68 * @param now Current time snapshot used for comparison.
69 * @return `true` if a header timeout was detected and an error response was queued,
70 * otherwise `false`.
71 *
72 * @note This does not close the connection immediately; the caller is expected to
73 * keep the socket writable to send the queued 408 response.
74 */
75bool SocketManager::isHeaderTimeout(int fd, time_t now) {
76 ClientInfo& client = _client_info[fd];
77
78 // No ongoing send, header not complete, some header bytes seen,
79 // and the header phase exceeded the configured timeout.
80 if (client.responses.empty() && client.current_raw_response.empty() && !client.headerComplete &&
81 client.headerBytesReceived > 0 &&
82 now - client.connectionStartTime > HEADER_TIMEOUT_SECONDS) {
83
84 Logger::logFrom(LogLevel::WARN, "SocketManager", "Timeout on fd: " + std::to_string(fd));
85
86 // Queue 408 so the writer path can flush it to the client.
87 respondError(fd, 408);
88 return true;
89 }
90
91 return false;
92}

94/**
95 * @brief Detects request body timeout for a client connection.
96 *
97 * @details
98 * This timeout applies when:
99 * - No response is currently being sent (`responses` and `current_raw_response` empty),
100 * - The HTTP headers have already been received completely (`headerComplete == true`),
101 * - And the elapsed time since `connectionStartTime` exceeds @c TIMEOUT.
102 *
103 * This typically indicates that the client started a request but did not finish
104 * sending the body in time (e.g. stalled upload or slowloris-style attack).
105 *
106 * On timeout, a `408 Request Timeout` is queued via @ref respondError and the function
107 * returns `true`, so the caller may disable reads (`POLLIN`) and flush the error.
108 *
109 * @param fd Client socket file descriptor.
110 * @param now Current server time snapshot.
111 * @return `true` if a body timeout occurred and an error response was queued,
112 * otherwise `false`.
113 *
114 * @note The connection is not closed immediately; it is marked for sending the 408
115 * response before shutdown.
116 */
117bool SocketManager::isBodyTimeout(int fd, time_t now) {
118 ClientInfo& client = _client_info[fd];
119
120 // No ongoing send, header complete, and body phase exceeded TIMEOUT.
121 if (client.responses.empty() && client.current_raw_response.empty() && client.headerComplete &&
122 now - client.connectionStartTime > TIMEOUT) {
123
124 Logger::logFrom(LogLevel::WARN, "SocketManager", "Timeout on fd: " + std::to_string(fd));
125
126 respondError(fd, 408); // Queue HTTP 408 Request Timeout
127 return true;
128 }
129
130 return false;
131}

133/**
134 * @brief Detects send timeout for a client connection.
135 *
136 * @details
137 * This timeout applies when:
138 * - There are still responses waiting to be sent (`!responses.empty()`),
139 * - A response is actively being transmitted (`!current_raw_response.empty()`),
140 * - But no progress has been made for more than @c TIMEOUT seconds
141 * (based on `lastSendAttemptTime`).
142 *
143 * This usually indicates that the client has stopped reading from its socket
144 * (e.g., slow or dead client), preventing the server from completing the response.
145 *
146 * On timeout, no error response is queued (since we are already mid-send),
147 * but the function signals the caller (`true`) so that the connection can be
148 * closed and cleaned up.
149 *
150 * @param fd Client socket file descriptor.
151 * @param now Current server time snapshot.
152 * @return `true` if a send timeout occurred and the connection should be closed,
153 * otherwise `false`.
154 *
155 * @note Unlike header/body timeouts, this does not queue a 408 response,
156 * because the server was already in the process of sending a response.
157 */
158bool SocketManager::isSendTimeout(int fd, time_t now) {
159 ClientInfo& client = _client_info[fd];
160
161 // Active send in progress but stalled for longer than TIMEOUT.
162 if (!client.responses.empty() && !client.current_raw_response.empty() &&
163 now - client.lastSendAttemptTime > TIMEOUT) {
164
165 Logger::logFrom(LogLevel::WARN, "SocketManager",
166 "Send timeout on fd: " + std::to_string(fd));
167 return true;
168 }
169
170 return false;
171}

173/**
174 * @brief Detects idle timeout for a client connection.
175 *
176 * @details
177 * This timeout applies when:
178 * - The client is not running a CGI process (`cgiProcess` is empty),
179 * - No response is in progress (`responses` and `current_raw_response` are empty),
180 * - No headers have been fully received (`headerComplete == false`),
181 * - No header bytes have been received at all (`headerBytesReceived == 0`),
182 * - And the time since the last recorded client activity (`lastRequestTime`)
183 * exceeds @c TIMEOUT.
184 *
185 * This typically indicates an idle TCP connection where the client opened
186 * a socket but never sent any meaningful data.
187 *
188 * On timeout, no explicit error response is queued; the caller is expected to
189 * close the connection after receiving `true`.
190 *
191 * @param fd Client socket file descriptor.
192 * @param now Current server time snapshot.
193 * @return `true` if the connection is considered idle and should be closed,
194 * otherwise `false`.
195 *
196 * @note CGI processes are exempted from idle timeout handling because they may
197 * take significant time before producing output.
198 */
199bool SocketManager::isIdleTimeout(int fd, time_t now) {
200 ClientInfo& client = _client_info[fd];
201
202 // Skip idle timeout checks if a CGI is running.
203 if (client.cgiProcess.has_value()) {
204 return false;
205 }
206
207 // No response, no header progress, and idle for too long.
208 if (client.responses.empty() && client.current_raw_response.empty() && !client.headerComplete &&
209 client.headerBytesReceived == 0 && now - client.lastRequestTime > TIMEOUT) {
210
211 Logger::logFrom(LogLevel::WARN, "SocketManager",
212 "Idle timeout on fd: " + std::to_string(fd));
213 return true;
214 }
215
216 return false;
217}

219/**
220 * @brief Checks and applies timeout rules for a client connection.
221 *
222 * @details
223 * This function orchestrates all timeout checks for a client:
224 * - **Idle timeout**: connection opened but no activity (`isIdleTimeout`).
225 * - **Send timeout**: stalled during an ongoing response (`isSendTimeout`).
226 * - **Header timeout**: headers started but not completed in time (`isHeaderTimeout`).
227 * - **Body timeout**: headers done but body not finished within limit (`isBodyTimeout`).
228 *
229 * Behavior:
230 * - If idle or send timeout occurs → the connection is closed immediately via
231 * `cleanupClientConnectionClose()`, and the function returns `false`.
232 * - If header or body timeout occurs → the connection stays open long enough to
233 * send a timeout response (408). To prevent further reads, `POLLIN` is disabled
234 * for this fd, and the function returns `true`.
235 * - If no timeout is triggered, returns `false`.
236 *
237 * @param client_fd The socket file descriptor of the client.
238 * @param index Index of the client's pollfd entry in `_poll_fds`.
239 * @return `true` if a recoverable timeout (header/body) was detected and the
240 * connection is left open for sending a response,
241 * `false` if the connection was closed or no timeout occurred.
242 *
243 * @note This function integrates multiple specialized timeout checks into the
244 * main poll loop, ensuring the server proactively cleans up stale or
245 * non-responsive connections.
246 */
247bool SocketManager::checkClientTimeouts(int client_fd, size_t index) {
248 if (!_client_info.count(client_fd))
249 return false;
250
251 time_t now = getCurrentTime();
252
253 // Hard timeouts → connection closed
254 if (isIdleTimeout(client_fd, now) || isSendTimeout(client_fd, now)) {
255 cleanupClientConnectionClose(client_fd, index);
256 return false;
257 }
258
259 // Soft timeouts → send 408 and disable POLLIN
260 if (isHeaderTimeout(client_fd, now) || isBodyTimeout(client_fd, now)) {
261 _poll_fds[index].events &= ~POLLIN;
262 return true;
263 }
264
265 return false;
266}

268/**
269 * @brief Enforces request size limits for an active client connection.
270 *
271 * @details
272 * This function checks whether the accumulated request headers for a client
273 * exceed the configured maximum length (`HEADER_MAX_LENGTH`).
274 *
275 * Behavior:
276 * - If the header is incomplete **and** the number of bytes received so far
277 * exceeds the allowed maximum, the server immediately queues a
278 * **431 Request Header Fields Too Large** error response using
279 * `respondError()`, and returns `true`.
280 * - Otherwise, the request is considered within limits and processing continues.
281 *
282 * @param fd The socket file descriptor of the client being checked.
283 * @return `true` if the request exceeded the configured header limit
284 * (error response queued),
285 * `false` if still within allowed limits.
286 *
287 * @note This check is only enforced **before header completion**. Once headers
288 * are fully parsed, the limit is no longer evaluated here.
289 */
290bool SocketManager::checkRequestLimits(int fd) {
291 ClientInfo& client = _client_info[fd];
292
293 // Enforce header-length limit only while headers are incomplete
294 if (client.headerBytesReceived > HEADER_MAX_LENGTH) {
295 Logger::logFrom(LogLevel::WARN, "SocketManager",
296 "Request header too large from fd: " + std::to_string(fd));
297 respondError(fd, 431); // 431: Request Header Fields Too Large
298 return true;
299 }
300
301 return false;
302}
