26bool SocketManager::isHeaderTimeout(int fd, time_t now) {
27 ClientInfo& client = _client_info[fd];
28
29 if (client.responses.empty() && client.current_raw_response.empty() && !client.headerComplete &&
30 client.headerBytesReceived > 0 &&
31 now - client.connectionStartTime > HEADER_TIMEOUT_SECONDS) {
32 Logger::logFrom(LogLevel::WARN, "SocketManager", "Timeout on fd: " + std::to_string(fd));
33 respondError(fd, 408);
34 return true;
35 }
36
37 return false;
38}

62bool SocketManager::isIdleTimeout(int fd, time_t now) {
63 ClientInfo& client = _client_info[fd];
64 if (client.cgiProcess.has_value()) {
65 return false;
66 }
67 if (client.responses.empty() && client.current_raw_response.empty() && !client.headerComplete &&
68 client.headerBytesReceived == 0 && now - client.lastRequestTime > TIMEOUT) {
69 Logger::logFrom(LogLevel::WARN, "SocketManager",
70 "Idle timeout on fd: " + std::to_string(fd));
71 return true;
72 }

76bool SocketManager::checkClientTimeouts(int client_fd, size_t index) {
77 if (!_client_info.count(client_fd))
78 return false;
79
80 time_t now = getCurrentTime();
81 if (isIdleTimeout(client_fd, now) || isSendTimeout(client_fd, now)) {
82 cleanupClientConnectionClose(client_fd, index);
83 return false;
84 }
85 if (isHeaderTimeout(client_fd, now) || isBodyTimeout(client_fd, now)) {
86 _poll_fds[index].events &= ~POLLIN;
87 return true;
88 }
89 return false;
90}
