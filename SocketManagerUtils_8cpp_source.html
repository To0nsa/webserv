50void SocketManager::cleanupClientState(int client_fd) {
51 auto it = _client_info.find(client_fd);
52 if (it == _client_info.end())
53 return;
54
55 ClientInfo& client = it->second;
56
57 while (!client.responses.empty()) {
58 HttpResponse& resp = client.responses.front();
59 if (resp.isCgiTempFile()) {
60 CGI::unlinkWithErrorLog(resp.getCgiTempFile(), "out temp file");
61 }
62 client.responses.pop();
63 }
64
65 if (client.cgiProcess) {
66 CGI::errorOnCgi(*client.cgiProcess);
67 client.cgiProcess.reset();
68 client.isCgiProcessRunning = false;
69 client.currentCgiRequest = HttpRequest();
70 }
71
72 if (client.file_stream.is_open()) {
73 client.file_stream.close();
74 }
75
76 _client_info.erase(it);
77}

87void SocketManager::resetRequestState(int client_fd) {
88 if (!_client_info.count(client_fd))
89 return;
90 // If there is still any data in requestBuffer, treat it as a partial header:
91 if (!_client_info[client_fd].requestBuffer.empty()) {
92 _client_info[client_fd].headerComplete = false;
93 // headerBytesReceived should reflect how many bytes are already in the buffer.
94 // But if we are just about to parse a brand‐new header, headerBytesReceived
95 // should have already been set by receiveFromClient(...) when those bytes first arrived.
96 // So here we do NOT zero it out—leave it alone so the header‐timer can still tick.
97 return;
98 }
99 // If requestBuffer is empty, then there is no partial header in progress.
100 _client_info[client_fd].headerComplete = false;
101 _client_info[client_fd].headerBytesReceived = 0;
102 _client_info[client_fd].bodyBytesReceived = 0;
103}
