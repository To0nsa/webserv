13/**
14 * @file SocketManagerUtils.cpp
15 * @brief Utility methods for connection lifecycle management in SocketManager.
16 *
17 * @details
18 * This file implements helper functions used internally by the
19 * `SocketManager` class to manage client state, poll descriptors,
20 * error handling, and cleanup routines. These utilities ensure proper
21 * resource deallocation and safe connection teardown.
22 *
23 * Main responsibilities:
24 * - Cleaning up CGI processes associated with a client (`cleanupCgiForClient`).
25 * - Removing `pollfd` entries from the monitored list (`removePollFd`).
26 * - Resetting client request parsing state (`resetRequestState`).
27 * - Gracefully closing client connections and releasing resources
28 * (`cleanupClientState`, `cleanupClientConnectionClose`).
29 * - Generating and queuing error responses (`respondError`).
30 *
31 * These functions are invoked by higher-level event loop and request
32 * handling methods to enforce robustness, prevent resource leaks, and
33 * handle exceptional conditions.
34 *
35 * @ingroup socket_manager
36 */

55/**
56 * @brief Cleans up and resets CGI state for a specific client.
57 *
58 * @details
59 * If the client identified by @p client_fd has an active CGI process,
60 * this function terminates it via `CGI::cleanupCgi`, resets its state,
61 * and clears the currently running CGI request.
62 * If no CGI process is active, the function returns without action.
63 *
64 * @param client_fd The file descriptor of the client whose CGI state should be cleaned up.
65 *
66 * @note This function does not close the client connection itself — it only
67 * resets CGI-related state. Connection cleanup is handled separately.
68 */
69void SocketManager::cleanupCgiForClient(int client_fd) {
70 if (!_client_info.contains(client_fd))
71 return;
72
73 ClientInfo& client = _client_info[client_fd];
74 if (!client.cgiProcess)
75 return;
76
77 CGI::cleanupCgi(*client.cgiProcess);
78 client.cgiProcess.reset();
79 client.isCgiProcessRunning = false;
80 client.currentCgiRequest = HttpRequest();
81}

83/**
84 * @brief Removes a file descriptor entry from the poll list.
85 *
86 * @details
87 * This function erases the `pollfd` structure at the given index
88 * from the `_poll_fds` vector, which is used by `poll()` in the
89 * main event loop to monitor active sockets.
90 *
91 * If the index is out of range, no action is performed.
92 *
93 * @param index The position in `_poll_fds` corresponding to the
94 * client or listening socket to be removed.
95 *
96 * @note This does not close the socket itself. The caller is responsible
97 * for performing connection cleanup separately (see
98 * `cleanupClientConnectionClose`).
99 */
100void SocketManager::removePollFd(size_t index) {
101 if (index < _poll_fds.size()) {
102 _poll_fds.erase(_poll_fds.begin() + index);
103 }
104}

106/**
107 * @brief Cleans up all state associated with a client connection.
108 *
109 * @details
110 * This function safely releases all resources tied to a client:
111 * - Drains the response queue and deletes any CGI temporary files.
112 * - Cleans up and terminates any active CGI process.
113 * - Closes any open file streams linked to the client.
114 * - Erases the client entry from `_client_info`.
115 *
116 * Unlike `cleanupClientConnectionClose`, this function does not close
117 * the socket or modify the poll list — it only clears per-client state.
118 *
119 * @param client_fd The socket file descriptor of the client whose state
120 * should be cleaned up.
121 *
122 * @note Call this when tearing down a client session to ensure no resource
123 * leaks (file handles, temp files, or CGI processes).
124 */
125void SocketManager::cleanupClientState(int client_fd) {
126 auto it = _client_info.find(client_fd);
127 if (it == _client_info.end())
128 return;
129
130 ClientInfo& client = it->second;
131
132 // Clean up pending responses and unlink any CGI temp files
133 while (!client.responses.empty()) {
134 HttpResponse& resp = client.responses.front();
135 if (resp.isCgiTempFile()) {
136 CGI::unlinkWithErrorLog(resp.getCgiTempFile(), "out temp file");
137 }
138 client.responses.pop();
139 }
140
141 // Handle any active CGI process
142 if (client.cgiProcess) {
143 CGI::errorOnCgi(*client.cgiProcess);
144 client.cgiProcess.reset();
145 client.isCgiProcessRunning = false;
146 client.currentCgiRequest = HttpRequest();
147 }
148
149 // Close file stream if open
150 if (client.file_stream.is_open()) {
151 client.file_stream.close();
152 }
153
154 // Finally remove client record
155 _client_info.erase(it);
156}

158/**
159 * @brief Gracefully closes a client connection and cleans up its state.
160 *
161 * @details
162 * This function performs a full teardown of a client connection:
163 * - Removes the file descriptor from the poll list (`removePollFd`).
164 * - Cleans up all per-client state (`cleanupClientState`), including
165 * pending responses, CGI processes, and open streams.
166 * - Closes the actual socket file descriptor via `close()`.
167 * - Logs the closure event.
168 *
169 * @param client_fd The socket file descriptor of the client being closed.
170 * @param index The index in `_poll_fds` corresponding to the client_fd.
171 *
172 * @note Use this when the connection must be terminated (e.g. client
173 * disconnect, fatal error, or `Connection: close` response).
174 * For state cleanup without closing the socket, see `cleanupClientState()`.
175 */
176void SocketManager::cleanupClientConnectionClose(int client_fd, size_t index) {
177 removePollFd(index);
178 cleanupClientState(client_fd);
179 close(client_fd);
180 Logger::logFrom(LogLevel::INFO, "SocketManager cleanupClientConnectionClose",
181 "Closed FD (Connection: close): " + std::to_string(client_fd));
182}

184/**
185 * @brief Resets the parsing state of a client’s current HTTP request.
186 *
187 * @details
188 * This function clears or resets request-related counters for a given client:
189 * - If the `requestBuffer` still contains data, the connection is treated
190 * as having an incomplete header. In this case, `headerComplete` is reset
191 * to false, but `headerBytesReceived` is preserved so that header timeout
192 * tracking continues.
193 * - If the buffer is empty, the function resets all parsing counters
194 * (`headerComplete`, `headerBytesReceived`, `bodyBytesReceived`).
195 *
196 * @param client_fd The file descriptor of the client whose request
197 * parsing state should be reset.
198 *
199 * @note This is typically called after a request has been fully parsed
200 * or after an error, so the connection can be reused for the next request
201 * (in keep-alive scenarios).
202 */
203void SocketManager::resetRequestState(int client_fd) {
204 if (!_client_info.count(client_fd))
205 return;
206
207 // If there is still data in requestBuffer, consider it a partial header
208 if (!_client_info[client_fd].requestBuffer.empty()) {
209 _client_info[client_fd].headerComplete = false;
210 // Preserve headerBytesReceived so timeout logic remains valid
211 return;
212 }
213
214 // No partial header in progress → fully reset state
215 _client_info[client_fd].headerComplete = false;
216 _client_info[client_fd].headerBytesReceived = 0;
217 _client_info[client_fd].bodyBytesReceived = 0;
218}

220/**
221 * @brief Queues an HTTP error response for a client.
222 *
223 * @details
224 * This function generates a standardized HTTP error response with the
225 * given status code (e.g. 400, 404, 500) and pushes it onto the client’s
226 * response queue.
227 *
228 * - Uses the first server in the client’s `serversOnPort` list as a fallback
229 * context for generating the response.
230 * - The request object passed to `ResponseBuilder::generateError` is empty,
231 * since the error may occur before a valid request could be parsed.
232 *
233 * @param fd The client’s socket file descriptor.
234 * @param status_code The HTTP status code to send (e.g. 400, 404, 500).
235 *
236 * @note This does not immediately send the response — it only enqueues it.
237 * The response will be transmitted later in the event loop
238 * by `sendResponse()`.
239 */
240void SocketManager::respondError(int fd, int status_code) {
241 HttpRequest empty;
242 const Server& fallback = _client_info[fd].serversOnPort.front();
243 HttpResponse err = ResponseBuilder::generateError(status_code, fallback, empty);
244 _client_info[fd].responses.push(err);
245}
