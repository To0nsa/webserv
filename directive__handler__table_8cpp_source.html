56static void validateIPv4Address(const std::string& ip, int line, int column,
57 const std::function<std::string()>& context_provider) {
58 std::istringstream iss(ip);
59 std::string segment;
60 int count = 0;
61
62 if (ip == "localhost")
63 return;
64
65 while (std::getline(iss, segment, '.')) {
66 if (++count > 4) {
67 throw SyntaxError(formatError("Too many octets in IP address: " + ip, line, column),
68 context_provider());
69 }
70
71 int octet = parseInt(segment, "host", line, column, context_provider);
72
73 if (octet < 0 || octet > 255) {
74 throw SyntaxError(
75 formatError("Invalid IP octet '" + segment + "' in host: " + ip, line, column),
76 context_provider());
77 }
78 }
79 if (count != 4) {
80 throw SyntaxError(
81 formatError("Invalid IP address format (expected 4 octets): " + ip, line, column),
82 context_provider());
83 }
84}

86static void validateCgiExtension(const std::string& ext, int line, int column,
87 const std::function<std::string()>& context_provider) {
88 if (ext.empty() || ext == "." || ext[0] != '.') {
89 throw SyntaxError(formatError("Invalid CGI extension: '" + ext + "'", line, column),
90 context_provider());
91 }
92
93 for (std::size_t i = 1; i < ext.size(); ++i) {
94 char c = ext[i];
95 if (!std::isalnum(c)) {
96 throw SyntaxError(
97 formatError("Invalid character in CGI extension: '" + ext + "'", line, column),
98 context_provider());
99 }
100 }

103const std::unordered_map<std::string, ServerHandler>& serverHandlers() {
104
105 static const std::unordered_map<std::string, ServerHandler> map = {
106 {"listen",
107 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
108 requireArgCount(v, 1, "listen", line, column, ctx);
109 int port = parseInt(v[0], "listen", line, column, [&]() { return ctx; });
110 if (port < 0 || port > 65535) {
111 throw SyntaxError(
112 formatError("Port number out of valid range (0-65535): " + v[0], line, column),
113 ctx);
114 }
115 s.setPort(port);
116 }},
117 {"host",
118 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
119 requireArgCount(v, 1, "host", line, column, ctx);
120 const std::string& ip = v[0];
121 validateIPv4Address(ip, line, column, [&]() { return ctx; });
122 s.setHost(ip);
123 }},
124 {"server_name",
125 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
126 requireMinArgCount(v, 1, "server_name", line, column, ctx);
127 for (const auto& name : v) {
128 s.addServerName(name);
129 }
130 }},
131
132 {"client_max_body_size",
133 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
134 requireArgCount(v, 1, "client_max_body_size", line, column, ctx);
135 s.setClientMaxBodySize(
136 parseByteSize(v[0], "client_max_body_size", line, column, [&]() { return ctx; }));
137 }},
138 {"error_page",
139 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
140 requireMinArgCount(v, 2, "error_page", line, column, ctx);
141 std::string uri = v.back();
142 for (std::size_t i = 0; i + 1 < v.size(); ++i) {
143 int code = parseInt(v[i], "error_page", line, column, [&]() { return ctx; });
144 s.setErrorPage(code, uri);
145 }
146 }},
147 };
148 return map;
149}

151const std::unordered_map<std::string, LocationHandler>& locationHandlers() {
152 static const std::unordered_map<std::string, LocationHandler> map = {
153 {"root",
154 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
155 requireArgCount(v, 1, "root", line, column, ctx);
156 loc.setRoot(resolveToAbsolute(v[0]));
157 }},
158 {"index",
159 [](Location& loc, const auto& args, int line, int column, const std::string& ctx) {
160 requireMinArgCount(args, 1, "index", line, column, ctx);
161 for (const std::string& raw : args) {
162 size_t start = 0, end;
163 while ((end = raw.find(',', start)) != std::string::npos) {
164 std::string idx = raw.substr(start, end - start);
165 if (!idx.empty())
166 loc.addIndexFile(idx);
167 start = end + 1;
168 }
169 std::string idx = raw.substr(start);
170 if (!idx.empty())
171 loc.addIndexFile(idx);
172 }
173 }},
174 {"autoindex",
175 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
176 requireArgCount(v, 1, "autoindex", line, column, ctx);
177 if (v[0] == "on") {
178 loc.setAutoindex(true);
179 } else if (v[0] == "off") {
180 loc.setAutoindex(false);
181 } else {
182 throw SyntaxError(
183 formatError("Invalid value for 'autoindex': " + v[0], line, column), ctx);
184 }
185 }},
186 {"methods",
187 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
188 if (v.empty()) {
189 throw SyntaxError(
190 formatError("Directive 'methods' requires at least one HTTP method", line,
191 column),
192 ctx);
193 }
194 static const std::set<std::string> valid_methods = {"GET", "POST", "DELETE"};
195 for (const auto& m : v) {
196 if (!valid_methods.count(m)) {
197 throw SyntaxError(formatError("Invalid HTTP method: " + m, line, column), ctx);
198 }
199 loc.addMethod(m);
200 }
201 }},
202 {"upload_store",
203 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
204 requireArgCount(v, 1, "upload_store", line, column, ctx);
205 loc.setUploadStore(resolveToAbsolute(v[0]));
206 }},
207 {"cgi_extension",
208 [](Location& loc, const auto& args, int line, int column, const std::string& ctx) {
209 requireMinArgCount(args, 1, "cgi_extension", line, column, ctx);
210 for (const std::string& raw : args) {
211 size_t start = 0, end;
212 while ((end = raw.find(',', start)) != std::string::npos) {
213 std::string ext = raw.substr(start, end - start);
214 if (!ext.empty()) {
215 validateCgiExtension(ext, line, column, [&]() { return ctx; });
216 loc.addCgiExtension(ext);
217 }
218 start = end + 1;
219 }
220 std::string ext = raw.substr(start);
221 if (!ext.empty()) {
222 validateCgiExtension(ext, line, column, [&]() { return ctx; });
223 loc.addCgiExtension(ext);
224 }
225 }
226 }},
227 {"cgi_interpreter",
228 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
229 requireArgCount(v, 2, "cgi_interpreter", line, column, ctx);
230 const std::string& ext = v[0];
231 const std::string& path = v[1];
232 validateCgiExtension(ext, line, column, [&]() { return ctx; });
233 if (!std::filesystem::is_regular_file(path) || access(path.c_str(), X_OK) != 0) {
234 throw SyntaxError(
235 formatError("Interpreter not executable or not found: " + path, line, column),
236 ctx);
237 }
238 if (!loc.getCgiInterpreter(ext).empty()) {
239 throw SyntaxError(
240 formatError("Duplicate cgi_interpreter for " + ext + ": already defined", line,
241 column),
242 ctx);
243 }
244 loc.addCgiInterpreter(ext, path);
245 }},
246 {"return",
247 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
248 requireArgCount(v, 2, "return", line, column, ctx);
249 int code = parseInt(v[0], "return", line, column, [&]() { return ctx; });
250 loc.setRedirect(v[1], code);
251 }},
252 };
253 return map;
254}
