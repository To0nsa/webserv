14 * @file directive_handler_table.cpp
15 * @brief Implements directive handler maps for `server` and `location` blocks.
16 *
17 * @details Defines the concrete handler tables returned by
18 * directive::serverHandlers and directive::locationHandlers.
19 * Each handler validates its arguments, reports precise diagnostics
20 * (line/column + contextual snippet), and mutates the target
21 * configuration object (Server or Location).
22 *
23 * This file also contains small internal helpers used by handlers:
24 * - `resolveToAbsolute()` — canonicalizes filesystem paths.
25 * - `requireArgCount()` / `requireMinArgCount()` — arity checks.
26 * - `validateIPv4Address()` — IPv4 format validation (with `localhost` allowed).
27 * - `validateCgiExtension()` — enforces dot-prefixed alnum extensions.
28 *
29 * @ingroup config_parsing

47/**
48 * @brief Resolves a filesystem path to its absolute, normalized form.
49 *
50 * @details Converts the given raw path string into an absolute path
51 * using `std::filesystem::absolute()`, then applies lexical
52 * normalization (removing redundant `.` and `..` components
53 * and duplicate separators). This ensures the result is a
54 * canonicalized path string safe for use in configuration
55 * directives (e.g., `root`, `upload_store`).
56 *
57 * @param rawPath The raw input path (may be relative or contain `.`/`..`).
58 * @return A string representing the absolute, lexically normalized path.
59 *
60 * @throw std::filesystem::filesystem_error If the path cannot be resolved
61 * due to invalid characters or inaccessible filesystem.
62 *
63 * @ingroup config_parser
64 */
65static std::string resolveToAbsolute(const std::string& rawPath) {
66 std::filesystem::path abs = std::filesystem::absolute(rawPath);
67 return abs.lexically_normal().string();
68}

70/**
71 * @brief Ensures that a directive has exactly the expected number of arguments.
72 *
73 * @details Used during configuration parsing to validate that a directive
74 * (e.g., `listen`, `root`, `client_max_body_size`) is provided with
75 * the correct number of arguments. If the count does not match
76 * the expectation, a `SyntaxError` is thrown with contextual
77 * information about the directive, line, column, and source snippet.
78 *
79 * @param args The list of argument strings parsed for the directive.
80 * @param expected The exact number of arguments required.
81 * @param directive The name of the directive being validated.
82 * @param line The line number in the config file where the directive appears.
83 * @param column The column number in the config file where the directive appears.
84 * @param ctx A snippet of the surrounding source text for error reporting.
85 *
86 * @throw SyntaxError If the directive does not have exactly `expected` arguments.
87 *
88 * @ingroup config_parser
89 */
90static void requireArgCount(const std::vector<std::string>& args, std::size_t expected,
91 const std::string& directive, int line, int column,
92 const std::string& ctx) {
93 // Check if the argument count matches the expected number
94 if (args.size() != expected) {
95 // If not, throw a SyntaxError with detailed context
96 throw SyntaxError(formatError("Directive '" + directive + "' takes exactly " +
97 std::to_string(expected) + " argument(s), but got " +
98 std::to_string(args.size()),
99 line, column),
100 ctx);
101 }
102}

117 * @param column The column number in the config file where the directive appears.

125 const std::string& directive, int line, int column,

127 // Check if number of parsed arguments is below the minimum

129 // Throw a SyntaxError with detailed error message and context
130 throw SyntaxError(formatError("Directive '" + directive + "' requires at least " +
131 std::to_string(min) + " argument(s), but got " +

138/**
139 * @brief Validates that a given string is a valid IPv4 address.
140 *
141 * @details This function ensures that the provided `ip` string is either
142 * `"localhost"` (special case allowed) or a well-formed IPv4 address
143 * in dotted-decimal notation (e.g., `127.0.0.1`, `192.168.1.42`).
144 * It splits the input into octets, parses each into an integer, and
145 * validates that:
146 * - There are exactly 4 octets.
147 * - Each octet is between 0 and 255.
148 * - No extra segments exist.
149 *
150 * @param ip The IP address string to validate.
151 * @param line Line number in the config file where the directive appears.
152 * @param column Column number in the config file where the directive appears.
153 * @param context_provider Lazy-evaluated function returning a source context snippet.
154 *
155 * @throw SyntaxError If the IP address is malformed (wrong number of octets,
156 * non-numeric values, or values outside 0–255).
157 *
158 * @ingroup config_parser
159 */
160static void validateIPv4Address(const std::string& ip, int line, int column,
161 const std::function<std::string()>& context_provider) {
162 std::istringstream iss(ip);
163 std::string segment;
164 int count = 0;
165
166 // Special case: allow "localhost" without further validation
167 if (ip == "localhost")
168 return;
169
170 // Split the string by '.' and validate each segment
171 while (std::getline(iss, segment, '.')) {
172 if (++count > 4) {
173 // More than 4 parts → invalid IPv4
174 throw SyntaxError(formatError("Too many octets in IP address: " + ip, line, column),
175 context_provider());
176 }
177
178 // Convert segment to integer, throws if non-numeric
179 int octet = parseInt(segment, "host", line, column, context_provider);
180
181 // Check range validity (0–255)
182 if (octet < 0 || octet > 255) {
183 throw SyntaxError(
184 formatError("Invalid IP octet '" + segment + "' in host: " + ip, line, column),
185 context_provider());
186 }
187 }
188
189 // Must have exactly 4 octets
190 if (count != 4) {
191 throw SyntaxError(
192 formatError("Invalid IP address format (expected 4 octets): " + ip, line, column),
193 context_provider());
194 }
195}

197/**
198 * @brief Validates the syntax of a CGI file extension.
199 *
200 * @details This function ensures that a given CGI extension string is valid.
201 * A valid extension must:
202 * - Be non-empty.
203 * - Begin with a dot (`.`).
204 * - Contain only alphanumeric characters after the dot.
205 * - Not be just `"."` with no following characters.
206 *
207 * @param ext The extension string to validate (e.g., ".php").
208 * @param line Line number in the config file for error reporting.
209 * @param column Column number in the config file for error reporting.
210 * @param context_provider Lazy-evaluated function returning a source context snippet.
211 *
212 * @throw SyntaxError If the extension is malformed or contains invalid characters.
213 *
214 * @ingroup config_parser
215 */
216static void validateCgiExtension(const std::string& ext, int line, int column,
217 const std::function<std::string()>& context_provider) {
218 // Must not be empty, a lone '.', and must start with a dot
219 if (ext.empty() || ext == "." || ext[0] != '.') {
220 throw SyntaxError(formatError("Invalid CGI extension: '" + ext + "'", line, column),
221 context_provider());
222 }
223
224 // Check that all characters after '.' are alphanumeric
225 for (std::size_t i = 1; i < ext.size(); ++i) {
226 char c = ext[i];
227 if (!std::isalnum(c)) {
228 throw SyntaxError(
229 formatError("Invalid character in CGI extension: '" + ext + "'", line, column),
230 context_provider());
231 }
232 }

235/**
236 * @brief Returns the mapping of server-level directives to their handler functions.
237 *
238 * @details
239 * This function builds (once, statically) an `unordered_map` that associates each
240 * directive keyword valid inside a `server { ... }` block with a corresponding
241 * validation and handler lambda. Each handler is responsible for:
242 * - Checking argument count and validity.
243 * - Validating the argument format (IP, port, byte sizes, etc.).
244 * - Updating the `Server` object with the parsed value.
245 *
246 * Supported directives:
247 * - **listen <port>;**
248 * Validates port number (0–65535) and stores it in the server object.
249 * - **host <ip>;**
250 * Validates IPv4 address (or "localhost") and assigns it to the server.
251 * - **server_name <name...>;**
252 * Accepts one or more hostnames; each is registered as a valid server name.
253 * - **client_max_body_size <size>;**
254 * Validates a byte-size expression (e.g., "10M") and sets the upload limit.
255 * - **error_page <code...> <uri>;**
256 * Accepts one or more HTTP error codes followed by a URI.
257 * Maps each error code to the provided URI.
258 *
259 * @return A reference to the immutable directive-to-handler map.
260 *
261 * @ingroup config_parser
262 */
263const std::unordered_map<std::string, ServerHandler>& serverHandlers() {
264
265 // Static so the map is created once and reused on subsequent calls.
266 static const std::unordered_map<std::string, ServerHandler> map = {
267 // ───────────────────────────────
268 // `listen <port>;`
269 // Validates that the port number is within range [0, 65535].
270 {"listen",
271 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
272 requireArgCount(v, 1, "listen", line, column, ctx); // must have exactly one argument
273 int port = parseInt(v[0], "listen", line, column, [&]() { return ctx; });
274 if (port < 0 || port > 65535) {
275 throw SyntaxError(
276 formatError("Port number out of valid range (0-65535): " + v[0], line, column),
277 ctx);
278 }
279 s.setPort(port); // assign validated port to server
280 }},
281
282 // ───────────────────────────────
283 // `host <ip>;`
284 // Validates IPv4 address (or "localhost").
285 {"host",
286 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
287 requireArgCount(v, 1, "host", line, column, ctx); // must have exactly one argument
288 const std::string& ip = v[0];
289 validateIPv4Address(ip, line, column, [&]() { return ctx; }); // ensure valid format
290 s.setHost(ip); // store host in server
291 }},
292
293 // ───────────────────────────────
294 // `server_name <name...>;`
295 // Allows one or more server names (aliases).
296 {"server_name",
297 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
298 requireMinArgCount(v, 1, "server_name", line, column, ctx); // at least one name
299 for (const auto& name : v) {
300 s.addServerName(name); // register each alias
301 }
302 }},
303
304 // ───────────────────────────────
305 // `client_max_body_size <size>;`
306 // Validates a byte-size string (e.g., "10M") and sets upload limit.
307 {"client_max_body_size",
308 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
309 requireArgCount(v, 1, "client_max_body_size", line, column, ctx); // one argument
310 s.setClientMaxBodySize(parseByteSize(v[0], "client_max_body_size", line, column,
311 [&]() { return ctx; })); // convert and set
312 }},
313
314 // ───────────────────────────────
315 // `error_page <code...> <uri>;`
316 // Maps one or more error codes to a redirect URI.
317 {"error_page",
318 [](Server& s, const auto& v, int line, int column, const std::string& ctx) {
319 requireMinArgCount(v, 2, "error_page", line, column, ctx); // at least one code + URI
320 std::string uri = v.back(); // last argument is the URI
321 for (std::size_t i = 0; i + 1 < v.size(); ++i) {
322 int code = parseInt(v[i], "error_page", line, column, [&]() { return ctx; });
323 s.setErrorPage(code, uri); // map each code to the URI
324 }
325 }},
326 };
327
328 return map;
329}

331/**
332 * @brief Returns the mapping of location-level directives to their handler functions.
333 *
334 * @details
335 * This function builds (once, statically) an `unordered_map` that associates each
336 * directive keyword valid inside a `location { ... }` block with its validation
337 * and handler lambda. Each handler checks argument counts, validates syntax,
338 * and updates the `Location` object accordingly.
339 *
340 * Supported directives:
341 * - **root <path>;**
342 * Sets the filesystem root for this location (absolute, normalized path).
343 * - **index <file...>;**
344 * Specifies one or more default index files (supports comma-separated values).
345 * - **autoindex on|off;**
346 * Enables or disables directory listing.
347 * - **methods <HTTP-method...>;**
348 * Restricts allowed HTTP methods (GET, POST, DELETE).
349 * - **upload_store <path>;**
350 * Defines the directory for file uploads (absolute, normalized path).
351 * - **cgi_extension <.ext...>;**
352 * Lists extensions handled by CGI (e.g., `.php`, `.py`).
353 * - **cgi_interpreter <.ext> <path>;**
354 * Associates a CGI extension with an executable interpreter.
355 * - **return <code> <uri>;**
356 * Configures a redirect with HTTP status code.
357 *
358 * @return A reference to the immutable directive-to-handler map.
359 *
360 * @ingroup config_parser
361 */
362const std::unordered_map<std::string, LocationHandler>& locationHandlers() {
363 static const std::unordered_map<std::string, LocationHandler> map = {
364 // ───────────────────────────────
365 // `root <path>;`
366 // Sets the root directory for this location.
367 {"root",
368 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
369 requireArgCount(v, 1, "root", line, column, ctx); // must have exactly one path
370 loc.setRoot(resolveToAbsolute(v[0])); // normalize to absolute path
371 }},
372
373 // ───────────────────────────────
374 // `index <file...>;`
375 // Defines one or more index files (comma-separated allowed).
376 {"index",
377 [](Location& loc, const auto& args, int line, int column, const std::string& ctx) {
378 requireMinArgCount(args, 1, "index", line, column, ctx); // at least one file
379 for (const std::string& raw : args) {
380 size_t start = 0, end;
381 // Split on commas
382 while ((end = raw.find(',', start)) != std::string::npos) {
383 std::string idx = raw.substr(start, end - start);
384 if (!idx.empty())
385 loc.addIndexFile(idx); // add each file
386 start = end + 1;
387 }
388 std::string idx = raw.substr(start);
389 if (!idx.empty())
390 loc.addIndexFile(idx);
391 }
392 }},
393
394 // ───────────────────────────────
395 // `autoindex on|off;`
396 // Enables or disables directory listing.
397 {"autoindex",
398 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
399 requireArgCount(v, 1, "autoindex", line, column, ctx);
400 if (v[0] == "on") {
401 loc.setAutoindex(true);
402 } else if (v[0] == "off") {
403 loc.setAutoindex(false);
404 } else {
405 throw SyntaxError(
406 formatError("Invalid value for 'autoindex': " + v[0], line, column), ctx);
407 }
408 }},
409
410 // ───────────────────────────────
411 // `methods <method...>;`
412 // Restricts allowed HTTP methods (must be GET, POST, or DELETE).
413 {"methods",
414 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
415 if (v.empty()) {
416 throw SyntaxError(
417 formatError("Directive 'methods' requires at least one HTTP method", line,
418 column),
419 ctx);
420 }
421 static const std::set<std::string> valid_methods = {"GET", "POST", "DELETE"};
422 for (const auto& m : v) {
423 if (!valid_methods.count(m)) {
424 throw SyntaxError(formatError("Invalid HTTP method: " + m, line, column), ctx);
425 }
426 loc.addMethod(m); // store validated method
427 }
428 }},
429
430 // ───────────────────────────────
431 // `upload_store <path>;`
432 // Sets the directory where file uploads will be saved.
433 {"upload_store",
434 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
435 requireArgCount(v, 1, "upload_store", line, column, ctx);
436 loc.setUploadStore(resolveToAbsolute(v[0])); // normalize path
437 }},
438
439 // ───────────────────────────────
440 // `cgi_extension <.ext...>;`
441 // Declares CGI file extensions (e.g., `.php`).
442 {"cgi_extension",
443 [](Location& loc, const auto& args, int line, int column, const std::string& ctx) {
444 requireMinArgCount(args, 1, "cgi_extension", line, column, ctx);
445 for (const std::string& raw : args) {
446 size_t start = 0, end;
447 // Split multiple extensions separated by commas
448 while ((end = raw.find(',', start)) != std::string::npos) {
449 std::string ext = raw.substr(start, end - start);
450 if (!ext.empty()) {
451 validateCgiExtension(ext, line, column, [&]() { return ctx; });
452 loc.addCgiExtension(ext);
453 }
454 start = end + 1;
455 }
456 std::string ext = raw.substr(start);
457 if (!ext.empty()) {
458 validateCgiExtension(ext, line, column, [&]() { return ctx; });
459 loc.addCgiExtension(ext);
460 }
461 }
462 }},
463
464 // ───────────────────────────────
465 // `cgi_interpreter <.ext> <path>;`
466 // Maps a CGI extension to its interpreter binary.
467 {"cgi_interpreter",
468 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
469 requireArgCount(v, 2, "cgi_interpreter", line, column, ctx); // ext + path required
470 const std::string& ext = v[0];
471 const std::string& path = v[1];
472
473 validateCgiExtension(ext, line, column, [&]() { return ctx; }); // must be valid ext
474 if (!std::filesystem::is_regular_file(path) || access(path.c_str(), X_OK) != 0) {
475 throw SyntaxError(
476 formatError("Interpreter not executable or not found: " + path, line, column),
477 ctx);
478 }
479 if (!loc.getCgiInterpreter(ext).empty()) {
480 throw SyntaxError(
481 formatError("Duplicate cgi_interpreter for " + ext + ": already defined", line,
482 column),
483 ctx);
484 }
485 loc.addCgiInterpreter(ext, path); // store extension → interpreter mapping
486 }},
487
488 // ───────────────────────────────
489 // `return <code> <uri>;`
490 // Defines a redirect (e.g., `return 301 /newpath;`).
491 {"return",
492 [](Location& loc, const auto& v, int line, int column, const std::string& ctx) {
493 requireArgCount(v, 2, "return", line, column, ctx);
494 int code = parseInt(v[0], "return", line, column, [&]() { return ctx; });
495 loc.setRedirect(v[1], code); // assign redirect URI + status code
496 }},
497 };
498
499 return map;
500}
