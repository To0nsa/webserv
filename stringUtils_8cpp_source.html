46/**
47 * @ingroup string_utils
48 * @brief Parses a non-negative integer from a string with context-aware error reporting.
49 *
50 * @details Attempts to parse the entire @p value string into a non-negative integer
51 * using `std::from_chars` for efficient, allocation-free conversion.
52 * If parsing fails, the string contains extra characters, or the result
53 * is negative, a @ref ConfigParseError is thrown. The error includes:
54 * - The field name for context.
55 * - The offending value.
56 * - The line and column numbers.
57 * - Additional context from @p context_provider, such as a snippet of the
58 * configuration file being parsed.
59 *
60 * @param value String representation of the integer to parse.
61 * @param field Name of the configuration field being parsed.
62 * @param line Line number in the configuration source.
63 * @param column Column number in the configuration source.
64 * @param context_provider Callable returning a string snippet or context for diagnostics.
65 * @return Parsed integer value (guaranteed to be non-negative).
66 *
67 * @throws ConfigParseError If parsing fails, if extra characters remain after parsing,
68 * or if the parsed value is negative.
69 *
70 * @note This function is typically used when reading numeric configuration values
71 * (e.g., port numbers, limits) that must be whole, non-negative integers.
72 */
73int parseInt(const std::string& value, const std::string& field, int line, int column,
74 const std::function<std::string()>& context_provider) {
75 int result = 0;
76
77 // Try to convert the full string to an integer using from_chars (no allocations, fast)
78 auto [ptr, ec] = std::from_chars(value.data(), value.data() + value.size(), result);
79
80 // Check if conversion failed, or if there were leftover characters, or if result is negative
81 if (ec != std::errc() || ptr != value.data() + value.size() || result < 0) {
82 // Throw an error with precise source location and contextual snippet for diagnostics
83 throw ConfigParseError(
84 formatError("Invalid value for '" + field + "': " + value, line, column),
85 context_provider());
86 }
87
88 return result;
89}

91/**
92 * @ingroup string_utils
93 * @brief Parses a human-readable byte size string into a byte count.
94 *
95 * @details Converts a size string (e.g., `"10K"`, `"5M"`, `"2G"`, or raw bytes `"512"`)
96 * into a `std::size_t` representing the number of bytes.
97 * The suffix, if present, is case-insensitive and supports:
98 * - `K` = kibibytes (× 1024)
99 * - `M` = mebibytes (× 1024²)
100 * - `G` = gibibytes (× 1024³)
101 * Throws a @ref ConfigParseError if:
102 * - The value is empty.
103 * - Parsing fails or extra non-numeric characters remain.
104 * - The computed size exceeds the 4 GiB hard limit from the Webserv spec.
105 *
106 * @param value String containing the size to parse (may include suffix).
107 * @param field Name of the configuration field being parsed.
108 * @param line Line number in the configuration source.
109 * @param column Column number in the configuration source.
110 * @param context_provider Callable returning a string snippet or context for diagnostics.
111 * @return Parsed size in bytes.
112 *
113 * @throws ConfigParseError If the string is empty, cannot be parsed as a size, contains
114 * leftover characters, or exceeds the maximum allowed size.
115 *
116 * @note Used primarily to parse configuration directives like `client_max_body_size`.
117 */
118std::size_t parseByteSize(const std::string& value, const std::string& field, int line, int column,
119 const std::function<std::string()>& context_provider) {
120 if (value.empty()) {
121 throw ConfigParseError(formatError("Empty size for '" + field + "'", line, column),
122 context_provider());
123 }
124
125 char suffix = value.back();
126 std::string number_part = value;
127 std::size_t multiplier = 1;
128
129 if (suffix == 'k' || suffix == 'K') {
130 multiplier = 1024;
131 number_part.pop_back();
132 } else if (suffix == 'm' || suffix == 'M') {
133 multiplier = 1024 * 1024;
134 number_part.pop_back();
135 } else if (suffix == 'g' || suffix == 'G') {
136 multiplier = 1024ULL * 1024ULL * 1024ULL;
137 number_part.pop_back();
138 }
139
140 std::size_t number = 0;
141 auto [ptr, ec] =
142 std::from_chars(number_part.data(), number_part.data() + number_part.size(), number);
143
144 if (ec != std::errc() || ptr != number_part.data() + number_part.size()) {
145 throw ConfigParseError(
146 formatError("Invalid size format for '" + field + "': " + value, line, column),
147 context_provider());
148 }
149
150 constexpr std::size_t MAX_BODY_SIZE_LIMIT = 4ULL * 1024 * 1024 * 1024;
151 if (number > MAX_BODY_SIZE_LIMIT / multiplier) {
152 throw ConfigParseError(
153 formatError("client_max_body_size exceeds maximum allowed (4GiB)", line, column),
154 context_provider());
155 }
156
157 return number * multiplier;
158}

160std::string toLower(const std::string& str) {
161 std::string result = str; // Create a mutable copy of the input string
162
163 // Transform each character in 'result' to lowercase in-place
164 std::transform(result.begin(), result.end(), result.begin(), [](unsigned char c) {
165 return std::tolower(c); // Convert character to lowercase using locale rules
166 });
167
168 return result; // Return the transformed string
169}

172 std::string result = str;

178std::string formatBytes(std::size_t bytes) {
179 const char* suffixes[] = {"B", "KiB", "MiB", "GiB", "TiB"};
180 double size = static_cast<double>(bytes);
181 int i = 0;
182
183 while (size >= 1024.0 && i < 4) {
184 size /= 1024.0;
185 ++i;
186 }
187
188 std::ostringstream oss;
189 oss.precision(2);
190 oss << std::fixed << size << ' ' << suffixes[i];
191 return oss.str();
192}

194std::string joinStrings(const std::vector<std::string>& list, const std::string& delim) {
195 std::ostringstream oss;
196 for (std::size_t i = 0; i < list.size(); ++i) {
197 oss << list[i];
198 if (i + 1 < list.size())
199 oss << delim;
200 }
201 return oss.str();
202}

204std::string trim(const std::string& s) {
205 size_t start = s.find_first_not_of(" \t\r\n");
206 size_t end = s.find_last_not_of(" \t\r\n");
207 return (start == std::string::npos || end == std::string::npos)
208 ? ""
209 : s.substr(start, end - start + 1);
210}
