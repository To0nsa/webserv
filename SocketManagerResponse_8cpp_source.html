13/**
14 * @file SocketManagerResponse.cpp
15 * @brief Implements response sending logic for SocketManager.
16 *
17 * @details
18 * This file contains all methods of @ref SocketManager related to
19 * **writing HTTP responses back to clients**:
20 * - Logging response status codes (@ref logResponseStatus).
21 * - Sending file-backed responses in chunks (@ref sendFileResponse).
22 * - Sending in-memory/raw responses (@ref sendRawResponse).
23 * - Driving the high-level send path for a client (@ref sendResponse).
24 *
25 * Responsibilities:
26 * - Stream headers and body data over non-blocking sockets.
27 * - Support both static files and CGI-generated temporary files.
28 * - Handle partial writes (tracking progress via `bytes_sent`).
29 * - Manage connection lifecycle (`keep-alive` vs `close`).
30 * - Ensure resource cleanup (closing file streams, unlinking CGI temp files).
31 *
32 * This file implements the **write-path** of the event loop,
33 * complementing @ref SocketManagerRequest.cpp which handles the read-path.
34 *
35 * @ingroup socket_manager
36 */

58/**
59 * @brief Logs the HTTP response status being sent to a client.
60 *
61 * @details
62 * - Formats a message indicating the status code and target file descriptor.
63 * - Logs at different severity levels depending on status code range:
64 * - `< 400`: Informational (success / redirection).
65 * - `400–499`: Warning (client error).
66 * - `>= 500`: Error (server error).
67 *
68 * This function is called before sending a response to provide
69 * visibility into the server’s behavior and help with debugging.
70 *
71 * @param status The HTTP status code of the response.
72 * @param fd The client file descriptor receiving the response.
73 */
74void SocketManager::logResponseStatus(int status, int fd) {
75 std::string message = "Sending HTTP " + std::to_string(status) + " → fd " + std::to_string(fd);
76 if (status < 400)
77 Logger::logFrom(LogLevel::INFO, "SocketManager sendResponse", message);
78 else if (status < 500)
79 Logger::logFrom(LogLevel::WARN, "SocketManager sendResponse", message);
80 else
81 Logger::logFrom(LogLevel::ERROR, "SocketManager sendResponse", message);
82}

84/**
85 * @brief Sends a file-based HTTP response to a client.
86 *
87 * @details
88 * Handles responses that stream a file (static resource or CGI temp file)
89 * to the client in a **non-blocking, incremental fashion**:
90 *
91 * Workflow:
92 * 1. **First call (headers not sent yet):**
93 * - Opens the response’s file path in binary mode.
94 * - If CGI body offset is set, seeks past CGI headers.
95 * - Builds and stores the HTTP status line and headers in
96 * `client.current_raw_response`.
97 *
98 * 2. **Send headers (if not fully sent yet):**
99 * - Writes remaining header bytes using `send(MSG_DONTWAIT)`.
100 * - Updates offset tracking (`bytes_sent`).
101 * - Returns early if partial write occurs (retry later).
102 *
103 * 3. **Send file body (chunked):**
104 * - Reads next chunk (8 KB) from the file into a buffer.
105 * - Sends it to the client socket.
106 * - Returns early if partial write occurs.
107 *
108 * 4. **Completion:**
109 * - On EOF (or 0 bytes read), closes file stream.
110 * - Clears `current_raw_response` and resets offset.
111 * - If response came from a CGI temp file, deletes it.
112 * - Pops the response from the queue.
113 * - Handles connection policy:
114 * - If `Connection: close` → closes client FD.
115 * - If keep-alive and no more responses → disables POLLOUT.
116 *
117 * Error handling:
118 * - On file open failure → queues a `500 Internal Server Error`.
119 * - On socket send failure → logs error, closes the connection.
120 *
121 * @param fd Client socket file descriptor.
122 * @param index Index of the client FD in the poll list.
123 * @param response The HTTP response object to send.
124 *
125 * @return `true` if more data remains to be sent,
126 * `false` if connection was closed or error occurred.
127 */
128bool SocketManager::sendFileResponse(int fd, size_t index, HttpResponse& response) {
129 ClientInfo& client = _client_info[fd];
130
131 // Step 1: open the file and build headers (first call)
132 if (!client.file_stream.is_open()) {
133 client.file_stream.open(response.getFilePath(), std::ios::binary);
134 if (!client.file_stream.is_open()) {
135 respondError(fd, 500);
136 return false;
137 }
138 if (response.getCgiBodyOffset() > 0)
139 client.file_stream.seekg(response.getCgiBodyOffset());
140
141 // Build HTTP response headers
142 std::ostringstream head;
143 head << "HTTP/1.1 " << response.getStatusCode() << " " << response.getStatusMessage()
144 << "\r\n";
145 for (const auto& header : response.getHeaders())
146 head << header.first << ": " << header.second << "\r\n";
147 head << "\r\n";
148
149 client.current_raw_response = head.str();
150 }
151
152 // Step 2: send HTTP headers if not finished
153 std::string& raw = client.current_raw_response;
154 size_t& offset = client.bytes_sent;
155 if (offset < raw.size()) {
156 ssize_t sent = send(fd, raw.c_str() + offset, raw.size() - offset, MSG_DONTWAIT);
157 if (sent < 0) {
158 Logger::logFrom(LogLevel::ERROR, "SocketManager",
159 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
160 cleanupClientConnectionClose(fd, index);
161 return false;
162 }
163 offset += sent;
164 client.lastSendAttemptTime = getCurrentTime();
165 if (offset < raw.size())
166 return true; // partial header write → retry later
167 }
168
169 // Step 3: stream file body in chunks
170 char buffer[8192];
171 client.file_stream.read(buffer, sizeof(buffer));
172 std::streamsize bytes_read = client.file_stream.gcount();
173 if (bytes_read > 0) {
174 ssize_t sent = send(fd, buffer, bytes_read, MSG_DONTWAIT);
175 if (sent < 0) {
176 Logger::logFrom(LogLevel::ERROR, "SocketManager",
177 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
178 cleanupClientConnectionClose(fd, index);
179 return false;
180 }
181 client.lastSendAttemptTime = getCurrentTime();
182 return true; // still more file left
183 }
184
185 // Step 4: completion (EOF reached)
186 if (client.file_stream.eof() || bytes_read == 0) {
187 Logger::logFrom(LogLevel::INFO, "SocketManager",
188 "[DONE] We sent full FILE RESPONSE to fd:" + std::to_string(fd));
189 client.file_stream.close();
190 client.current_raw_response.clear();
191 offset = 0;
192
193 // Cleanup CGI temp file if response came from CGI
194 if (response.isCgiTempFile()) {
195 CGI::unlinkWithErrorLog(response.getCgiTempFile(), "out temp file");
196 response.setCgiTempFile("");
197 }
198
199 // Pop response and handle connection policy
200 bool shouldClose = response.isConnectionClose();
201 client.responses.pop();
202 if (shouldClose) {
203 Logger::logFrom(LogLevel::INFO, "SocketManager",
204 "Connection: close - closing the connection");
205 cleanupClientConnectionClose(fd, index);
206 return false;
207 } else if (client.responses.empty()) {
208 Logger::logFrom(LogLevel::INFO, "SocketManager",
209 "Connection: keep-alive - keeping the connection open");
210 _poll_fds[index].events &= ~POLLOUT; // disable POLLOUT until more data
211 }
212 }
213
214 return true;
215}

217/**
218 * @brief Sends an in-memory (raw) HTTP response to a client.
219 *
220 * @details
221 * Handles responses that are already fully materialized in memory:
222 * 1) On first call, serialize the @ref HttpResponse to a byte string
223 * (`toHttpString()`) and reset the per-client send offset.
224 * 2) Perform non-blocking `send()` of the remaining bytes; update `bytes_sent`.
225 * 3) When all bytes are sent:
226 * - Pop the response from the queue,
227 * - Clear the staging buffer and reset offset,
228 * - Honor connection policy:
229 * * `Connection: close` → close FD,
230 * * keep-alive → if no more responses, drop POLLOUT to avoid busy loops.
231 *
232 * Error handling:
233 * - On `send()` failure, logs the error and closes the connection.
234 *
235 * @param fd Client socket file descriptor.
236 * @param index Index of the client FD in the poll list.
237 * @param response The response object to serialize and send.
238 *
239 * @return `true` if the socket remains usable (may need more sends),
240 * `false` if the connection was closed or a fatal error occurred.
241 */
242bool SocketManager::sendRawResponse(int fd, size_t index, HttpResponse& response) {
243 ClientInfo& client = _client_info[fd];
244 size_t& offset = client.bytes_sent;
245
246 // First invocation for this response: serialize headers + body into a single string.
247 if (client.current_raw_response.empty()) {
248 client.current_raw_response = response.toHttpString();
249 offset = 0;
250 }
251
252 // Attempt to send the remaining bytes (non-blocking).
253 std::string& raw = client.current_raw_response;
254 if (offset < raw.size()) {
255 ssize_t sent = send(fd, raw.c_str() + offset, raw.size() - offset, MSG_DONTWAIT);
256 if (sent < 0) {
257 Logger::logFrom(LogLevel::ERROR, "SocketManager",
258 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
259 cleanupClientConnectionClose(fd, index);
260 return false;
261 }
262 offset += sent;
263 client.lastSendAttemptTime = getCurrentTime();
264 }
265
266 // If everything is sent, finalize and apply connection policy.
267 if (offset >= raw.size()) {
268 bool shouldClose = response.isConnectionClose();
269 client.responses.pop();
270 offset = 0;
271 Logger::logFrom(LogLevel::INFO, "SocketManager",
272 "[DONE] We sent full RESPONSE to fd:" + std::to_string(fd));
273 client.current_raw_response.clear();
274
275 if (shouldClose) {
276 Logger::logFrom(LogLevel::INFO, "SocketManager",
277 "Connection: close - closing the connection");
278 cleanupClientConnectionClose(fd, index);
279 return false;
280 } else if (client.responses.empty()) {
281 // No more data to write → stop polling for POLLOUT to avoid spin.
282 Logger::logFrom(LogLevel::INFO, "SocketManager",
283 "Connection: keep-alive - keeping the connection open");
284 _poll_fds[index].events &= ~POLLOUT;
285 }
286 }
287
288 return true;
289}

291/**
292 * @brief Dispatches sending of the next queued response for a client.
293 *
294 * @details
295 * Retrieves the front @ref HttpResponse for the given client, logs its status
296 * via @ref logResponseStatus, then delegates to the appropriate write path:
297 * - File-backed response → @ref sendFileResponse (streams headers + file body).
298 * - In-memory response → @ref sendRawResponse (single serialized buffer).
299 *
300 * Both send paths are **non-blocking**; they may return early when the socket
301 * can’t accept more bytes. Any fatal error (including allocation failure) is
302 * caught here; the client connection is then closed to keep the server healthy.
303 *
304 * @param client_fd Client socket file descriptor.
305 * @param index Index of the client FD in the poll list.
306 *
307 * @note This function assumes there is at least one queued response for
308 * `client_fd`. Callers should check the queue before enabling POLLOUT.
309 */
310void SocketManager::sendResponse(int client_fd, size_t index) {
311 try {
312 // Peek the next response to send for this client.
313 HttpResponse& response = _client_info[client_fd].responses.front();
314
315 // Log status code with severity (info/warn/error).
316 logResponseStatus(response.getStatusCode(), client_fd);
317
318 // Choose write path: file-backed (streamed) vs raw (in-memory).
319 if (response.isFileResponse()) {
320 if (!sendFileResponse(client_fd, index, response))
321 return; // connection closed or fatal error handled inside
322 } else {
323 if (!sendRawResponse(client_fd, index, response))
324 return; // connection closed or fatal error handled inside
325 }
326 return; // keep connection; more data may remain for later POLLOUT
327 } catch (const std::bad_alloc& e) {
328 Logger::logFrom(LogLevel::ERROR, "SocketManager",
329 "Fatal memory allocation error while sending response to fd " +
330 std::to_string(client_fd));
331 } catch (const std::exception& e) {
332 // Any recoverable runtime error while sending this response.
333 Logger::logFrom(LogLevel::ERROR, "SocketManager",
334 "Exception in sendResponse for fd " + std::to_string(client_fd) + ": " +
335 e.what());
336 } catch (...) {
337 // Last-resort safety net.
338 Logger::logFrom(LogLevel::ERROR, "SocketManager",
339 "Unknown fatal error in sendResponse for fd " + std::to_string(client_fd));
340 }
341 // On fatal errors, close the client cleanly to avoid undefined state.
342 cleanupClientConnectionClose(client_fd, index);
343}
