43bool SocketManager::sendFileResponse(int fd, size_t index, HttpResponse& response) {
44 ClientInfo& client = _client_info[fd];
45
46 if (!client.file_stream.is_open()) {
47 client.file_stream.open(response.getFilePath(), std::ios::binary);
48 if (!client.file_stream.is_open()) {
49 respondError(fd, 500);
50 return false;
51 }
52 if (response.getCgiBodyOffset() > 0)
53 client.file_stream.seekg(response.getCgiBodyOffset());
54
55 std::ostringstream head;
56 head << "HTTP/1.1 " << response.getStatusCode() << " " << response.getStatusMessage()
57 << "\r\n";
58 for (const auto& header : response.getHeaders())
59 head << header.first << ": " << header.second << "\r\n";
60 head << "\r\n";
61 client.current_raw_response = head.str();
62 }
63
64 std::string& raw = client.current_raw_response;
65 size_t& offset = client.bytes_sent;
66 if (offset < raw.size()) {
67 ssize_t sent = send(fd, raw.c_str() + offset, raw.size() - offset, MSG_DONTWAIT);
68 if (sent < 0) {
69 Logger::logFrom(LogLevel::ERROR, "SocketManager",
70 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
71 cleanupClientConnectionClose(fd, index);
72 return false;
73 }
74 offset += sent;
75 client.lastSendAttemptTime = getCurrentTime();
76 if (offset < raw.size())
77 return true;
78 }
79
80 char buffer[8192];
81 client.file_stream.read(buffer, sizeof(buffer));
82 std::streamsize bytes_read = client.file_stream.gcount();
83 if (bytes_read > 0) {
84 ssize_t sent = send(fd, buffer, bytes_read, MSG_DONTWAIT);
85 if (sent < 0) {
86 Logger::logFrom(LogLevel::ERROR, "SocketManager",
87 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
88 cleanupClientConnectionClose(fd, index);
89 return false;
90 }
91 client.lastSendAttemptTime = getCurrentTime();
92 return true;
93 }
94
95 if (client.file_stream.eof() || bytes_read == 0) {
96 Logger::logFrom(LogLevel::INFO, "SocketManager",
97 "[DONE] We sent full FILE RESPONSE to fd:" + std::to_string(fd));
98 client.file_stream.close();
99 client.current_raw_response.clear();
100 offset = 0;
101
102 if (response.isCgiTempFile()) {
103 CGI::unlinkWithErrorLog(response.getCgiTempFile(), "out temp file");
104 response.setCgiTempFile("");
105 }
106 bool shouldClose = response.isConnectionClose();
107 client.responses.pop();
108 if (shouldClose) {
109 Logger::logFrom(LogLevel::INFO, "SocketManager",
110 "Connection: close - closing the connection");
111 cleanupClientConnectionClose(fd, index);
112 return false;
113 } else if (client.responses.empty()) {
114 Logger::logFrom(LogLevel::INFO, "SocketManager",
115 "Connection: keep-alive - keeping the connection open");
116 _poll_fds[index].events &= ~POLLOUT; // Disable POLLOUT for keep-alive
117 }
118 }
119
120 return true;
121}

123bool SocketManager::sendRawResponse(int fd, size_t index, HttpResponse& response) {
124 ClientInfo& client = _client_info[fd];
125 size_t& offset = client.bytes_sent;
126
127 if (client.current_raw_response.empty()) {
128 client.current_raw_response = response.toHttpString();
129 offset = 0;
130 }
131
132 std::string& raw = client.current_raw_response;
133 if (offset < raw.size()) {
134 ssize_t sent = send(fd, raw.c_str() + offset, raw.size() - offset, MSG_DONTWAIT);
135 if (sent < 0) {
136 Logger::logFrom(LogLevel::ERROR, "SocketManager",
137 "send() failed on fd " + std::to_string(fd) + ": " + strerror(errno));
138 cleanupClientConnectionClose(fd, index);
139 return false;
140 }
141 offset += sent;
142 client.lastSendAttemptTime = getCurrentTime();
143 }
144
145 if (offset >= raw.size()) {
146 bool shouldClose = response.isConnectionClose();
147 client.responses.pop();
148 offset = 0;
149 Logger::logFrom(LogLevel::INFO, "SocketManager",
150 "[DONE] We sent full RESPONSE to fd:" + std::to_string(fd));
151 client.current_raw_response.clear();
152 if (shouldClose) {
153 Logger::logFrom(LogLevel::INFO, "SocketManager",
154 "Connection: close - closing the connection");
155 cleanupClientConnectionClose(fd, index);
156 return false;
157 } else if (client.responses.empty()) {
158 Logger::logFrom(LogLevel::INFO, "SocketManager",
159 "Connection: keep-alive - keeping the connection open");
160 _poll_fds[index].events &= ~POLLOUT; // Disable POLLOUT for keep-alive
161 }
162 }
163
164 return true;
165}

167void SocketManager::sendResponse(int client_fd, size_t index) {
168 try {
169 HttpResponse& response = _client_info[client_fd].responses.front();
170
171 logResponseStatus(response.getStatusCode(), client_fd);
172
173 if (response.isFileResponse()) {
174 if (!sendFileResponse(client_fd, index, response))
175 return;
176 } else {
177 if (!sendRawResponse(client_fd, index, response))
178 return;
179 }
180 return;
181 } catch (const std::bad_alloc& e) {
182 Logger::logFrom(LogLevel::ERROR, "SocketManager",
183 "Fatal memory allocation error while sending response to fd " +
184 std::to_string(client_fd));
185 } catch (const std::exception& e) {
186 Logger::logFrom(LogLevel::ERROR, "SocketManager",
187 "Exception in sendResponse for fd " + std::to_string(client_fd) + ": " +
188 e.what());
189 } catch (...) {
190 Logger::logFrom(LogLevel::ERROR, "SocketManager",
191 "Unknown fatal error in sendResponse for fd " + std::to_string(client_fd));
192 }
193 cleanupClientConnectionClose(client_fd, index);
194}
