56/**
57 * @brief Attempts to delete a file safely, producing an error response on failure.
58 *
59 * @details This helper encapsulates the low-level checks and filesystem
60 * operations for handling HTTP `DELETE` requests. It verifies:
61 * - Existence of the target (returns 404 if not found).
62 * - That the target is **not** a directory or special file (returns 403).
63 * - That the target is a regular file (otherwise rejected with 403).
64 * - Filesystem deletion via `std::filesystem::remove`, with
65 * detailed error mapping:
66 * - EACCES / EPERM → 403 Forbidden
67 * - ENOENT (file vanished) → 404 Not Found
68 * - Other errors → 500 Internal Server Error
69 *
70 * On success, the file is removed and the function returns `true`.
71 * On failure, `outError` is populated with an appropriate
72 * @ref HttpResponse.
73 *
74 * @param filepath Full path to the file to delete.
75 * @param req Original HTTP request (used in error responses).
76 * @param server Active server context (used in error responses).
77 * @param outError Populated with a generated error response if deletion fails.
78 *
79 * @return `true` if the file was successfully unlinked, `false` otherwise.
80 *
81 * @ingroup request_handler
82 */
83bool unlinkFile(const std::string& filepath, const HttpRequest& req, const Server& server,
84 HttpResponse& outError) {
85 struct stat st;
86
87 // 1) Stat the file: reject if not found
88 if (stat(filepath.c_str(), &st) != 0) {
89 Logger::logFrom(LogLevel::WARN, "Delete Handler",
90 "File not found → rejecting DELETE: " + filepath);
91 outError = ResponseBuilder::generateError(404, server, req);
92 return false;
93 }
94
95 // 2) Reject directories outright
96 if (S_ISDIR(st.st_mode)) {
97 Logger::logFrom(LogLevel::WARN, "Delete Handler",
98 "Target is a directory → rejecting DELETE: " + filepath);
99 outError = ResponseBuilder::generateError(403, server, req);
100 return false;
101 }
102
103 // 3) Reject non-regular files (devices, sockets, etc.)
104 if (!S_ISREG(st.st_mode)) {
105 Logger::logFrom(LogLevel::WARN, "Delete Handler",
106 "Target is not a regular file → rejecting DELETE: " + filepath);
107 outError = ResponseBuilder::generateError(403, server, req);
108 return false;
109 }
110
111 // 4) Attempt deletion with detailed error handling
112 std::error_code ec;
113 if (!std::filesystem::remove(filepath, ec)) {
114 if (ec.value() == EACCES || ec.value() == EPERM) {
115 Logger::logFrom(LogLevel::WARN, "Delete Handler",
116 "Permission denied when deleting: " + filepath + " (" + ec.message() +
117 ")");
118 outError = ResponseBuilder::generateError(403, server, req);
119 } else if (ec.value() == ENOENT) {
120 Logger::logFrom(LogLevel::WARN, "Delete Handler",
121 "File disappeared before deletion: " + filepath);
122 outError = ResponseBuilder::generateError(404, server, req);
123 } else {
124 Logger::logFrom(LogLevel::WARN, "Delete Handler",
125 "Unexpected error deleting file: " + filepath + " (" + ec.message() +
126 ")");
127 outError = ResponseBuilder::generateError(500, server, req);
128 }
129 return false;
130 }
131
132 // Success: file removed
133 return true;
134}

136/**
137 * @brief Generates a simple HTML confirmation page for a deleted file.
138 *
139 * @details Builds a minimal HTML5 document containing a title and message
140 * confirming that the requested file has been successfully deleted.
141 * The filename is HTML-escaped before insertion to prevent XSS.
142 *
143 * @param filename Name of the deleted file (not a path). Will be sanitized
144 * via @ref htmlEscape before being injected into the HTML.
145 *
146 * @return A fully-formed HTML string suitable as the body of a 200 OK response.
147 *
148 * @ingroup request_handler
149 */
150std::string generateDeleteHtml(const std::string& filename) {
151 std::stringstream html;
152 html << R"(<!DOCTYPE html>
153<html lang="en">
154<head>
155 <meta charset="UTF-8">
156 <title>Deleted: )"
157 << htmlEscape(filename) << R"(</title>
158</head>
159<body>
160 <div class="container">
161 <h1>File )"
162 << htmlEscape(filename) << R"( deleted.</h1>
163 <p>The requested file has been successfully removed.</p>
164 </div>
165</body>
166</html>)";
167 return html.str();
168}

172/**
173 * @brief Handles an HTTP DELETE request for a given resource.
174 *
175 * @details This function enforces server security rules and attempts
176 * to remove the requested file if permitted:
177 * 1. Resolve the request path to a physical filesystem path.
178 * 2. Reject empty resolutions (403).
179 * 3. Reject symlinks (403) to prevent symlink attacks.
180 * 4. Reject directory-like URIs (403) to avoid recursive deletion.
181 * 5. Attempt file deletion with @ref unlinkFile, which validates
182 * type, permissions, and maps errors to proper responses.
183 * 6. On success, build and return a 200 OK response containing
184 * a confirmation HTML page (via @ref generateDeleteHtml).
185 *
186 * @param req The incoming HTTP request object.
187 * @param server Active server context.
188 * @param loc The matched location block from the configuration.
189 *
190 * @return A fully constructed @ref HttpResponse:
191 * - 200 OK with HTML confirmation if deletion succeeds.
192 * - Error response (403, 404, or 500) on failure.
193 *
194 * @ingroup request_handler
195 */
196HttpResponse handleDelete(const HttpRequest& req, const Server& server, const Location& loc) {
197 // 1) Resolve the requested URI to a filesystem path
198 std::string path = resolvePhysicalPath(req, loc);
199 if (path.empty()) {
200 Logger::logFrom(LogLevel::WARN, "Delete Handler",
201 "Path is empty → rejecting DELETE for URI: " + req.getPath());
202 return ResponseBuilder::generateError(403, server, req);
203 }
204
205 // 2) Reject symbolic links to avoid symlink traversal exploits
206 if (isSymlink(path)) {
207 Logger::logFrom(LogLevel::WARN, "Delete Handler",
208 "Target is a symlink → rejecting DELETE for URI: " + req.getPath());
209 return ResponseBuilder::generateError(403, server, req);
210 }
211
212 // 3) Reject URIs ending with '/' (treated as directories)
213 if (req.getPath().back() == '/') {
214 Logger::logFrom(LogLevel::WARN, "Delete Handler",
215 "Request URI ends with '/' → rejecting DELETE for directory-like path: " +
216 req.getPath());
217 return ResponseBuilder::generateError(403, server, req);
218 }
219
220 // 4) Try to delete the file; errors are mapped to proper HTTP responses
221 HttpResponse errResp;
222 if (!unlinkFile(path, req, server, errResp)) {
223 return errResp;
224 }
225
226 // 5) On success: build a sanitized confirmation page
227 std::string rawName = extractFilenameFromUri(req.getPath());
228 std::string safeName = htmlEscape(rawName); // escape to prevent XSS
229 std::string body = generateDeleteHtml(safeName); // build confirmation HTML
230 Logger::logFrom(LogLevel::INFO, "Delete Handler",
231 "Successfully deleted “" + safeName + "” → sending HTML confirmation");
232
233 // 6) Return a 200 OK response with confirmation body
234 return ResponseBuilder::generateSuccess(200, body, "text/html", req);
235}
