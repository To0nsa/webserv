31bool unlinkFile(const std::string& filepath, const HttpRequest& req, const Server& server,
32 HttpResponse& outError) {
33 struct stat st;
34 if (stat(filepath.c_str(), &st) != 0) {
35 Logger::logFrom(LogLevel::WARN, "Delete Handler",
36 "File not found → rejecting DELETE: " + filepath);
37 outError = ResponseBuilder::generateError(404, server, req);
38 return false;
39 }
40 if (S_ISDIR(st.st_mode)) {
41 Logger::logFrom(LogLevel::WARN, "Delete Handler",
42 "Target is a directory → rejecting DELETE: " + filepath);
43 outError = ResponseBuilder::generateError(403, server, req);
44 return false;
45 }
46 if (!S_ISREG(st.st_mode)) {
47 Logger::logFrom(LogLevel::WARN, "Delete Handler",
48 "Target is not a regular file → rejecting DELETE: " + filepath);
49 outError = ResponseBuilder::generateError(403, server, req);
50 return false;
51 }
52
53 std::error_code ec;
54 if (!std::filesystem::remove(filepath, ec)) {
55 if (ec.value() == EACCES || ec.value() == EPERM) {
56 Logger::logFrom(LogLevel::WARN, "Delete Handler",
57 "Permission denied when deleting: " + filepath + " (" + ec.message() +
58 ")");
59 outError = ResponseBuilder::generateError(403, server, req);
60 } else if (ec.value() == ENOENT) {
61 Logger::logFrom(LogLevel::WARN, "Delete Handler",
62 "File disappeared before deletion: " + filepath);
63 outError = ResponseBuilder::generateError(404, server, req);
64 } else {
65 Logger::logFrom(LogLevel::WARN, "Delete Handler",
66 "Unexpected error deleting file: " + filepath + " (" + ec.message() +
67 ")");
68 outError = ResponseBuilder::generateError(500, server, req);
69 }
70 return false;
71 }
72
73 return true;
74}

98HttpResponse handleDelete(const HttpRequest& req, const Server& server, const Location& loc) {
99 std::string path = resolvePhysicalPath(req, loc);
100 if (path.empty()) {
101 Logger::logFrom(LogLevel::WARN, "Delete Handler",
102 "Path is empty → rejecting DELETE for URI: " + req.getPath());
103 return ResponseBuilder::generateError(403, server, req);
104 }
105
106 if (isSymlink(path)) {
107 Logger::logFrom(LogLevel::WARN, "Delete Handler",
108 "Target is a symlink → rejecting DELETE for URI: " + req.getPath());
109 return ResponseBuilder::generateError(403, server, req);
110 }
111
112 if (req.getPath().back() == '/') {
113 Logger::logFrom(LogLevel::WARN, "Delete Handler",
114 "Request URI ends with '/' → rejecting DELETE for directory-like path: " +
115 req.getPath());
116 return ResponseBuilder::generateError(403, server, req);
117 }
118
119 HttpResponse errResp;
120 if (!unlinkFile(path, req, server, errResp)) {
121 return errResp;
122 }
123
124 std::string rawName = extractFilenameFromUri(req.getPath());
125 std::string safeName = htmlEscape(rawName);
126 std::string body = generateDeleteHtml(safeName);
127 Logger::logFrom(LogLevel::INFO, "Delete Handler",
128 "Successfully deleted “" + safeName + "” → sending HTML confirmation");
129 return ResponseBuilder::generateSuccess(200, body, "text/html", req);
130}
