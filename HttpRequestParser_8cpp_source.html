13/**
14 * @file HttpRequestParser.cpp
15 * @brief Implements the HttpRequestParser class and HTTP request parsing logic.
16 *
17 * @details This file defines the parsing pipeline that converts raw client
18 * input into structured @ref HttpRequest objects:
19 *
20 * - **Request line parsing**: method, request-target, and HTTP version.
21 * - **Header validation**: validates and inserts headers, handling
22 * conflicts between `Content-Length` and `Transfer-Encoding`.
23 * - **Path normalization**: decodes percent-encoding, strips fragments,
24 * validates against traversal attempts, and normalizes paths.
25 * - **Server matching**: selects the most appropriate @ref Server
26 * configuration from the available virtual hosts, based on the Host header.
27 * - **Body parsing**: supports both fixed-length bodies (via
28 * `Content-Length`) and chunked transfer encoding, enforcing
29 * `client_max_body_size` limits.
30 * - **Validation**: ensures method support, mandatory headers,
31 * and content type restrictions for POST requests.
32 *
33 * Utility functions inside anonymous namespaces support RFC-compliant
34 * token validation, path checking, URL parsing, and body handling.
35 *
36 * The main entry point is @ref HttpRequestParser::parse, which integrates
37 * all steps to safely parse complete requests, handle errors, and detect
38 * pipelined requests.
39 *
40 * @ingroup http
41 */

66/**
67 * @brief Validates whether a string is a valid HTTP method token.
68 *
69 * @details According to RFC 7230 §3.1.1, HTTP methods are case-sensitive
70 * tokens that may include letters, digits, and a limited set
71 * of special characters:
72 *
73 * ```
74 * token = 1*tchar
75 * tchar = "!" / "#" / "$" / "%" / "&" / "'" / "*"
76 * / "+" / "-" / "." / "^" / "_" / "`" / "|" / "~"
77 * / DIGIT / ALPHA
78 * ```
79 *
80 * This function checks that:
81 * - The method is non-empty.
82 * - Every character is alphanumeric or belongs to the allowed set.
83 *
84 * @param method The candidate method string (e.g. "GET", "POST", "PUT").
85 * @return true if the string is a valid HTTP method token, false otherwise.
86 */
87static bool isValidHttpMethodToken(const std::string& method) {
88 if (method.empty())
89 return false;
90
91 for (char c : method) {
92 unsigned char uc = static_cast<unsigned char>(c);
93 // Reject if not alphanumeric and not in the allowed special characters
94 if (!std::isalnum(uc) && c != '!' && c != '#' && c != '$' && c != '%' && c != '&' &&
95 c != '\'' && c != '*' && c != '+' && c != '-' && c != '.' && c != '^' && c != '_' &&
96 c != '`' && c != '|' && c != '~') {
97 return false;
98 }
99 }
100 return true;
101}

103/**
104 * @brief Validates that a normalized request path is safe and well-formed.
105 *
106 * @details This function enforces basic security and syntax rules for
107 * HTTP request targets:
108 * - Must start with a leading `/` (absolute path).
109 * - The root path `/` is always valid.
110 * - Rejects empty paths or those not beginning with `/`.
111 * - Rejects traversal attempts such as `/..`, `/../foo`, or paths ending in `/..`.
112 * - Rejects sequences that attempt to escape the root directory.
113 *
114 * Note: Double slashes (`"//"`) were considered for rejection but are
115 * currently allowed (commented out).
116 *
117 * @param rawPath The request path string (after decoding and normalization).
118 * @return true if the path is valid and safe, false otherwise.
119 */
120static bool isValidPath(const std::string& rawPath) {
121 fs::path p(rawPath);
122 std::string s = p.string();
123
124 if (s == "/")
125 return true; // root path is always valid
126 if (s.empty() || s.front() != '/')
127 return false; // must start with "/"
128
129 // Optional stricter check for double slashes (currently disabled)
130 /* if (s.find("//") != std::string::npos)
131 return false; */
132
133 // Reject directory traversal attempts
134 if (s == "/.." || s.find("/../") != std::string::npos || s.ends_with("/.."))
135 return false;
136
137 return true;
138}

140/**
141 * @brief Parses a full URL string into its components (HTTP/1.1 version).
142 *
143 * @details This function is specialized for HTTP/1.1 requests:
144 * - Verifies that the mandatory `Host` header is present when
145 * the request version is `HTTP/1.1` (RFC 7230 §5.4).
146 * - Uses a regular expression to parse the URL into its components:
147 * - Scheme (e.g., `http`, `https`)
148 * - User (optional username)
149 * - Password (optional password)
150 * - Host (domain or IP address)
151 * - Port (optional numeric port)
152 * - Path (resource path, e.g., `/index.html`)
153 * - Query (optional `?param=value` part)
154 * - Fragment (optional `#section` part)
155 * - Throws `std::invalid_argument` if the URL is invalid.
156 *
157 * @param req The associated @ref HttpRequest, used to validate presence of
158 * the `Host` header when using HTTP/1.1.
159 * @param url The raw URL string extracted from the request line.
160 *
161 * @return A populated @ref Url structure containing parsed fields.
162 *
163 * @throws std::invalid_argument If the `Host` header is missing (HTTP/1.1)
164 * or the URL does not conform to the expected syntax.
165 */
166static Url parseUrlHttpVersion1_1(HttpRequest& req, const std::string& url) {
167 if (req.getVersion() == "HTTP/1.1" && req.getHeader("HOST").empty()) {
168 throw std::invalid_argument("Missing HOST header (required in HTTP/1.1)");
169 }
170 Url res;
171 static const std::regex re(
172 R"((https?://)?(?:([^:@]+)(?::([^:@]*))?@)?([^:/?#]+)(?::(\d+))?(/[^?#]*)?(?:\?([^#]*))?(?:#(.*))?)");
173 std::smatch m;
174 if (!std::regex_match(url, m, re)) {
175 throw std::invalid_argument("Invalid URL");
176 }
177
178 // Assign captured groups to URL components
179 res.scheme = m[1].str();
180 res.user = m[2].str();
181 res.password = m[3].str();
182 res.host = m[4].str();
183 res.port = m[5].str();
184 res.path = m[6].str();
185 res.query = m[7].str();
186 res.fragment = m[8].str();
187
188 return res;
189}

193 *
194 * @details This function uses a regular expression to extract standard
195 * URL parts from a string. It supports optional components such as:
196 * - Scheme (`http`, `https`)
197 * - User credentials (`user:password`)
198 * - Hostname
199 * - Port
200 * - Path
201 * - Query string
202 * - Fragment identifier
203 *
204 * Example:
205 * ```

214 * // u.fragment = "frag"

216 *
217 * @param url A URL string to parse. It may be absolute or relative,
218 * but must match the expected format.
219 *
220 * @throws std::invalid_argument If the string does not conform to the
221 * URL grammar.
222 *
223 * @return A populated @ref Url struct containing the parsed components.
224 *
225 * @ingroup http
226 */
227static Url parseUrl(const std::string& url) {
228 Url res;
229 static const std::regex re(
230 R"((https?://)?(?:([^:@]+)(?::([^:@]*))?@)?([^:/?#]+)(?::(\d+))?(/[^?#]*)?(?:\?([^#]*))?(?:#(.*))?)");
231 std::smatch m;
232 if (!std::regex_match(url, m, re)) {
233 throw std::invalid_argument("Invalid URL");
234 }
235 res.scheme = m[1].str();
236 res.user = m[2].str();
237 res.password = m[3].str();
238 res.host = m[4].str();
239 res.port = m[5].str();
240 res.path = m[6].str();
241 res.query = m[7].str();

247/**
248 * @brief Selects the best-matching server configuration for a given Host header.
249 *
250 * @details This function determines which @ref Server instance should handle
251 * an incoming request based on the `Host` header:
252 * - If the header includes a port (e.g. `"example.com:8080"`),
253 * the port portion is ignored when matching.
254 * - The function iterates over all provided servers and checks whether
255 * the hostname matches any configured server name (via
256 * @ref Server::hasServerName).
257 * - If a match is found, that server is returned.
258 * - If no match is found, the first server in the list is returned as
259 * the default fallback.
260 *
261 * Logging is performed at INFO level to record whether a match
262 * was found or if the fallback server is being used.
263 *
264 * @param servers List of @ref Server objects available on the current port.
265 * @param hostHeader The raw `Host` header string from the request
266 * (may include a port suffix, e.g. `"example.com:8080"`).
267 *
268 * @return A reference to the best-matching @ref Server configuration.
269 *
270 * @note The caller must ensure that `servers` is non-empty.
271 *
272 * @ingroup http
273 */
274static const Server& searchBestMatchedServers(std::vector<Server>& servers,
275 const std::string& hostHeader) {
276 // Extract hostname (strip port if present)
277 std::string hostname = hostHeader;
278 size_t colonPos = hostHeader.find(':');
279 if (colonPos != std::string::npos) {
280 hostname = hostHeader.substr(0, colonPos); // ignore port in Host format "host:port"
281 }
282
283 for (Server& server : servers) {
284 if (server.hasServerName(hostname)) {
285 Logger::logFrom(LogLevel::INFO, "HttpRequestParser",
286 "Found matching server for Host: " + hostname + " on port " +
287 std::to_string(server.getPort()));
288 return server;
289 }
290 }
291
292 Logger::logFrom(LogLevel::INFO, "HttpRequestParser",
293 "No specific match found, returning default server for Host: " +
294 servers[0].getHost() + " on port " + std::to_string(servers[0].getPort()));
295 return servers[0]; // fallback
296}

299/**
300 * @brief Validates and inserts a single HTTP header into the request.
301 *
302 * @details This function performs centralized header validation and insertion
303 * according to HTTP/1.1 rules and RFC 7230/7231:
304 * - Checks header name validity (non-empty, no control chars).
305 * - Rejects empty values.
306 * - Special validation for `Content-Length`, `Transfer-Encoding`,
307 * and `Expect` headers.
308 * - Enforces conflict rules (e.g., `Content-Length` vs
309 * `Transfer-Encoding`).
310 * - Prevents duplicates for non-mergeable headers (`Host`,
311 * `Content-Type`, etc.).
312 * - Merges values for mergeable headers (per RFC 7230 §3.2.2).
313 *
314 * On error, an appropriate HTTP status code is set in
315 * `errorCode` (e.g., 400 Bad Request, 411 Length Required,
316 * 417 Expectation Failed).
317 *
318 * @param req Reference to the @ref HttpRequest being populated.
319 * @param key Header field name (case-insensitive).
320 * @param value Header field value.
321 * @param errorCode Output parameter: set to HTTP error code if validation fails.
322 *
323 * @return `true` if the header was successfully validated and inserted,
324 * `false` otherwise (with `errorCode` set).
325 *
326 * @ingroup http
327 */
328bool insertValidatedHeader(HttpRequest& req, const std::string& key, const std::string& value,
329 int& errorCode) {
330 std::string normKey = toUpper(key);
331
332 // 1) Validate header name: must not be empty or contain control chars
333 if (normKey.empty() || std::any_of(normKey.begin(), normKey.end(), [](char c) {
334 unsigned char uc = static_cast<unsigned char>(c);
335 return uc <= 0x1F || uc == 0x7F;
336 })) {
337 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
338 "Invalid or empty header name: " + normKey);
339 errorCode = 400;
340 return false;
341 }
342
343 // 2) Reject empty header values
344 if (value.empty()) {
345 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Header missing value: " + normKey);
346 errorCode = 400;
347 return false;
348 }
349
350 bool first = !req.hasHeader(normKey);
351
352 // 3) Special case: Content-Length must be numeric and unique
353 if (normKey == "CONTENT-LENGTH") {
354 if (!std::all_of(value.begin(), value.end(),
355 [](char c) { return std::isdigit(static_cast<unsigned char>(c)); })) {
356 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
357 "Invalid Content-Length value: " + value);
358 errorCode = 411;
359 return false;
360 }
361 unsigned long long len;
362 try {
363 len = std::stoull(value);
364 } catch (...) {
365 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
366 "Invalid Content-Length value: " + value);
367 errorCode = 411;
368 return false;
369 }
370 req.setContentLength(len);
371 }
372
373 // 4) Special case: Expect header → reject 100-continue
374 if (normKey == "EXPECT") {
375 std::string lower = toLower(value);
376 if (lower == "100-continue") {
377 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
378 "Unsupported Expect header: " + value);
379 errorCode = 417;
380 return false;
381 }
382 }
383
384 // 5) Special case: Transfer-Encoding → only "chunked" allowed
385 if (normKey == "TRANSFER-ENCODING") {
386 std::string lower = toLower(value);
387 if (lower != "chunked") {
388 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
389 "Unsupported Transfer-Encoding: " + lower);
390 errorCode = 501;
391 return false;
392 }
393 if (req.getMethod() == "GET") {
394 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
395 "Chunked Transfer-Encoding not allowed for GET");
396 errorCode = 400;
397 return false;
398 }
399 }
400
401 // 6) Check for Content-Length ↔ Transfer-Encoding conflict
402 if (first) {
403 if ((normKey == "CONTENT-LENGTH" && !req.getHeader("TRANSFER-ENCODING").empty()) ||
404 (normKey == "TRANSFER-ENCODING" && req.getContentLength() > 0)) {
405 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
406 "Both Content-Length and Transfer-Encoding present");
407 errorCode = 400;
408 return false;
409 }
410 }
411
412 // 7) Handle duplicates: some headers must not be repeated
413 static const std::set<std::string> nonMergeable = {
414 "HOST", "CONTENT-LENGTH", "CONTENT-TYPE", "TRANSFER-ENCODING", "EXPECT", "CONNECTION"};
415
416 if (!first) {
417 if (nonMergeable.count(normKey)) {
418 // Special case: identical Content-Length values are tolerated
419 if (normKey == "CONTENT-LENGTH") {
420 if (req.getHeader(normKey) != value) {
421 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
422 "Conflicting Content-Length headers");
423 errorCode = 400;
424 return false;
425 }
426 return true;
427 }
428 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
429 "Duplicate " + normKey + " header");
430 errorCode = 400;
431 return false;
432 }
433
434 // Otherwise, merge values (e.g., Accept: a, b)
435 req.setHeader(normKey, req.getHeader(normKey) + ", " + value);
436 } else {
437 req.setHeader(normKey, value);
438 }
439
440 return true;
441}

443/**
444 * @brief Parses and validates the request line and headers of an HTTP request.
445 *
446 * @details This function processes the "head" section of an HTTP request,
447 * which includes the request line (method, target, version) and all
448 * header fields. It performs:
449 *
450 * - Validation of the request line:
451 * • Ensures it contains exactly 3 fields (method, target, version).
452 * • Validates that the method token is syntactically valid.
453 * • Rejects empty fields.
454 * - Decoding and validation of the request-target:
455 * • Percent-decoding.
456 * • Rejects control characters.
457 * • Forbids absolute-URIs and fragments (`#...`).
458 * • Splits query parameters from the path.
459 * • Enforces maximum URI length.
460 * • Normalizes the path and blocks traversal attempts (`..`).
461 * - Header parsing:
462 * • Reads each header line, ensures proper format (`key: value`).
463 * • Delegates validation/merging logic to @ref insertValidatedHeader.
464 * - URL building:
465 * • Uses the `Host` header and matched @ref Server configuration.
466 * • Sets normalized host and stores full @ref Url in the request.
467 *
468 * On error, this function sets `errorCode` to the appropriate HTTP
469 * status code (e.g., 400, 403, 414, 505) and returns `false`.
470 *
471 * @param req Reference to the @ref HttpRequest being populated.
472 * @param headerPart Raw string containing the request line + headers.
473 * @param errorCode Output parameter: set if parsing fails.
474 * @param servers List of servers listening on the same port (for
475 * Host/ServerName matching).
476 * @param clientMaxBodySize Output parameter: max body size derived from the
477 * selected server configuration.
478 *
479 * @return `true` if parsing succeeds and `req` is populated,
480 * `false` otherwise (with `errorCode` set).
481 *
482 * @ingroup http
483 */
484bool parseReqHeader(HttpRequest& req, const std::string& headerPart, int& errorCode,
485 std::vector<Server> servers, size_t& clientMaxBodySize) {
486 std::istringstream stream(headerPart);
487 std::string line;
488
489 // 1) Parse request line (must have exactly 3 parts)
490 if (!std::getline(stream, line) || line.empty()) {
491 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
492 "Empty or missing request start line");
493 errorCode = 400;
494 return false;
495 }
496 size_t sp1 = line.find(' ');
497 size_t sp2 = sp1 == std::string::npos ? std::string::npos : line.find(' ', sp1 + 1);
498 size_t sp3 = sp2 == std::string::npos ? std::string::npos : line.find(' ', sp2 + 1);
499 if (sp1 == std::string::npos || sp2 == std::string::npos || sp3 != std::string::npos) {
500 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
501 "Malformed request line (must have exactly two spaces)");
502 errorCode = 400;
503 return false;
504 }
505
506 std::string method = line.substr(0, sp1);
507 std::string rawTarget = line.substr(sp1 + 1, sp2 - sp1 - 1);
508 std::string version = trim(line.substr(sp2 + 1));
509
510 // Validate method and basic request line fields
511 if (!isValidHttpMethodToken(method)) {
512 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Method Not Allowed: " + method);
513 errorCode = 400;
514 return false;
515 }
516 if (method.empty() || rawTarget.empty() || version.empty()) {
517 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Invalid request start line fields");
518 errorCode = 400;
519 return false;
520 }
521
522 // 2) Decode and validate path
523 std::string decoded;
524 try {
525 decoded = decodePercentEncoding(rawTarget);
526 } catch (const std::exception& e) {
527 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", e.what());
528 errorCode = 400;
529 return false;
530 }
531 for (unsigned char c : decoded) {
532 if (std::iscntrl(c)) {
533 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
534 "Path contains control characters");
535 errorCode = 400;
536 return false;
537 }
538 }
539
540 // Forbid absolute URIs (http://...) and fragments (#...)
541 if (decoded.rfind("http://", 0) == 0 || decoded.rfind("https://", 0) == 0) {
542 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
543 "Rejected absolute-URI request-target: " + decoded);
544 errorCode = 400;
545 return false;
546 }
547 size_t hashPos = decoded.find('#');
548 if (hashPos != std::string::npos) {
549 decoded.erase(hashPos); // strip fragment
550 }
551
552 // Split into path and query string
553 std::string pathOnly = decoded, query;
554 size_t qpos = decoded.find('?');
555 if (qpos != std::string::npos) {
556 pathOnly = decoded.substr(0, qpos);
557 query = decoded.substr(qpos + 1);
558 }
559
560 // Reject overly long URIs
561 static constexpr size_t MAX_URI_LEN = 2048;
562 if (pathOnly.size() > MAX_URI_LEN) {
563 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Request-URI Too Long: " + pathOnly);
564 errorCode = 414;
565 return false;
566 }
567
568 // 3) Initialize request object
569 req = HttpRequest();
570 req.setMethod(method);
571
572 // Reject traversal attempts via ".." segments
573 {
574 std::istringstream segstream(pathOnly);
575 std::string seg;
576 while (std::getline(segstream, seg, '/')) {
577 if (seg == "..") {
578 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
579 "Rejected traversal attempt: " + pathOnly);
580 errorCode = 403;
581 return false;
582 }
583 }
584 }
585
586 // Normalize path and set fields
587 std::string norm = normalizePath(pathOnly);
588 if (norm.empty()) {
589 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Path escapes root: " + pathOnly);
590 errorCode = 403;
591 return false;
592 }
593 req.setPath(norm);
594 req.setQuery(query);
595 req.setVersion(version);
596
597 // Only support HTTP/1.0 and HTTP/1.1
598 if (version != "HTTP/1.0" && version != "HTTP/1.1") {
599 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Invalid HTTP version: " + version);
600 errorCode = 505;
601 return false;
602 }
603
604 // 4) Parse headers line by line
605 while (std::getline(stream, line)) {
606 if (!line.empty() && line.back() == '\r')
607 line.pop_back();
608 if (line.empty())
609 continue;
610
611 size_t colon = line.find(':');
612 if (colon == std::string::npos || colon == 0 || colon + 1 >= line.size()) {
613 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Malformed header line: " + line);
614 errorCode = 400;
615 return false;
616 }
617
618 std::string key = line.substr(0, colon);
619 if (key.empty() || key.find_first_of(" \t") != std::string::npos) {
620 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Invalid header name: " + key);
621 errorCode = 400;
622 return false;
623 }
624
625 std::string value = line.substr(colon + 1);
626 value.erase(0, value.find_first_not_of(" \t\r\n"));
627 value.erase(value.find_last_not_of(" \t\r\n") + 1);
628
629 // Delegate validation/insertion to helper
630 if (!insertValidatedHeader(req, key, value, errorCode))
631 return false;
632 }
633
634 // 5) Build full URL using Host header + server config
635 std::string hostHeader = req.getHeader("HOST");
636 try {
637 Url url;
638 const Server& foundServer = searchBestMatchedServers(servers, hostHeader);
639 clientMaxBodySize = foundServer.getClientMaxBodySize();
640
641 std::string urlStr =
642 "http://" + (hostHeader.empty() ? foundServer.getHost() : hostHeader) + req.getPath();
643
644 // Use stricter parsing for HTTP/1.1
645 if (req.getVersion() == "HTTP/1.1") {
646 url = parseUrlHttpVersion1_1(req, urlStr);
647 } else {
648 url = parseUrl(urlStr);
649 }
650
651 req.setUrl(url);
652
653 // Normalize and set host explicitly
654 std::string& headerAfterParseUrl = url.host;
655 std::transform(headerAfterParseUrl.begin(), headerAfterParseUrl.end(),
656 headerAfterParseUrl.begin(), ::tolower);
657 req.setHost(headerAfterParseUrl);
658
659 Logger::logFrom(LogLevel::INFO, "HttpRequestParser",
660 "Parsed URL: " + urlStr + " with host: " + headerAfterParseUrl);
661 } catch (const std::exception& e) {
662 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", e.what());
663 errorCode = 400;
664 return false;
665 }
666
667 return true;

670/**
671 * @brief Checks if a chunked HTTP request body is complete.
672 *
673 * @details According to RFC 7230 §4.1, a chunked transfer ends when:
674 * - A zero-length chunk (`0\r\n`) is received.
675 * - Followed by an optional trailer section.
676 * - And terminated by a blank line (`\r\n\r\n`).
677 *
678 * This function inspects the body buffer to determine if the final
679 * zero-size chunk and terminating CRLF sequence are present.
680 *
681 * @param bodyPart The raw body data received so far.
682 * @return `true` if the chunked body is complete,
683 * `false` if more data is required.
684 *
685 * @ingroup http
686 */
687bool isChunkedBodyComplete(const std::string& bodyPart) {
688 // 1. Look for the "0\r\n" marker (start of the last chunk).
689 std::size_t zeroPos = bodyPart.find("0\r\n");
690 if (zeroPos == std::string::npos)
691 return false; // no terminator chunk yet
692
693 // 2. Ensure the trailers (if any) are properly terminated with CRLFCRLF.
694 std::size_t trailerEnd = bodyPart.find("\r\n\r\n", zeroPos);
695 return trailerEnd != std::string::npos;
696}

698/**
699 * @brief Parses a chunked HTTP request body and assembles it into the request.
700 *
701 * @details Implements RFC 7230 §4.1 "Chunked Transfer Coding".
702 * - Reads each chunk size (hexadecimal) from the body stream.
703 * - Copies the corresponding number of bytes into a contiguous body buffer.
704 * - Stops when a zero-size chunk is encountered.
705 * - Rejects trailers (unsupported in this implementation).
706 * - Enforces the per-client maximum body size limit.
707 *
708 * @param req Reference to the @ref HttpRequest being built.
709 * @param bodyPart The raw body data received so far (may include CRLFs, chunks, trailers).
710 * @param clientMaxBodySize Maximum allowed body size for this client (bytes).
711 * @param errorCode Output parameter: set to an HTTP error status (e.g. 400, 413) on failure.
712 * @param consumedBytes Output parameter: number of bytes successfully consumed from the buffer.
713 *
714 * @note If trailers are present after the final 0–chunk, the request is rejected with `400`.
715 * @note If the accumulated body exceeds @p clientMaxBodySize, the request is rejected with `413`.
716 *
717 * @ingroup http
718 */
719void chunkReqHandler(HttpRequest& req, const std::string& bodyPart, std::size_t clientMaxBodySize,
720 int& errorCode, std::size_t& consumedBytes) {
721 std::istringstream stream(bodyPart);
722 std::string line;
723 std::string fullBody;
724 std::size_t total = 0, local = 0;
725
726 // Process each line (either chunk size, CRLF, or trailers)
727 while (std::getline(stream, line)) {
728 local += line.size() + 1; // count CRLF too
729 if (!line.empty() && line.back() == '\r')
730 line.pop_back();
731
732 // 1) Parse chunk size (hexadecimal)
733 std::size_t chunkSize;
734 try {
735 chunkSize = std::stoul(line, nullptr, 16);
736 } catch (...) {
737 errorCode = 400; // Bad Request: malformed chunk size
738 consumedBytes += local;
739 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Invalid chunk size format");
740 return;
741 }
742
743 // 2) Final 0–chunk indicates end of body
744 if (chunkSize == 0) {
745 // Consume the CRLF after the last chunk
746 std::getline(stream, line);
747 local += line.size() + 1;
748
749 // Reject any trailers (not supported)
750 std::string trailer;
751 while (std::getline(stream, trailer)) {
752 local += trailer.size() + 1;
753 if (trailer.empty())
754 break; // empty line terminates trailers
755 errorCode = 400; // reject any non-empty trailer
756 consumedBytes += local;
757 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
758 "Unsupported trailer header: " + trailer);
759 return;
760 }
761 break;
762 }
763
764 // 3) Enforce max body size limit
765 Logger::logFrom(LogLevel::WARN, "PARSER",
766 " Totalsize: " + std::to_string(total + chunkSize));
767 if (total + chunkSize > clientMaxBodySize) {
768 errorCode = 413; // Payload Too Large
769 consumedBytes += local + chunkSize;
770 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
771 "Exceeded max body size in chunked transfer");
772 return;
773 }
774
775 // 4) Read the chunk data
776 std::string data(chunkSize, '\0');
777 stream.read(&data[0], chunkSize);
778 std::streamsize got = stream.gcount();
779 local += got;
780 fullBody.append(data, 0, got);
781 total += got;
782
783 // Consume CRLF after the chunk data
784 std::getline(stream, line);
785 local += line.size() + 1;
786 }
787
788 // 5) Finalize request
789 req.setBody(fullBody);
790 req.setContentLength(fullBody.size());
791 errorCode = 0;
792 consumedBytes += local;
793}

795/**
796 * @brief Heuristically checks if a string looks like the start line of an HTTP request.
797 *
798 * @details This helper is used to detect pipelined requests:
799 * - It tests whether the string begins with one of the known HTTP
800 * method tokens followed by a space (e.g., "GET ", "POST ").
801 * - Only a small subset of standard methods is recognized
802 * (GET, POST, DELETE, PUT, HEAD, OPTIONS, PATCH).
803 *
804 * This is not a full request-line parser, but a lightweight
805 * heuristic to distinguish request boundaries in a raw buffer.
806 *
807 * @param s The string segment to test (usually leftover buffer data).
808 * @return true if @p s looks like a valid request start line, false otherwise.
809 *
810 * @ingroup http
811 */
812static bool isLikelyStartLine(const std::string& s) {
813 // List of method tokens followed by a space
814 static const std::array<std::string_view, 7> methods = {"GET ", "POST ", "DELETE ", "PUT ",
815 "HEAD ", "OPTIONS ", "PATCH "};
816
817 // Compare the beginning of s with each known method token
818 for (auto m : methods) {
819 if (s.size() >= m.size() && std::string_view(s).substr(0, m.size()) == m) {
820 return true; // Found a match → likely start line
821 }
822 }
823
824 return false; // No match → probably not a start line
825}

827/**
828 * @brief Parses and validates the HTTP request body according to Content-Length or
829 * Transfer-Encoding.
830 *
831 * @details This function handles two body transfer mechanisms:
832 * - **Chunked transfer encoding**: waits until the terminating `0\r\n\r\n` is received,
833 * then delegates parsing to @ref chunkReqHandler. Rejects incomplete or malformed
834 * chunked bodies.
835 * - **Fixed-length bodies**: uses the `Content-Length` header to determine how many bytes
836 * to consume, rejecting requests that exceed @p clientMaxBodySize or contain mismatched
837 * lengths.
838 *
839 * It also supports HTTP/1.1 pipelining:
840 * - If extra bytes follow a complete body, the function checks whether they look like the
841 * start of another request (via @ref isLikelyStartLine).
842 * - If so, only the declared body is consumed and parsing can continue with the next
843 * request.
844 *
845 * @param req The @ref HttpRequest object being filled.
846 * @param bodyPart Raw body data extracted from the client buffer.
847 * @param clientMaxBodySize Maximum allowed body size (from server configuration).
848 * @param errorCode Reference set to HTTP error code on failure (0 on success).
849 * @param consumedBytes Reference updated with the number of bytes consumed from @p bodyPart.
850 *
851 * @return true if the body is complete and valid, false if more data is needed or if an error
852 * occurred.
853 *
854 * @ingroup http
855 */
856bool parseReqBody(HttpRequest& req, const std::string& bodyPart, std::size_t clientMaxBodySize,
857 int& errorCode, std::size_t& consumedBytes) {
858 const std::string& te = req.getHeader("TRANSFER-ENCODING");
859
860 // 1) Handle chunked transfer encoding
861 if (te == "chunked") {
862 if (!isChunkedBodyComplete(bodyPart)) {
863 // Body not yet complete → wait for more data
864 Logger::logFrom(LogLevel::INFO, "HttpRequestParser",
865 "Incomplete chunked body, waiting for more");
866 errorCode = 0;
867 return false;
868 }
869 // Fully received → delegate to chunked handler
870 chunkReqHandler(req, bodyPart, clientMaxBodySize, errorCode, consumedBytes);
871 return (errorCode == 0);
872 }
873
874 // 2) Handle fixed-length bodies
875 std::size_t len = req.getContentLength();
876
877 // Reject if declared length exceeds configured max
878 if (len > 0 && len >= clientMaxBodySize) {
879 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
880 "Exceeded max body size in non-chunked transfer");
881 errorCode = 413; // Payload Too Large
882 consumedBytes += bodyPart.size();
883 return false;
884 }
885
886 // Not enough data yet → wait for more
887 if (bodyPart.size() < len) {
888 errorCode = 0;
889 return false;
890 }
891
892 // 3) Detect pipelined requests
893 if (bodyPart.size() > len) {
894 // Look at the bytes immediately after the declared body
895 std::string_view remainder(bodyPart.c_str() + len, bodyPart.size() - len);
896
897 if (isLikelyStartLine(std::string(remainder))) {
898 // Treat as pipelined request → consume only declared body
899 req.setBody(bodyPart.substr(0, len));
900 consumedBytes += len;
901 errorCode = 0;
902 return true;
903 }
904
905 // Extra data is invalid → reject
906 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
907 "Content-Length mismatch: body too long");
908 errorCode = 400; // Bad Request
909 consumedBytes += bodyPart.size();
910 return false;
911 }
912
913 // 4) Exact match → accept body
914 req.setBody(bodyPart.substr(0, len));
915 consumedBytes += len;
916 errorCode = 0;
917 return true;
918}

920/**
921 * @brief Performs final validation of a parsed HTTP request.
922 *
923 * @details This function applies protocol-level validation rules after the request line
924 * and headers have been parsed. Specifically:
925 * - Ensures the HTTP method is supported (GET, POST, DELETE).
926 * - Validates the normalized request path (must not escape root).
927 * - Checks the `Connection` header (only "keep-alive" or "close" are accepted).
928 * - For POST requests:
929 * * Requires either a `Content-Length` or `Transfer-Encoding` header.
930 * * Requires a valid `Content-Type` header, chosen from a limited set
931 * (supports parameters like `multipart/form-data; boundary=...`).
932 *
933 * @param req The @ref HttpRequest object to validate.
934 * @param errorCode Reference set to HTTP error code if validation fails.
935 *
936 * @return true if the request is valid, false otherwise (with @p errorCode set).
937 *
938 * @ingroup http
939 */
940bool validateReq(HttpRequest& req, int& errorCode) {
941 // Allowed methods
942 static const std::set<std::string> methods = {"GET", "POST", "DELETE"};
943
944 if (!methods.count(req.getMethod())) {
945 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
946 "Method Not Allowed: " + req.getMethod());
947 errorCode = 405; // NOTE: semantically could be 501, but 405 is used for tests
948 return false;
949 }
950
951 // Path validation
952 if (!isValidPath(req.getPath())) {
953 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
954 "Invalid request path: " + req.getPath());
955 errorCode = 403; // Forbidden
956 return false;
957 }
958
959 // Connection header
960 std::string conn = req.getHeader("CONNECTION");
961 if (!conn.empty()) {
962 std::string lower = toLower(conn);
963 if (lower != "keep-alive" && lower != "close") {
964 // Normalize invalid values to "close" but still reject
965 req.setHeader("CONNECTION", "close");
966 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
967 "Invalid Connection header value: " + conn);
968 errorCode = 400; // Bad Request
969 return false;
970 }
971 }
972
973 // POST-specific checks
974 if (req.getMethod() == "POST") {
975 // Must declare either Content-Length or Transfer-Encoding
976 if (req.getContentLength() == 0 && req.getHeader("TRANSFER-ENCODING").empty()) {
977 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
978 "POST without Content-Length or Transfer-Encoding");
979 errorCode = 411; // Length Required
980 return false;
981 }
982
983 // Require Content-Type
984 std::string ct = req.getHeader("CONTENT-TYPE");
985 if (ct.empty()) {
986 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser", "Missing Content-Type for POST");
987 errorCode = 415; // Unsupported Media Type
988 return false;
989 }
990
991 // Allowed Content-Types (prefix matching for params like boundary=...)
992 static const std::set<std::string> validTypes = {"application/x-www-form-urlencoded",
993 "multipart/form-data",
994 "text/plain",
995 "application/json",
996 "application/octet-stream",
997 "test/file"};
998
999 std::string lowerCt = toLower(ct);
1000 bool valid = false;
1001 for (const std::string& type : validTypes) {
1002 if (lowerCt.rfind(type, 0) == 0) { // prefix match
1003 valid = true;
1004 break;
1005 }
1006 }
1007
1008 if (!valid) {
1009 Logger::logFrom(LogLevel::ERROR, "HttpRequestParser",
1010 "Unsupported Content-Type for POST: " + ct);
1011 errorCode = 415; // Unsupported Media Type
1012 return false;
1013 }
1014 }
1015
1016 return true;
1017}

1019/**
1020 * @brief Parses a raw HTTP request buffer into an @ref HttpRequest object.
1021 *
1022 * @details This is the main entry point for HTTP request parsing. It coordinates
1023 * header parsing, body parsing, and final validation. The workflow is:
1024 *
1025 * 1. Locate the end of the header section (`\r\n\r\n`).
1026 * 2. Split the buffer into a header block and a body block.
1027 * 3. Parse the request line and headers via @ref parseReqHeader.
1028 * 4. Validate the parsed request with @ref validateReq.
1029 * 5. If the method does not expect a body (e.g. GET/DELETE), exit early.
1030 * - Detect and support HTTP pipelining (multiple requests in one buffer).
1031 * 6. Otherwise, parse the body with @ref parseReqBody.
1032 * 7. On success, return a fully populated @ref HttpRequest.
1033 *
1034 * Error handling:
1035 * - If parsing fails, the function sets @p errorCode to the appropriate
1036 * HTTP status code (400, 405, 411, 413, 414, 415, 505, etc.).
1037 * - @p consumedBytes is always updated to reflect how much of the buffer
1038 * was processed (even if parsing fails).
1039 *
1040 * @param req The @ref HttpRequest to populate.
1041 * @param buffer Raw input buffer containing HTTP request(s).
1042 * @param serversOnPort List of available @ref Server objects for virtual-host matching.
1043 * @param errorCode Output parameter set to HTTP error code on failure.
1044 * @param consumedBytes Output parameter set to the number of bytes successfully consumed.
1045 *
1046 * @return true if the request was successfully parsed, false otherwise.
1047 *
1048 * @ingroup http
1049 */
1050bool HttpRequestParser::parse(HttpRequest& req, const std::string& buffer,
1051 std::vector<Server> serversOnPort, int& errorCode,
1052 std::size_t& consumedBytes) {
1053 // 1) Find end of header block
1054 std::size_t headerEndPos = buffer.find("\r\n\r\n");
1055 if (headerEndPos == std::string::npos) {
1056 Logger::logFrom(LogLevel::INFO, "HttpRequestParser", "Incomplete header, waiting for more");
1057 errorCode = 0; // Not an error, just incomplete
1058 return false;
1059 }
1060 std::size_t headerLen = headerEndPos + 4;
1061
1062 // 2) Split buffer into header and body
1063 std::string headerPart = buffer.substr(0, headerLen);
1064 std::string bodyPart = buffer.substr(headerLen);
1065 consumedBytes = headerLen;
1066
1067 // 3) Parse request line + headers
1068 std::size_t clientMaxBodySize;
1069 if (!parseReqHeader(req, headerPart, errorCode, serversOnPort, clientMaxBodySize)) {
1070 return false; // parseReqHeader sets errorCode
1071 }
1072
1073 // 4) Validate request semantics
1074 if (!validateReq(req, errorCode)) {
1075 return false; // validateReq sets errorCode
1076 }
1077
1078 // 5) Early exit for bodyless methods (GET, DELETE)
1079 std::string method = req.getMethod();
1080 if (method == "GET") {
1081 if (buffer.size() == headerLen) {
1082 // Clean GET: no extra data
1083 consumedBytes = headerLen;
1084 errorCode = 0;
1085 return true;
1086 }
1087
1088 // Check if extra data looks like a new request (pipelining)
1089 char next = buffer[headerLen];
1090 if (next >= 'A' && next <= 'Z') {
1091 // Treat as pipelined request
1092 consumedBytes = headerLen;
1093 errorCode = 0;
1094 return true;
1095 }
1096
1097 // Otherwise, ignore any stray "body"
1098 consumedBytes = buffer.size();
1099 errorCode = 0;
1100 return true;
1101 }
1102
1103 // 6) Parse body for methods that require it (e.g. POST)
1104 std::size_t bodyConsumed = 0;
1105 bool bodyOk = parseReqBody(req, bodyPart, clientMaxBodySize, errorCode, bodyConsumed);
1106 if (!bodyOk) {
1107 consumedBytes = headerLen + bodyConsumed;
1108 return false; // parseReqBody sets errorCode
1109 }
1110
1111 // 7) Success
1112 consumedBytes = headerLen + bodyConsumed;
1113 errorCode = 0;
1114 return true;
1115}
