31std::string resolvePhysicalPath(const HttpRequest& req, const Location& loc) {
32 // Normalize the request path
33 std::string requestPath = normalizePath(req.getPath());
34 if (requestPath.empty())
35 return "";
36
37 // 1) Try resolving against the real root
38 std::string rootPath = buildFilePath(req, loc);
39 // If this isn't a POST *and* the file actually exists under root, use it
40 if (req.getMethod() != "POST" && !rootPath.empty() && fs::exists(rootPath)) {
41 return rootPath;
42 }
43
44 // 2) Otherwise, if upload_store is enabled and the URI matches the location prefix,
45 // map into the upload‐store
46 std::string locPrefix = normalizePath(loc.getPath());
47 if (loc.isUploadEnabled() && requestPath.rfind(locPrefix, 0) == 0) {
48 // strip leading slashes from the remainder
49 std::string relative = requestPath.substr(locPrefix.size());
50 while (!relative.empty() && relative.front() == '/')
51 relative.erase(0, 1);
52
53 // figure out the absolute upload‐store path
54 std::string uploadRoot = normalizePath(loc.getUploadStore());
55 if (!uploadRoot.empty() && uploadRoot.front() != '/')
56 uploadRoot = joinPath(normalizePath(loc.getRoot()), uploadRoot);
57
58 return joinPath(uploadRoot, relative);
59 }
60
61 // 3) Fallback to the root path (even if it doesn't exist)
62 return rootPath;
63}

69std::string make_temp_name(const std::string& prefix, unsigned& counter) {
70 // 1) Where to put it (e.g. "/tmp" on Linux, or $TMPDIR)
71 fs::path tmpdir = fs::temp_directory_path();
72
73 // 2) High-precision timestamp (nanoseconds since epoch)
74 auto now = std::chrono::high_resolution_clock::now();
75 auto ns = std::chrono::duration_cast<std::chrono::nanoseconds>(now.time_since_epoch()).count();
76
77 // 3) Build "<prefix>_<pid>_<nanoseconds>_<counter>.tmp"
78 std::ostringstream ss;
79 ss << prefix << "_" << ns << "_" << counter++ << ".tmp";
80
81 return (tmpdir / ss.str()).string();
82}

84std::string normalizePath(const std::string& path) {
85 if (path.empty())
86 return "/";
87
88 std::string collapsed;
89 bool prevWasSlash = false;
90 for (char c : path) {
91 if (c == '/') {
92 if (!prevWasSlash) {
93 collapsed += '/';
94 prevWasSlash = true;
95 }
96 } else {
97 collapsed += c;
98 prevWasSlash = false;
99 }
100 }
101
102 std::vector<std::string> segments;
103 std::string segment;
104 std::istringstream stream(collapsed);
105 bool hadTrailingSlash = collapsed.back() == '/';
106
107 while (std::getline(stream, segment, '/')) {
108 if (segment.empty() || segment == ".")
109 continue;
110 if (segment == "..") {
111 if (!segments.empty()) {
112 segments.pop_back();
113 } else {
114 return "";
115 }
116 } else {
117 segments.push_back(segment);
118 }
119 }
120
121 std::string result = "/";
122 for (std::size_t i = 0; i < segments.size(); ++i) {
123 result += segments[i];
124 if (i + 1 < segments.size())
125 result += "/";
126 }
127
128 if (hadTrailingSlash && result != "/")
129 result += "/";
130
131 return result;
132}

142std::string buildFilePath(const HttpRequest& request, const Location& loc) {
143 std::string req_path = normalizePath(request.getPath());
144 std::string loc_path = normalizePath(loc.getPath());
145 std::string loc_root = loc.getRoot();
146
147 if (req_path.rfind(loc_path, 0) != 0) {
148 return "";
149 }
150
151 std::string suffix = req_path.substr(loc_path.size());
152 if (!suffix.empty() && suffix[0] == '/')
153 suffix.erase(0, 1);
154
155 return joinPath(loc_root, suffix);
156}

169bool mkdirRecursive(const std::string& path) {
170 std::vector<std::string> parts = splitPath(path);
171 std::string current = path[0] == '/' ? "/" : "";
172
173 for (size_t i = 0; i < parts.size(); ++i) {
174 current = joinPath(current, parts[i]);
175 if (isFile(current)) {
176 return false;
177 }
178 if (mkdir(current.c_str(), 0777) == -1) {
179 if (errno != EEXIST) {
180 return false;
181 }
182 }
183 }
184 return true;
185}

205std::string sanitizeFilename(const std::string& raw) {
206 namespace fs = std::filesystem;
207 // 1) Drop any leading path components
208 fs::path p(raw);
209 std::string name = p.filename().string();
210
211 // 2) Remove path separators and control chars, keep everything else (including Unicode bytes)
212 name.erase(
213 std::remove_if(name.begin(), name.end(),
214 [](unsigned char c) { return c == '/' || c == '\\' || std::iscntrl(c); }),
215 name.end());
216
217 // 3) If that produced empty or “.”/“..”, fallback
218 if (name.empty() || name == "." || name == "..") {
219 name = makeFallbackName();
220 }
221
222 return name;
223}

225// Returns empty on any error or if the resolved path would leave uploadRoot.
226std::string makeSafeUploadPath(const std::string& uploadRoot, const std::string& rawRelativePath) {
227 namespace fs = std::filesystem;
228 std::error_code ec;
229
230 // 1) Canonicalize uploadRoot
231 fs::path root(uploadRoot);
232 fs::path canonRoot = fs::weakly_canonical(root, ec);
233 if (ec) {
234 // bad uploadRoot
235 return {};
236 }
237
238 // 2) Split the rawRelativePath on “/” and sanitize each segment
239 fs::path candidate = root;
240 std::stringstream ss(rawRelativePath);
241 std::string segment;
242 while (std::getline(ss, segment, '/')) {
243 if (segment.empty())
244 continue;
245 std::string safeSeg = sanitizeFilename(segment);
246 candidate /= safeSeg;
247 }
248
249 // 3) For a non‐existent leaf, weakly_canonical will strip it off,
250 // so canonicalize the parent then re‐append the leaf.
251 fs::path leaf = candidate.filename();
252 fs::path canonParent = fs::weakly_canonical(candidate.parent_path(), ec);
253 if (ec) {
254 // parent doesn’t exist or broken symlink
255 return {};
256 }
257 fs::path canonCandidate = canonParent / leaf;
258
259 // 4) Bound‐check: ensure canonCandidate is inside canonRoot
260 auto rootStr = canonRoot.generic_string();
261 auto candStr = canonCandidate.generic_string();
262 if (candStr.size() < rootStr.size() || candStr.compare(0, rootStr.size(), rootStr) != 0 ||
263 (candStr.size() > rootStr.size() && candStr[rootStr.size()] != '/')) {
264 // outside of uploadRoot
265 return {};
266 }
267
268 return candStr;
269}
