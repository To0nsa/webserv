68/**
69 * @brief Checks whether a directive name is allowed (not duplicated).
70 *
71 * @details Tracks encountered directive names within a block (`server` or `location`).
72 * If the directive is marked as repeatable, it is always accepted.
73 * Otherwise, the name is inserted into the `seen` set:
74 * - If insertion succeeds, the directive is new and valid.
75 * - If insertion fails, the directive is a duplicate and rejected.
76 *
77 * @param name The directive keyword being checked.
78 * @param seen Set of already encountered directive names in the current block.
79 * @param repeatable Set of directive names allowed to appear multiple times.
80 *
81 * @return `true` if the directive is valid (new or repeatable),
82 * `false` if it is a duplicate and not allowed.
83 *
84 * @ingroup config_parser
85 */
86bool checkDuplicateDirective(const std::string& name, std::unordered_set<std::string>& seen,
87 const std::unordered_set<std::string>& repeatable) {
88 return repeatable.contains(name) || seen.insert(name).second;
89}

91/**
92 * @brief Dispatches a configuration directive to its registered handler.
93 *
94 * @details Looks up the directive keyword (from the current token) in the
95 * provided handler table. If a matching handler is found, it is
96 * invoked with the target object (Server or Location), parsed values,
97 * source position (line/column), and contextual snippet. If no handler
98 * exists for the directive name, a @ref SyntaxError is thrown.
99 *
100 * @tparam T The type of the configuration target (e.g., `Server` or `Location`).
101 * @tparam HandlerMap A mapping from directive name (`std::string`) to a handler
102 * function/lambda capable of applying the directive.
103 *
104 * @param target The configuration object being populated (Server/Location).
105 * @param token The token representing the directive keyword.
106 * @param values List of string arguments parsed for the directive.
107 * @param handlers Map of directive names to handler functions.
108 * @param line Line number in the source config where the directive occurs.
109 * @param column Column number in the source config where the directive occurs.
110 * @param ctx A snippet of the source line for contextual error messages.
111 *
112 * @throws SyntaxError If the directive name is unknown or unhandled.
113 *
114 * @ingroup config_parser
115 */
116template <typename T, typename HandlerMap>
117void parseDirective(T& target, const Token& token, std::vector<std::string>& values,
118 const HandlerMap& handlers, int line, int column, const std::string& ctx) {
119 const std::string& directiveName = token.value;
120
121 // Look up the directive in the handler table
122 auto handlerIt = handlers.find(directiveName);
123 if (handlerIt == handlers.end()) {
124 throw SyntaxError(formatError("Unknown directive: '" + token.value + "'", line, column),
125 ctx);
126 }
127
128 // Dispatch to the registered handler function
129 auto& handler = handlerIt->second;
130 handler(target, values, line, column, ctx);
131}

138 * @brief Constructs a ConfigParser instance from raw configuration text.
139 *
140 * @details Initializes the internal tokenizer with the provided configuration

142 * token sequence for parsing. This makes subsequent parsing methods

144 * token stream rather than raw text.

146 * @param source Raw configuration string to be parsed.

150ConfigParser::ConfigParser(std::string source) : _tokenizer(std::move(source)) {
151 _tokens = _tokenizer.tokenize();

154/**
155 * @brief Parses the full configuration input into a Config object.
156 *
157 * @details Expects the top-level configuration to consist of one or more
158 * `server` blocks. Ensures that the input is non-empty, validates
159 * the presence of `server` keywords, and parses each block in turn.
160 * If invalid structure or unexpected tokens are encountered, a
161 * `SyntaxError` is thrown with contextual information.
162 *
163 * @return A fully constructed Config object containing all parsed servers.
164 *
165 * @throws SyntaxError If the configuration is empty, missing `server` blocks,
166 * or contains unexpected tokens after a block.
167 *
168 * @ingroup config_parser
169 */
170Config ConfigParser::parseConfig() {
171 Config config;
172
173 // Reject completely empty configuration
174 if (isAtEnd()) {
175 throw SyntaxError(formatError("Empty configuration", 1, 1), getLineSnippet());
176 }
177
178 // Parse one or more server blocks
179 while (!isAtEnd()) {
180 // Each top-level block must start with the 'server' keyword
181 if (current().type != TokenType::KEYWORD_SERVER) {
182 throw SyntaxError(
183 formatError("Expected 'server' block", current().line, current().column),
184 getLineSnippet());
185 }
186
187 // Parse the full server block and add it to the Config
188 config.addServer(parseServer());
189
190 // After a server block, only EOF or another 'server' is allowed
191 if (!isAtEnd() && current().type != TokenType::KEYWORD_SERVER &&
192 current().type != TokenType::END_OF_FILE) {
193 throw SyntaxError(formatError("Unexpected token after server block", current().line,
194 current().column),
195 getLineSnippet());
196 }
197 }
198
199 return config;
200}

204/**
205 * @brief Parses a single `server` block into a Server object.
206 *
207 * @details Ensures the block starts with the `server` keyword and opening brace,
208 * then iterates through its directives and nested `location` blocks.
209 * Each directive is validated against duplicates (unless repeatable)
210 * and dispatched to the appropriate handler.
211 *
212 * @return A fully populated Server object representing the parsed block.
213 *
214 * @throws SyntaxError If the block is malformed, contains duplicate directives,
215 * or is missing the expected closing brace.
216 *
217 * @ingroup config_parser
218 */
219Server ConfigParser::parseServer() {
220 // Validate required "server {" start sequence
221 expect(TokenType::KEYWORD_SERVER, "server block");
222 expect(TokenType::LBRACE, "start of server block");
223
224 Server server; // New Server instance
225 std::unordered_set<std::string> seen; // Track seen directives (for duplicates)
226
227 // Loop until closing brace or EOF
228 while (!isAtEnd() && current().type != TokenType::RBRACE) {
229 if (current().type == TokenType::KEYWORD_LOCATION) {
230 // Nested location block -> parse recursively
231 server.addLocation(parseLocation());
232 } else {
233 // Check for duplicate directives unless repeatable
234 const std::string name = current().value;
235 if (!checkDuplicateDirective(name, seen, kRepeatableServerDirectives)) {
236 throw SyntaxError(formatError("Duplicate directive: '" + name + "'", current().line,
237 current().column),
238 getLineSnippet());
239 }
240 // Parse directive and apply it to the server
241 parseServerDirective(server);
242 }
243 }
244
245 // Ensure proper block termination
246 expect(TokenType::RBRACE, "end of server block");
247 return server;
248}

250/**
251 * @brief Parses and applies a single directive inside a `server` block.
252 *
253 * @details Reads the current directive token (e.g., `listen`, `host`, etc.),
254 * collects its arguments, enforces a terminating semicolon, and then
255 * dispatches the directive to the appropriate handler function from
256 * the server handler table.
257 *
258 * @param server Reference to the Server object being populated.
259 *
260 * @throws SyntaxError If the directive is malformed, missing arguments, or
261 * missing its terminating semicolon.
262 *
263 * @ingroup config_parser
264 */
265void ConfigParser::parseServerDirective(Server& server) {
266 Token key = current(); // Capture the directive keyword token
267 advance(); // Consume the keyword (move to its arguments)
268
269 // Collect directive arguments (strings, numbers, or identifiers)
270 std::vector<std::string> values = collectArgs(kArgTokenTypes);
271
272 // Ensure directive ends with a semicolon
273 expect(TokenType::SEMICOLON, "semicolon after server directive");
274
275 // Dispatch directive to its handler, applying it to the server
276 parseDirective(server, key, values, directive::serverHandlers(), key.line, key.column,
277 getLineSnippet());
278}

282/**
283 * @brief Parses a single `location` block inside a `server` block.
284 *
285 * @details Ensures the block starts with the `location` keyword, validates and
286 * sets the location path, then iterates over its contained directives.
287 * Each directive is validated for duplicates (unless repeatable) and
288 * dispatched to the appropriate handler. The block must end with a
289 * closing brace `}`.
290 *
291 * @return A fully constructed `Location` object populated with directives.
292 *
293 * @throws SyntaxError If the block is malformed, the path is missing or invalid,
294 * contains duplicate directives, or is missing required
295 * braces.
296 *
297 * @ingroup config_parser
298 */
299Location ConfigParser::parseLocation() {
300 expect(TokenType::KEYWORD_LOCATION, "location block"); // Must start with `location`
301
302 Location location;
303
304 // Validate that the next token is a valid path (string or identifier)
305 TokenType type = current().type;
306 if (type != TokenType::STRING && type != TokenType::IDENTIFIER) {
307 throw SyntaxError(formatError("Expected location path after 'location', but got '" +
308 current().value + "'",
309 current().line, current().column),
310 getLineSnippet());
311 }
312
313 // Extract and validate the path token
314 Token pathTok = current();
315 if (pathTok.value.empty() || pathTok.value[0] != '/') {
316 throw SyntaxError(
317 formatError("Location path must start with '/' (got '" + pathTok.value + "')",
318 pathTok.line, pathTok.column),
319 getLineSnippet());
320 }
321 location.setPath(pathTok.value);
322 advance(); // Consume the path token
323
324 expect(TokenType::LBRACE, "start of location block"); // Require opening `{`
325
326 std::unordered_set<std::string> seen; // Track encountered directives to catch duplicates
327
328 // Parse directives until closing brace `}`
329 while (!isAtEnd() && current().type != TokenType::RBRACE) {
330 const std::string name = current().value;
331 if (!checkDuplicateDirective(name, seen, kRepeatableLocationDirectives)) {
332 throw SyntaxError(formatError("Duplicate directive: '" + name + "'", current().line,
333 current().column),
334 getLineSnippet());
335 }
336 parseLocationDirective(location); // Apply directive to the location
337 }
338
339 expect(TokenType::RBRACE, "end of location block"); // Ensure closing `}`
340 return location;
341}

343/**
344 * @brief Parses a single directive inside a `location` block.
345 *
346 * @details Reads the directive keyword, collects its arguments, enforces that
347 * it is terminated by a semicolon `;`, and dispatches it to the
348 * appropriate handler function via the directive table.
349 *
350 * @param location Reference to the `Location` object where parsed directive
351 * data will be applied.
352 *
353 * @throws SyntaxError If arguments are invalid, missing, or the directive
354 * is not followed by a semicolon.
355 *
356 * @ingroup config_parser
357 */
358void ConfigParser::parseLocationDirective(Location& location) {
359 Token key = current(); // Capture directive keyword token (e.g., "root", "index", etc.)
360 advance(); // Consume the directive keyword
361
362 // Collect arguments that follow the directive
363 std::vector<std::string> values = collectArgs(kArgTokenTypes);
364
365 // Ensure the directive is properly terminated by a semicolon
366 expect(TokenType::SEMICOLON, "semicolon after location directive");
367
368 // Dispatch the directive to the correct handler function from the table
369 parseDirective(location, key, values, directive::locationHandlers(), key.line, key.column,
370 getLineSnippet());
371}

460/**
461 * @brief Ensures that the current token matches a specific expected type.
462 *
463 * @details This method validates that the parser is currently positioned
464 * at a token of type @p expected. If the token matches, it is consumed
465 * (the parser advances by one). If not, an `UnexpectedToken` error is thrown
466 * with details about what was expected versus what was found.
467 *
468 * @param expected The exact token type required at this position.
469 * @param context Human-readable context for error reporting
470 * (unused here but provided for consistency).
471 *
472 * @throws UnexpectedToken If the current token does not match @p expected.
473 *
474 * @ingroup config_parser
475 */
476void ConfigParser::expect(TokenType expected, const std::string& context) {
477 (void) context; // Currently unused, but kept for interface consistency
478
479 // If at end or wrong type → prepare error
480 if (isAtEnd() || _tokens[_pos].type != expected) {
481 const Token* actual;
482
483 if (isAtEnd()) {
484 actual = &_tokens.back(); // fallback to EOF token
485 } else {
486 actual = &_tokens[_pos]; // offending token
487 }
488
489 // Build and throw detailed error
490 throw UnexpectedToken(formatError("Expected " + debugTokenType(expected) + ", but got " +
491 debugTokenType(actual->type),
492 actual->line, actual->column),
493 getLineSnippet());
494 }
495
496 ++_pos; // Consume token if matched
497}

499/**
500 * @brief Ensures the current token matches one of several expected types.
501 *
502 * @details This method is used when a directive or grammar rule allows
503 * multiple token types in the same position (e.g., `STRING` or `IDENTIFIER`).
504 * It checks if the current token matches any type in @p types:
505 *
506 * - If a match is found, the token is consumed via `advance()` and returned.
507 * - If none match, an `UnexpectedToken` exception is thrown with a
508 * descriptive error message listing all expected types and the actual one.
509 *
510 * @param types List of acceptable token types (initializer list).
511 * @param context Human-readable context for error reporting.
512 *
513 * @return The matched token, already consumed from the stream.
514 *
515 * @throws UnexpectedToken If the current token does not match any expected type.
516 *
517 * @ingroup config_parser
518 */
519Token ConfigParser::expectOneOf(std::initializer_list<TokenType> types,
520 const std::string& context) {
521 TokenType actual = current().type; // Current token type
522
523 // Try matching against the expected set
524 for (TokenType expected : types) {
525 if (actual == expected) {
526 return advance(); // Consume & return if found
527 }
528 }
529
530 // Build detailed error message if no match
531 std::ostringstream msg;
532 msg << "Expected ";
533 for (auto it = types.begin(); it != types.end(); ++it) {
534 if (it != types.begin()) {
535 msg << " or "; // Separate multiple expected types
536 }
537 msg << debugTokenType(*it); // Append readable token type
538 }
539 msg << " for " << context << ", but got " << debugTokenType(actual);
540
541 // Throw with contextual line snippet
542 throw UnexpectedToken(formatError(msg.str(), current().line, current().column),
543 getLineSnippet());
544}

546/**
547 * @brief Collects a list of argument values following a directive.
548 *
549 * @details This function consumes tokens that belong to the directive's argument list.
550 * It accepts only tokens of specific types (`STRING`, `NUMBER`, `IDENTIFIER`),
551 * provided in the @p validTypes span. Arguments may be separated by commas.
552 * Example:
553 * ```
554 * root /var/www;
555 * methods GET, POST, DELETE;
556 * ```
557 *
558 * - Stops parsing when a token does not match a valid argument type.
559 * - Supports comma-separated lists of arguments.
560 * - Throws if a comma is not followed by a valid argument.
561 *
562 * @param validTypes A span of allowed token types for directive arguments.
563 *
564 * @return A vector of string values representing the collected arguments.
565 *
566 * @throws SyntaxError If a comma is not followed by a valid argument type.
567 *
568 * @ingroup config_parser
569 */
570std::vector<std::string> ConfigParser::collectArgs(std::span<const TokenType> validTypes) {
571 std::vector<std::string> values; // Accumulate argument values here
572
573 // Continue as long as we are not at end-of-file
574 while (!isAtEnd()) {
575 TokenType t = current().type;
576
577 // If current token type is not among the allowed ones, stop collecting
578 if (std::find(validTypes.begin(), validTypes.end(), t) == validTypes.end())
579 break;
580
581 // Append the argument value (string/number/identifier)
582 values.push_back(current().value);
583 advance(); // Move to the next token
584
585 // Handle optional comma-separated arguments
586 while (match(TokenType::COMMA)) {
587 // A comma must be followed by another valid argument
588 if (isAtEnd() || std::find(validTypes.begin(), validTypes.end(), current().type) ==
589 validTypes.end()) {
590 throw SyntaxError(
591 formatError("Expected value after comma", current().line, current().column),
592 getLineSnippet());
593 }
594
595 // Append the value after the comma
596 values.push_back(current().value);
597 advance(); // Consume argument token
598 }
599 }
600
601 return values; // Return the collected argument list
602}
