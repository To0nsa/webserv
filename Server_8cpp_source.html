77/**
78 * @brief Adds a server name alias for this virtual host.
79 *
80 * @details Server names are used to match the `Host` header of incoming HTTP requests.
81 * Multiple names can be added to support name-based virtual hosting.
82 * This method appends without deduplication.
83 *
84 * @param name The server name to add (e.g., "example.com").
85 * @ingroup server_component
86 */
87void Server::addServerName(std::string_view name) {
88 // Append the given server name to the list of aliases.
89 _server_names.emplace_back(toLower(std::string(name)));
90}

253/**
254 * @brief Checks whether the server matches the given server name.
255 *
256 * @details Compares the provided name against the configured server names
257 * for this virtual host. Used during request routing based on the
258 * `Host` header in the HTTP request.
259 *
260 * @param name The server name to check (case-sensitive).
261 * @return `true` if the name matches one of the configured server names.
262 * @ingroup server_component
263 */
264bool Server::hasServerName(std::string_view name) const noexcept {
265 // Check if the given name matches any of the configured server names.
266 return std::any_of(_server_names.begin(), _server_names.end(),
267 [&name](const std::string& s) { return s == name; });
268}
