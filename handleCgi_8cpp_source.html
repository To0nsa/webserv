13/**
14 * @file handleCgi.cpp
15 * @brief Implements CGI (Common Gateway Interface) request handling.
16 *
17 * @details
18 * This module provides full lifecycle management of CGI processes:
19 * - Build CGI environment variables from the incoming `HttpRequest`.
20 * - Create temporary input/output files to stream request/response bodies.
21 * - Fork and exec the target CGI script with the correct interpreter.
22 * - Collect, parse, and forward CGI output as an `HttpResponse`.
23 * - Handle errors, timeouts, and cleanup of child processes.
24 *
25 * ### Workflow
26 * 1. **initCgiProcess**: validate script, create temp files, fork child, exec interpreter.
27 * 2. **Child process**: stdin → request body, stdout → temp output, `execve()` CGI script.
28 * 3. **Parent process**: monitor PID, wait for completion, enforce timeouts.
29 * 4. **finalizeCgi**: parse CGI headers (Status, Content-Type), build `HttpResponse`.
30 * 5. **cleanup**: delete temp files and reset `CgiProcess` state.
31 *
32 * ### Key Functions
33 * - `prepareEnv` → Build CGI environment (SCRIPT_NAME, PATH_INFO, QUERY_STRING, etc.).
34 * - `prepareCgiTempFiles` → Write request body to temp file, create output file.
35 * - `setupAndRunCgiChild` → Fork/exec child process safely.
36 * - `finalizeCgi` → Read output, parse headers, and generate HTTP response.
37 * - `errorOnCgi`, `cleanupCgi`, `tryTerminateCgi` → Robust process management.
38 *
39 * ### Limitations
40 * - Blocking I/O (temp files, fork/exec).
41 * - Only basic CGI spec (no FastCGI, no async streaming).
42 *
43 * @ingroup request_handler
44 */

74/**
75 * @brief Build the environment variables required for a CGI script execution.
76 *
77 * @details
78 * Constructs a `std::vector<std::string>` of `KEY=VALUE` pairs following the CGI/1.1
79 * specification. The environment includes:
80 * - Standard CGI variables (`SCRIPT_NAME`, `PATH_INFO`, `REQUEST_METHOD`, etc.).
81 * - Server metadata (`SERVER_NAME`, `SERVER_PORT`, `SERVER_PROTOCOL`, etc.).
82 * - Request headers, prefixed with `HTTP_` and normalized (uppercased, `-` → `_`).
83 * - Content metadata (`CONTENT_LENGTH`, `CONTENT_TYPE`).
84 *
85 * Special handling:
86 * - Computes `PATH_INFO` if the request URI extends beyond the script’s URI.
87 * - Sets `REDIRECT_STATUS=200` for PHP compatibility.
88 *
89 * @param req Incoming HTTP request.
90 * @param server Server configuration (name, port, defaults).
91 * @param loc Matched location block (provides root, CGI settings).
92 * @param scriptPath Absolute filesystem path of the CGI script to run.
93 *
94 * @return A vector of strings representing the CGI environment, suitable for `execve()`.
95 *
96 * @ingroup request_handler
97 */
98std::vector<std::string> prepareEnv(const HttpRequest& req, const Server& server,
99 const Location& loc, const std::string& scriptPath) {
100 std::vector<std::string> env;
101 auto set = [&](const std::string& k, const std::string& v) { env.push_back(k + "=" + v); };
102
103 // Normalize paths and extract script filename
104 std::string requestPath = normalizePath(req.getPath());
105 std::string locationPath = normalizePath(loc.getPath());
106 std::string scriptName = std::filesystem::path(scriptPath).filename().string();
107
108 // Build SCRIPT_NAME and PATH_INFO
109 std::string scriptUri = locationPath;
110 if (!scriptUri.empty() && scriptUri.back() != '/')
111 scriptUri += "/";
112 scriptUri += scriptName;
113
114 // Extract extra path after the script (PATH_INFO)
115 std::string pathInfo;
116 if (requestPath.size() > scriptUri.size() &&
117 requestPath.compare(0, scriptUri.size(), scriptUri) == 0) {
118 pathInfo = requestPath.substr(scriptUri.size());
119 if (!pathInfo.empty() && pathInfo[0] != '/')
120 pathInfo = "/" + pathInfo;
121 }
122
123 // Core CGI variables (script + path info)
124 set("SCRIPT_NAME", req.getPath());
125 if (pathInfo.empty()) {
126 set("PATH_INFO", req.getPath());
127 } else {
128 set("PATH_INFO", pathInfo);
129 }
130
131 // Request metadata
132 set("REQUEST_METHOD", req.getMethod());
133 set("QUERY_STRING", req.getQuery());
134 set("CONTENT_LENGTH", std::to_string(req.getContentLength()));
135 if (!req.getHeader("Content-Type").empty())
136 set("CONTENT_TYPE", req.getHeader("Content-Type"));
137
138 // Server and protocol metadata
139 set("SERVER_PROTOCOL", "HTTP/1.1");
140 set("GATEWAY_INTERFACE", "CGI/1.1");
141 set("SERVER_SOFTWARE", "webserv/1.0");
142 set("DOCUMENT_ROOT", loc.getRoot());
143 set("SERVER_NAME", server.getDefaultServerName());
144 set("SERVER_PORT", std::to_string(server.getPort()));
145
146 // Script execution context
147 set("PATH_TRANSLATED", scriptPath);
148 set("REMOTE_ADDR", "127.0.0.1"); // currently hardcoded
149 set("REQUEST_URI", req.getPath());
150 set("SCRIPT_FILENAME", scriptPath);
151 set("REDIRECT_STATUS", "200"); // required by some CGI implementations (e.g. PHP)
152
153 // Forward all HTTP headers as CGI variables (HTTP_FOO_BAR format)
154 for (const auto& [key, value] : req.getHeaders()) {
155 std::string envKey = "HTTP_" + toUpper(key);
156 std::replace(envKey.begin(), envKey.end(), '-', '_');
157 set(envKey, value);
158 }
159
160 return env;
161}

163/**
164 * @brief Convert a vector of strings into a null-terminated `char*` array.
165 *
166 * @details
167 * Prepares data for system calls like `execve()`, which require `char* argv[]`
168 * or `char* envp[]` format. Each string’s `c_str()` is cast away from `const`
169 * (safe here because the lifetime is managed by the original `std::string`
170 * vector). The returned array is null-terminated as required by POSIX.
171 *
172 * @param vs Input vector of strings.
173 * @return A `std::vector<char*>` containing pointers to each string, ending with `nullptr`.
174 *
175 * @note The caller must ensure that the original `std::vector<std::string>` outlives
176 * this array, since the pointers are non-owning.
177 */
178std::vector<char*> toCharPtrArray(const std::vector<std::string>& vs) {
179 std::vector<char*> out;
180 out.reserve(vs.size() + 1);
181 for (const auto& s : vs)
182 out.push_back(const_cast<char*>(s.c_str()));
183 out.push_back(nullptr);
184 return out;
185}

187/**
188 * @brief Read up to a maximum number of bytes from a file stream.
189 *
190 * @details
191 * Allocates a temporary buffer of size `maxBytes`, attempts to read from the
192 * given input file stream, and returns both:
193 * - The bytes read, wrapped into a `std::string`.
194 * - The actual count of bytes read (`gcount()`).
195 *
196 * This is typically used to capture the initial chunk of a CGI output file
197 * in order to parse headers before streaming the rest of the response.
198 *
199 * @param file Open input file stream (must be in binary mode).
200 * @param maxBytes Maximum number of bytes to read from the stream.
201 *
202 * @return A pair: `(stringData, bytesRead)`.
203 *
204 * @note If fewer than `maxBytes` are available, only the available bytes are read.
205 * @warning The file’s read position will advance by the number of bytes consumed.
206 */
207std::pair<std::string, std::streamsize> readInitialOutput(std::ifstream& file, size_t maxBytes) {
208 std::vector<char> buffer(maxBytes);
209 file.read(buffer.data(), maxBytes);
210 std::streamsize bytesRead = file.gcount();
211 return {std::string(buffer.data(), bytesRead), bytesRead};
212}

214/**
215 * @brief Locate the end of the HTTP header section in a CGI response.
216 *
217 * @details
218 * Searches for the standard header delimiter:
219 * - First tries `"\r\n\r\n"` (CRLF-terminated headers).
220 * - Falls back to `"\n\n"` if CRLF is not found.
221 *
222 * Updates `delimiterLength` with the length of the matched delimiter
223 * (either 4 or 2).
224 *
225 * @param data Input string containing CGI response data.
226 * @param delimiterLength Output; set to the delimiter size (4 or 2) if found.
227 *
228 * @return The position of the delimiter if found, otherwise `std::nullopt`.
229 */
230std::optional<size_t> findHeaderDelimiter(const std::string& data, size_t& delimiterLength) {
231 size_t pos = data.find("\r\n\r\n");
232 delimiterLength = 4;
233 if (pos == std::string::npos) {
234 pos = data.find("\n\n");
235 delimiterLength = 2;
236 }
237 if (pos != std::string::npos) {
238 return std::optional<size_t>(pos);
239 }
240 return std::nullopt;
241}

243/**
244 * @brief Parse CGI response headers for status code and content type.
245 *
246 * @details
247 * Iterates over each line in the provided header block:
248 * - Extracts the value of `Content-Type:` if present.
249 * - Extracts the numeric value of `Status:` if present (default = 200).
250 * - If parsing fails, logs a warning and falls back to `500`.
251 *
252 * Other headers are ignored by this function.
253 *
254 * @param header Raw header section of the CGI output (up to the header delimiter).
255 *
256 * @return A pair `(statusCode, contentType)`:
257 * - `statusCode` → HTTP status (int).
258 * - `contentType` → MIME type string (empty if not provided).
259 */
260std::pair<int, std::string> parseHeaders(const std::string& header) {
261 std::istringstream stream(header);
262 std::string line, contentType = "";
263 int statusCode = 200;
264
265 while (std::getline(stream, line)) {
266 if (line.find("Content-Type:") == 0)
267 contentType = trim(line.substr(13));
268 else if (line.find("Status:") == 0) {
269 try {
270 statusCode = std::stoi(trim(line.substr(7)));
271 } catch (...) {
272 Logger::logFrom(LogLevel::WARN, "CGI", "Invalid status code in CGI response");
273 statusCode = 500;
274 }
275 }
276 }
277 return {statusCode, contentType};
278}

280bool validateCgiScript(const std::filesystem::path& path, int& errorCode) {
281 if (!isFile(path)) {
282 Logger::logFrom(LogLevel::ERROR, "CGI", "File is invalid");
283 errorCode = 404;
284 return false;
285 }
286 if (access(path.c_str(), X_OK) != 0) {
287 Logger::logFrom(LogLevel::ERROR, "CGI", "File is not executable");
288 errorCode = 403;
289 return false;
290 }
291 return true;

294/**
295 * @brief Validate that a CGI script exists and is executable.
296 *
297 * @details
298 * Performs two checks on the given filesystem path:
299 * 1. Verifies the path refers to a regular file (`isFile`).
300 * - If not, sets `errorCode = 404` (Not Found).
301 * 2. Verifies the file has execute permissions (`access(..., X_OK)`).
302 * - If not, sets `errorCode = 403` (Forbidden).
303 *
304 * Logs errors if validation fails.
305 *
306 * @param path Filesystem path to the CGI script.
307 * @param errorCode Output; set to `404` or `403` on failure.
308 *
309 * @return true if the script exists and is executable, false otherwise.
310 */
311bool prepareCgiTempFiles(CgiProcess& cgi, const HttpRequest& req, int& bodyFd, int& outputFd) {
312 static unsigned counter = 0;
313 cgi.input_path = make_temp_name("webserv_in", counter);
314 cgi.output_path = make_temp_name("webserv_out", counter);
315
316 std::ofstream out(cgi.input_path, std::ios::binary);
317 if (!out.is_open()) {
318 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to create temp input file");
319 return false;
320 }
321 out.write(req.getBody().data(), req.getBody().size());
322 out.close();
323
324 bodyFd = open(cgi.input_path.c_str(), O_RDONLY);
325 if (bodyFd < 0) {
326 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to reopen temp_in file for CGI input");
327 return false;
328 }
329
330 outputFd = open(cgi.output_path.c_str(), O_RDWR | O_CREAT | O_TRUNC, 0600);
331 if (outputFd < 0) {
332 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to create CGI output file");
333 close(bodyFd);
334 return false;
335 }
336
337 return true;
338}

340/**
341 * @brief Setup file descriptors, environment, and execute a CGI script in the child process.
342 *
343 * @details
344 * This function is called after `fork()` in the CGI child. It:
345 * 1. Redirects `stdin` from the temporary body file (`dup2(body_fd, STDIN_FILENO)`).
346 * 2. Redirects `stdout` to the temporary output file (`dup2(output_fd, STDOUT_FILENO)`).
347 * 3. Closes all unrelated file descriptors to avoid leaks.
348 * 4. Resolves the CGI interpreter from the script extension (if configured).
349 * 5. Builds the `argv` array for `execve()` (`[interpreter?, scriptPath, NULL]`).
350 * 6. Prepares the CGI environment variables with `prepareEnv()`.
351 * 7. Changes working directory to the script’s directory for relative file access.
352 * 8. Calls `execve()` to replace the process image with the CGI script.
353 *
354 * If any step fails, logs the error and terminates the child with `exit(1)`.
355 *
356 * @param cgi CGI process metadata (script path, temp files).
357 * @param body_fd File descriptor for the request body (temp input file).
358 * @param output_fd File descriptor for CGI output (temp output file).
359 * @param req Incoming HTTP request (provides headers, body, method).
360 * @param server Server configuration (name, port, etc.).
361 * @param loc Location block configuration (provides CGI interpreter).
362 * @param poll_fds Active poll descriptors (closed to avoid leakage in the child).
363 *
364 * @warning Must only be called in the child process after `fork()`.
365 * @note On success, this function never returns (replaced by `execve()`).
366 */
367void setupAndRunCgiChild(const CgiProcess& cgi, int body_fd, int output_fd, const HttpRequest& req,
368 const Server& server, const Location& loc,
369 const std::vector<pollfd>& poll_fds) {
370 if (dup2(body_fd, STDIN_FILENO) == -1) {
371 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
372 "dup2 stdin failed: " + std::string(strerror(errno)));
373 exit(1);
374 }
375 close(body_fd);
376 if (dup2(output_fd, STDOUT_FILENO) == -1) {
377 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
378 "dup2 stdout failed: " + std::string(strerror(errno)));
379 exit(1);
380 }
381 close(output_fd);
382 for (std::vector<pollfd>::const_iterator it = poll_fds.begin(); it != poll_fds.end(); ++it) {
383 int fd = it->fd;
384 if (fd != STDIN_FILENO && fd != STDOUT_FILENO && fd != STDERR_FILENO) {
385 close(fd);
386 }
387 }
388
389 // 1. Store script and interpreter in scoped std::string
390 std::string scriptPath = cgi.script_path;
391 std::string ext = std::filesystem::path(scriptPath).extension().string();
392 std::string interp = loc.getCgiInterpreter(ext);
393
394 // 2. Build argv using references to scoped strings
395 std::vector<std::string> argvStorage;
396 if (!interp.empty()) {
397 argvStorage.push_back(interp);
398 }
399 argvStorage.push_back(scriptPath);
400
401 std::vector<char*> argv;
402 for (size_t i = 0; i < argvStorage.size(); ++i) {
403 argv.push_back(const_cast<char*>(argvStorage[i].c_str()));
404 }
405 argv.push_back(nullptr);
406
407 // 3. Environment: same principle
408 std::vector<std::string> envStrs = prepareEnv(req, server, loc, scriptPath);
409 std::vector<char*> envp = toCharPtrArray(envStrs);
410
411 // 4. chdir safely
412 const std::string cgiDir = std::filesystem::path(scriptPath).parent_path().string();
413 if (chdir(cgiDir.c_str()) != 0) {
414 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
415 "chdir failed: " + std::string(strerror(errno)));
416 exit(1);
417 }
418 execve(argv[0], argv.data(), envp.data());
419
420 // 5. If execve fails
421 Logger::logFrom(LogLevel::ERROR, "CGI CHILD", "execve failed: " + std::string(strerror(errno)));
422 exit(1);
423}

454/**
455 * @brief Initialize and start a CGI process for a given request.
456 *
457 * @details
458 * - Resolves the script path from the request URI and validates it with `validateCgiScript`.
459 * - Prepares temporary input/output files (`prepareCgiTempFiles`):
460 * - Writes the request body to a temp input file.
461 * - Opens a temp output file for capturing CGI output.
462 * - Forks the process:
463 * - In the child → calls `setupAndRunCgiChild()` (redirects FDs, sets env, execve).
464 * - In the parent → closes temp file descriptors and stores process metadata.
465 * - On error (invalid script, temp file failure, fork failure), sets `errorCode`
466 * appropriately and returns `false`.
467 *
468 * @param cgi Reference to a `CgiProcess` struct (will be populated with PID, paths,
469 * timestamps).
470 * @param req Incoming HTTP request (provides path, headers, body).
471 * @param server Server configuration (name, port, etc.).
472 * @param loc Location block configuration (provides root, CGI settings).
473 * @param poll_fds Current poll file descriptors (passed to child for cleanup).
474 * @param errorCode Output parameter; set to HTTP-like error codes (`404`, `403`, `500`) on failure.
475 *
476 * @return true if the CGI process was successfully started, false otherwise.
477 *
478 * @note On success, the child process never returns (replaced by `execve()`).
479 * @warning Must be followed by later cleanup (`finalizeCgi`, `errorOnCgi`, or `cleanupCgi`).
480 */
481bool initCgiProcess(CgiProcess& cgi, const HttpRequest& req, const Server& server,
482 const Location& loc, const std::vector<pollfd>& poll_fds, int& errorCode) {
483 cgi.last_activity = getCurrentTime();
484 cgi.script_path = std::filesystem::absolute(loc.resolveAbsolutePath(req.getPath()));
485 if (!validateCgiScript(cgi.script_path, errorCode)) {
486 return false;
487 }
488
489 int body_fd = -1, output_fd = -1;
490 if (!prepareCgiTempFiles(cgi, req, body_fd, output_fd)) {
491 errorCode = 500;
492 return false;
493 }
494 cgi.last_activity = getCurrentTime();
495
496 pid_t pid = fork();
497 if (pid < 0) {
498 close(body_fd);
499 close(output_fd);
500 return false;
501 }
502
503 if (pid == 0) {
504 setupAndRunCgiChild(cgi, body_fd, output_fd, req, server, loc, poll_fds);
505 }
506
507 close(body_fd);
508 close(output_fd);
509
510 cgi.pid = pid;
511 cgi.start_time = getCurrentTime();
512 cgi.last_activity = getCurrentTime();
513 return true;
514}

516/**
517 * @brief Finalize a CGI process by parsing its output and building an HTTP response.
518 *
519 * @details
520 * - Opens the CGI output file produced by the child process.
521 * - Scans up to the first 9KB (`MAX_HEADER_SCAN`) to locate the HTTP header delimiter.
522 * - If not found, logs an error and returns a `500` response.
523 * - Extracts and parses CGI headers:
524 * - `Status:` → HTTP status code (default = 200).
525 * - `Content-Type:` → MIME type (optional).
526 * - Computes the offset where the body starts and its size.
527 * - Builds an `HttpResponse` via `ResponseBuilder::generateSuccessFile`, which streams
528 * the body directly from the output file.
529 * - Marks the response with the CGI temp file path so it can be cleaned up later.
530 *
531 * @param cgi Reference to the `CgiProcess` containing CGI state (output path, timestamps).
532 * @param server Server configuration (used for error responses).
533 * @param req Original HTTP request (used for logging and response metadata).
534 *
535 * @return An `HttpResponse` containing either:
536 * - The CGI result (with headers + body).
537 * - Or an error response (500) if parsing/reading failed.
538 *
539 * @note Only the first 9KB of output is scanned for headers. Larger headers are unsupported.
540 * @warning Caller is responsible for eventually cleaning up the CGI temp file.
541 */
542HttpResponse finalizeCgi(CgiProcess& cgi, const Server& server, const HttpRequest& req) {
543
544 cgi.last_activity = getCurrentTime();
545 std::ifstream in(cgi.output_path, std::ios::binary);
546 if (!in.is_open()) {
547 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to open CGI output file");
548 return ResponseBuilder::generateError(500, server, req);
549 }
550
551 in.seekg(0, std::ios::end);
552 std::streamsize totalSize = in.tellg();
553 in.seekg(0, std::ios::beg);
554
555 // Read the first 9KB only to find headers
556 constexpr size_t MAX_HEADER_SCAN = 9 * 1024;
557 auto [initialData, bytesRead] = readInitialOutput(in, MAX_HEADER_SCAN);
558 size_t delimLen = 0;
559 auto headerEndOpt = findHeaderDelimiter(initialData, delimLen);
560 if (!headerEndOpt) {
561 Logger::logFrom(LogLevel::ERROR, "CGI", "Header delimiter not found in first 9KB");
562 return ResponseBuilder::generateError(500, server, req);
563 }
564
565 size_t headerPos = *headerEndOpt;
566 std::string headerSection = initialData.substr(0, headerPos);
567 auto [code, contentType] = parseHeaders(headerSection);
568
569 std::streamsize headerEnd = static_cast<std::streamsize>(headerPos + delimLen);
570 std::streamsize bodySize = totalSize - headerEnd;
571 in.close();
572 req.printRequest();
573 HttpResponse resp = ResponseBuilder::generateSuccessFile(code, cgi.output_path, contentType,
574 req, bodySize, headerEnd);
575 resp.setCgiTempFile(cgi.output_path);
576 return resp;
577}

579/**
580 * @brief Forcefully terminate a CGI process and clean up its resources.
581 *
582 * @details
583 * - Logs the intent to kill the CGI process.
584 * - Sends `SIGKILL` to the child process using its PID.
585 * - Calls `waitpid()` to reap the process and avoid zombies.
586 * - Logs an error if `waitpid` fails.
587 * - Deletes temporary input/output files via `unlinkWithErrorLog`.
588 * - Resets the `CgiProcess` state (`pid`, timestamps, paths).
589 *
590 * Typically invoked when a CGI process times out or becomes unresponsive.
591 *
592 * @param cgi Reference to the CGI process metadata to be cleaned up.
593 *
594 * @warning This is a hard kill (`SIGKILL`) — no graceful shutdown of the CGI script.
595 */
596void errorOnCgi(CgiProcess& cgi) {
597 Logger::logFrom(LogLevel::ERROR, "CGI",
598 "Killing CGI process with PID: " + std::to_string(cgi.pid));

601 Logger::logFrom(LogLevel::ERROR, "CGI",
602 "waitpid failed after killing CGI process: " +
603 std::string(strerror(errno)));
604 }
605 unlinkWithErrorLog(cgi.input_path, "input temp file");
606 unlinkWithErrorLog(cgi.output_path, "output temp file");
607
608 cgi.pid = -1;
609 cgi.start_time = 0;
610 cgi.last_activity = 0;
611 cgi.input_path.clear();
612 cgi.script_path.clear();
613}

617 *
618 * @details
619 * - Removes the temporary input file associated with the CGI process.
620 * - Resets process metadata (`pid`, timestamps, input path).
621 * - Does not touch the CGI output file (which may still be in use
622 * by the HTTP response).
623 *
624 * Typically called after `finalizeCgi()` has built the response.
625 *
626 * @param cgi Reference to the CGI process metadata to be reset.
627 */

629 unlinkWithErrorLog(cgi.input_path, "input temp file");

636/**
637 * @brief Check whether a CGI process has terminated, and reap it if so.
638 *
639 * @details
640 * - Calls `waitpid(pid, &status, WNOHANG)`:
641 * - Returns `false` if the process is still running (`result == 0`).
642 * - Returns `true` if the process has terminated (or if `waitpid` fails).
643 * - Logs an error if `waitpid` itself fails (`result == -1`).
644 *
645 * This function is non-blocking and can be safely called in the event loop
646 * to periodically check if a CGI process is done.
647 *
648 * @param cgi Reference to the CGI process metadata.
649 *
650 * @return `true` if the CGI process has terminated (or an error occurred),
651 * `false` if it is still running.
652 */
653bool tryTerminateCgi(CgiProcess& cgi) {
654 int status;
655 pid_t result = waitpid(cgi.pid, &status, WNOHANG);
656
657 if (result == 0) {
658 return false;
659 }
660
661 if (result == -1) {
662 Logger::logFrom(LogLevel::ERROR, "CGI", "waitpid failed: " + std::string(strerror(errno)));
663 return true;
664 }
665 return true;
666}
