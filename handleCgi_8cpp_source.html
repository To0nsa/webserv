40std::vector<std::string> prepareEnv(const HttpRequest& req, const Server& server,
41 const Location& loc, const std::string& scriptPath) {
42 std::vector<std::string> env;
43 auto set = [&](const std::string& k, const std::string& v) { env.push_back(k + "=" + v); };
44
45 std::string requestPath = normalizePath(req.getPath());
46 std::string locationPath = normalizePath(loc.getPath());
47 std::string scriptName = std::filesystem::path(scriptPath).filename().string();
48
49 // SCRIPT_NAME = URL path to the script (/directory/youpi.bla)
50 std::string scriptUri = locationPath;
51 if (!scriptUri.empty() && scriptUri.back() != '/')
52 scriptUri += "/";
53 scriptUri += scriptName;
54
55 std::string pathInfo;
56 if (requestPath.size() > scriptUri.size() &&
57 requestPath.compare(0, scriptUri.size(), scriptUri) == 0) {
58 pathInfo = requestPath.substr(scriptUri.size());
59 if (!pathInfo.empty() && pathInfo[0] != '/')
60 pathInfo = "/" + pathInfo;
61 }
62 set("SCRIPT_NAME", req.getPath());
63 if (pathInfo.empty()) {
64 set("PATH_INFO", req.getPath());
65 } else {
66 set("PATH_INFO", pathInfo);
67 }
68 set("REQUEST_METHOD", req.getMethod());
69 set("QUERY_STRING", req.getQuery());
70 set("CONTENT_LENGTH", std::to_string(req.getContentLength()));
71 if (!req.getHeader("Content-Type").empty())
72 set("CONTENT_TYPE", req.getHeader("Content-Type"));
73
74 set("SERVER_PROTOCOL", "HTTP/1.1");
75 set("GATEWAY_INTERFACE", "CGI/1.1");
76 set("SERVER_SOFTWARE", "webserv/1.0");
77 set("DOCUMENT_ROOT", loc.getRoot());
78 set("SERVER_NAME", server.getDefaultServerName());
79 set("SERVER_PORT", std::to_string(server.getPort()));
80 set("PATH_TRANSLATED", scriptPath);
81 set("REMOTE_ADDR", "127.0.0.1");
82 set("REQUEST_URI", req.getPath());
83 set("SCRIPT_FILENAME", scriptPath);
84 set("REDIRECT_STATUS", "200");
85
86 for (const auto& [key, value] : req.getHeaders()) {
87 std::string envKey = "HTTP_" + toUpper(key);
88 std::replace(envKey.begin(), envKey.end(), '-', '_');
89 set(envKey, value);
90 }
91
92 return env;
93}

96std::vector<char*> toCharPtrArray(const std::vector<std::string>& vs) {
97 std::vector<char*> out;
98 out.reserve(vs.size() + 1);
99 for (const auto& s : vs)
100 out.push_back(const_cast<char*>(s.c_str()));
101 out.push_back(nullptr);
102 return out;

112std::optional<size_t> findHeaderDelimiter(const std::string& data, size_t& delimiterLength) {
113 size_t pos = data.find("\r\n\r\n");
114 delimiterLength = 4;
115 if (pos == std::string::npos) {
116 pos = data.find("\n\n");
117 delimiterLength = 2;
118 }
119 if (pos != std::string::npos) {
120 return std::optional<size_t>(pos);
121 }
122 return std::nullopt;

125std::pair<int, std::string> parseHeaders(const std::string& header) {
126 std::istringstream stream(header);
127 std::string line, contentType = "";
128 int statusCode = 200;
129
130 while (std::getline(stream, line)) {
131 if (line.find("Content-Type:") == 0)
132 contentType = trim(line.substr(13));
133 else if (line.find("Status:") == 0) {
134 try {
135 statusCode = std::stoi(trim(line.substr(7)));
136 } catch (...) {
137 Logger::logFrom(LogLevel::WARN, "CGI", "Invalid status code in CGI response");
138 statusCode = 500;
139 }
140 }
141 }
142 return {statusCode, contentType};
143}

146 if (!isFile(path)) {

159bool prepareCgiTempFiles(CgiProcess& cgi, const HttpRequest& req, int& bodyFd, int& outputFd) {
160 static unsigned counter = 0;
161 cgi.input_path = make_temp_name("webserv_in", counter);
162 cgi.output_path = make_temp_name("webserv_out", counter);
163
164 std::ofstream out(cgi.input_path, std::ios::binary);
165 if (!out.is_open()) {
166 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to create temp input file");
167 return false;
168 }
169 out.write(req.getBody().data(), req.getBody().size());
170 out.close();
171
172 bodyFd = open(cgi.input_path.c_str(), O_RDONLY);
173 if (bodyFd < 0) {
174 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to reopen temp_in file for CGI input");
175 return false;
176 }
177
178 outputFd = open(cgi.output_path.c_str(), O_RDWR | O_CREAT | O_TRUNC, 0600);
179 if (outputFd < 0) {
180 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to create CGI output file");
181 close(bodyFd);
182 return false;
183 }
184
185 return true;
186}

188void setupAndRunCgiChild(const CgiProcess& cgi, int body_fd, int output_fd, const HttpRequest& req,
189 const Server& server, const Location& loc,
190 const std::vector<pollfd>& poll_fds) {
191 if (dup2(body_fd, STDIN_FILENO) == -1) {
192 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
193 "dup2 stdin failed: " + std::string(strerror(errno)));
194 exit(1);
195 }
196 close(body_fd);
197 if (dup2(output_fd, STDOUT_FILENO) == -1) {
198 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
199 "dup2 stdout failed: " + std::string(strerror(errno)));
200 exit(1);
201 }
202 close(output_fd);
203 for (std::vector<pollfd>::const_iterator it = poll_fds.begin(); it != poll_fds.end(); ++it) {
204 int fd = it->fd;
205 if (fd != STDIN_FILENO && fd != STDOUT_FILENO && fd != STDERR_FILENO) {
206 close(fd);
207 }
208 }
209
210 // 1. Store script and interpreter in scoped std::string
211 std::string scriptPath = cgi.script_path;
212 std::string ext = std::filesystem::path(scriptPath).extension().string();
213 std::string interp = loc.getCgiInterpreter(ext);
214
215 // 2. Build argv using references to scoped strings
216 std::vector<std::string> argvStorage;
217 if (!interp.empty()) {
218 argvStorage.push_back(interp);
219 }
220 argvStorage.push_back(scriptPath);
221
222 std::vector<char*> argv;
223 for (size_t i = 0; i < argvStorage.size(); ++i) {
224 argv.push_back(const_cast<char*>(argvStorage[i].c_str()));
225 }
226 argv.push_back(nullptr);
227
228 // 3. Environment: same principle
229 std::vector<std::string> envStrs = prepareEnv(req, server, loc, scriptPath);
230 std::vector<char*> envp = toCharPtrArray(envStrs);
231
232 // 4. chdir safely
233 const std::string cgiDir = std::filesystem::path(scriptPath).parent_path().string();
234 if (chdir(cgiDir.c_str()) != 0) {
235 Logger::logFrom(LogLevel::ERROR, "CGI CHILD",
236 "chdir failed: " + std::string(strerror(errno)));
237 exit(1);
238 }
239 execve(argv[0], argv.data(), envp.data());
240
241 // 5. If execve fails
242 Logger::logFrom(LogLevel::ERROR, "CGI CHILD", "execve failed: " + std::string(strerror(errno)));
243 exit(1);
244}

261bool initCgiProcess(CgiProcess& cgi, const HttpRequest& req, const Server& server,
262 const Location& loc, const std::vector<pollfd>& poll_fds, int& errorCode) {
263 cgi.last_activity = getCurrentTime();
264 cgi.script_path = std::filesystem::absolute(loc.resolveAbsolutePath(req.getPath()));
265 if (!validateCgiScript(cgi.script_path, errorCode)) {
266 return false;
267 }
268
269 int body_fd = -1, output_fd = -1;
270 if (!prepareCgiTempFiles(cgi, req, body_fd, output_fd)) {
271 errorCode = 500;
272 return false;
273 }
274 cgi.last_activity = getCurrentTime();
275
276 pid_t pid = fork();
277 if (pid < 0) {
278 close(body_fd);
279 close(output_fd);
280 return false;
281 }
282
283 if (pid == 0) {
284 setupAndRunCgiChild(cgi, body_fd, output_fd, req, server, loc, poll_fds);
285 }
286
287 close(body_fd);
288 close(output_fd);
289
290 cgi.pid = pid;
291 cgi.start_time = getCurrentTime();
292 cgi.last_activity = getCurrentTime();
293 return true;
294}

296HttpResponse finalizeCgi(CgiProcess& cgi, const Server& server, const HttpRequest& req) {
297
298 cgi.last_activity = getCurrentTime();
299 std::ifstream in(cgi.output_path, std::ios::binary);
300 if (!in.is_open()) {
301 Logger::logFrom(LogLevel::ERROR, "CGI", "Failed to open CGI output file");
302 return ResponseBuilder::generateError(500, server, req);
303 }
304
305 in.seekg(0, std::ios::end);
306 std::streamsize totalSize = in.tellg();
307 in.seekg(0, std::ios::beg);
308
309 // Read the first 9KB only to find headers
310 constexpr size_t MAX_HEADER_SCAN = 9 * 1024;
311 auto [initialData, bytesRead] = readInitialOutput(in, MAX_HEADER_SCAN);
312 size_t delimLen = 0;
313 auto headerEndOpt = findHeaderDelimiter(initialData, delimLen);
314 if (!headerEndOpt) {
315 Logger::logFrom(LogLevel::ERROR, "CGI", "Header delimiter not found in first 9KB");
316 return ResponseBuilder::generateError(500, server, req);
317 }
318
319 size_t headerPos = *headerEndOpt;
320 std::string headerSection = initialData.substr(0, headerPos);
321 auto [code, contentType] = parseHeaders(headerSection);
322
323 std::streamsize headerEnd = static_cast<std::streamsize>(headerPos + delimLen);
324 std::streamsize bodySize = totalSize - headerEnd;
325 in.close();
326 req.printRequest();
327 HttpResponse resp = ResponseBuilder::generateSuccessFile(code, cgi.output_path, contentType,
328 req, bodySize, headerEnd);
329 resp.setCgiTempFile(cgi.output_path);
330 return resp;
331}

333void errorOnCgi(CgiProcess& cgi) {
334 Logger::logFrom(LogLevel::ERROR, "CGI",
335 "Killing CGI process with PID: " + std::to_string(cgi.pid));
336 kill(cgi.pid, SIGKILL);
337 if (waitpid(cgi.pid, nullptr, 0) == -1) {
338 Logger::logFrom(LogLevel::ERROR, "CGI",
339 "waitpid failed after killing CGI process: " +
340 std::string(strerror(errno)));
341 }
342 unlinkWithErrorLog(cgi.input_path, "input temp file");
343 unlinkWithErrorLog(cgi.output_path, "output temp file");
344
345 cgi.pid = -1;
346 cgi.start_time = 0;
347 cgi.last_activity = 0;
348 cgi.input_path.clear();
349 cgi.script_path.clear();
350}

360bool tryTerminateCgi(CgiProcess& cgi) {
361 int status;
362 pid_t result = waitpid(cgi.pid, &status, WNOHANG);
363
364 if (result == 0) {
365 return false;
366 }
367
368 if (result == -1) {
