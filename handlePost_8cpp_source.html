36static HttpResponse handleUrlEncodedForm(const HttpRequest& request, const Server& server,
37 const std::string& fullpath, const std::string& filename) {
38 auto form = parseFormUrlEncoded(request.getBody());
39 if (form.empty()) {
40 Logger::logFrom(LogLevel::WARN, "Post Handler",
41 "Empty or malformed URL-encoded form body from client.");
42 return ResponseBuilder::generateError(400, server, request);
43 }
44
45 std::string html = "<html><body><h1>Form Received</h1>";
46 for (auto& formField : form) {
47 html += "<p><b>" + htmlEscape(formField.first) + ":</b> " + htmlEscape(formField.second) +
48 "</p>";
49 }
50 html += "</body></html>";
51
52 std::ofstream out(fullpath);
53 if (!out.is_open()) {
54 Logger::logFrom(LogLevel::ERROR, "Post Handler",
55 "Failed to open file for writing: " + fullpath);
56 return ResponseBuilder::generateError(500, server, request);
57 }
58
59 out << html;
60 out.close();
61 if (out.fail()) {
62 Logger::logFrom(LogLevel::ERROR, "Post Handler",
63 "Failed to write or close file: " + fullpath);
64 return ResponseBuilder::generateError(500, server, request);
65 }
66
67 Logger::logFrom(LogLevel::INFO, "Post Handler",
68 "Successfully wrote URL-encoded form to: " + fullpath);
69 return ResponseBuilder::generateSuccess(
70 201, "<h1>Form Received. File " + filename + " created.</h1>", "text/html", request);
71}

92 201, "<html><body><h1>File " + htmlEscape(filename) + " created.</h1></body></html>",
93 "text/html", request);
94}

98static std::string generateFilename() {
99 // 1) high-res time → nanoseconds since epoch
100 auto now = std::chrono::system_clock::now().time_since_epoch();
101 uint64_t ns = std::chrono::duration_cast<std::chrono::nanoseconds>(now).count();
102 // 2) per-process counter to break ties within the same nanosecond
103 uint64_t seq = uploadCounter.fetch_add(1, std::memory_order_relaxed);
104 std::ostringstream oss;
105 oss << "upload_" << ns << "_" << seq;
106 return oss.str();
107}

111static std::optional<HttpResponse>
112validatePostRequest(HttpRequest const& request, Server const& server, Location const& location) {
113 if (request.getBody().empty()) {
114 Logger::logFrom(LogLevel::WARN, "Post Handler",
115 "Empty body → rejecting POST for URI: " + request.getPath());
116 return ResponseBuilder::generateError(400, server, request);
117 }
118
119 if (request.getBody().size() > server.getClientMaxBodySize()) {
120 Logger::logFrom(LogLevel::WARN, "Post Handler",
121 "Body size " + std::to_string(request.getBody().size()) +
122 " exceeds max client body size " +
123 std::to_string(server.getClientMaxBodySize()) +
124 " → rejecting POST for URI: " + request.getPath());
125 return ResponseBuilder::generateError(413, server, request);
126 }
127
128 if (location.getUploadStore().empty()) {
129 Logger::logFrom(LogLevel::WARN, "Post Handler",
130 "No upload store configured → rejecting POST for URI: " +
131 request.getPath());
132 return ResponseBuilder::generateError(403, server, request);
133 }
134
135 return std::nullopt;
136}

139static std::optional<HttpResponse>
140preparePostTargetPath(HttpRequest const& request, Server const& server, Location const& location,
141 fs::path& outTargetPath, std::string& outTargetDirectory,
142 std::string& outTargetFilename) {
143 namespace fs = std::filesystem;
144
145 // 1) NGINX style: POST "/file.ext/" → 404 if file.ext exists
146 std::string reqPath = normalizePath(request.getPath());
147 std::string locPref = normalizePath(location.getPath());
148 if (!reqPath.empty() && reqPath.back() == '/' && reqPath.rfind(locPref, 0) == 0) {
149 std::string rel = reqPath.substr(locPref.size());
150 while (!rel.empty() && rel.front() == '/')
151 rel.erase(0, 1);
152 std::string rootFull = joinPath(normalizePath(location.getRoot()), rel);
153 if (!rootFull.empty() && rootFull.back() == '/')
154 rootFull.pop_back();
155 if (isFile(rootFull)) {
156 Logger::logFrom(LogLevel::WARN, "Post Handler",
157 "Trailing slash on file → rejecting POST for URI: " +
158 request.getPath());
159 return ResponseBuilder::generateError(404, server, request);
160 }
161 }
162
163 // 2) Compute the client‐side relative path under the upload_store
164 std::string rawRel = reqPath.substr(locPref.size());
165 while (!rawRel.empty() && rawRel.front() == '/')
166 rawRel.erase(0, 1);
167
168 // 3) Percent‐decode any “%20”, etc.
169 std::string decodedRel = decodePercentEncoding(rawRel);
170
171 // 4) Build the candidate path under the uploadStore
172 fs::path uploadRoot(location.getUploadStore());
173 fs::path candidate = uploadRoot / fs::path(decodedRel);
174
175 // 5) If they POST to a directory (or included a trailing slash), append a generated filename
176 bool endsWithSlash = !reqPath.empty() && reqPath.back() == '/';
177 if (endsWithSlash || (fs::exists(candidate) && fs::is_directory(candidate))) {
178 candidate /= generateFilename();
179 }
180
181 // 6) Canonicalize & bound‐check (reuse your makeSafeUploadPath taking a relative path)
182 // First compute the path _relative_ to uploadRoot:
183 fs::path relCand = candidate.lexically_relative(uploadRoot);
184 std::string relCandStr = relCand.generic_string();
185
186 std::string safeFull = makeSafeUploadPath(location.getUploadStore(), relCandStr);
187 if (safeFull.empty()) {
188 Logger::logFrom(LogLevel::WARN, "Post Handler",
189 "Unsafe upload path → rejecting POST for URI: " + request.getPath() +
190 " rel: " + relCandStr);
191 return ResponseBuilder::generateError(403, server, request);
192 }
193
194 // 7) Symlink-forbid: reject if any component under uploadRoot is a symlink
195 fs::path root = uploadRoot;
196 fs::path target(outTargetPath = safeFull);
197 fs::path relRooted = target.lexically_relative(root);
198 fs::path acc = root;
199 for (auto const& comp : relRooted) {
200 acc /= comp;
201 std::error_code ec;
202 if (fs::is_symlink(acc, ec) && !ec) {
203 Logger::logFrom(LogLevel::WARN, "Post Handler",
204 "Symlink in upload path → rejecting POST for URI: " +
205 request.getPath() + " component: " + acc.string());
206 return ResponseBuilder::generateError(403, server, request);
207 }
208 }
209
210 // 8) Populate outputs and ensure directory exists
211 outTargetPath = fs::path(safeFull);
212 outTargetDirectory = outTargetPath.parent_path().generic_string();
213 outTargetFilename = outTargetPath.filename().string();
214
215 if (!mkdirRecursive(outTargetDirectory)) {
216 Logger::logFrom(LogLevel::ERROR, "Post Handler",
217 "Failed to create directory: " + outTargetDirectory +
218 " → rejecting POST for URI: " + request.getPath());
219 return ResponseBuilder::generateError(500, server, request);
220 }
221
222 // 9) Reject if file already exists
223 if (isFile(outTargetPath.string())) {
224 Logger::logFrom(LogLevel::WARN, "Post Handler",
225 "File already exists → rejecting POST for URI: " + request.getPath() +
226 " path: " + outTargetPath.string());
227 return ResponseBuilder::generateError(400, server, request);
228 }
229
230 // 10) All clear
231 return std::nullopt;
232}

234static HttpResponse dispatchPostByContentType(HttpRequest const& request, Server const& server,
235 std::filesystem::path const& targetPath,
236 std::string const& targetDirectory,
237 std::string const& targetFilename) {
238 std::string contentTypeHeader = request.getHeader("Content-Type");
239
240 if (contentTypeHeader.find("multipart/form-data") != std::string::npos) {
241 return handleMultipartForm(request, server, targetDirectory);
242 }
243
244 if (contentTypeHeader.find("application/x-www-form-urlencoded") != std::string::npos) {
245 return handleUrlEncodedForm(request, server, targetPath.string(), targetFilename);
246 }
247
248 return handleRawBody(request, server, targetPath.string(), targetFilename);
249}

251HttpResponse handlePost(HttpRequest const& request, Server const& server,
252 Location const& location) {
253 // 1) Preconditions
254 std::optional<HttpResponse> maybeErrorResponse = validatePostRequest(request, server, location);
255 if (maybeErrorResponse.has_value()) {
256 return *maybeErrorResponse;
257 }
258
259 // 2) Filesystem path prep
260 std::filesystem::path targetPath;
261 std::string targetDirectory;
262 std::string targetFilename;
263
264 std::optional<HttpResponse> maybePreparationError = preparePostTargetPath(
265 request, server, location, targetPath, targetDirectory, targetFilename);
266 if (maybePreparationError.has_value()) {
267 return *maybePreparationError;
268 }
269
270 // 3) Content-type dispatch
271 return dispatchPostByContentType(request, server, targetPath, targetDirectory, targetFilename);
272}
