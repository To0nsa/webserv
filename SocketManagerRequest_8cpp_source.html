38bool SocketManager::receiveFromClient(int client_fd, size_t index) {
39 char buffer[RECV_BUFFER];
40 _client_info[client_fd].lastRequestTime = getCurrentTime();
41 int bytes = recv(client_fd, buffer, sizeof(buffer) - 1, MSG_DONTWAIT);
42 if (bytes == 0) {
43 Logger::logFrom(LogLevel::INFO, "SocketManager",
44 "Client fd " + std::to_string(client_fd) + " disconnected.");
45 cleanupClientConnectionClose(client_fd, index);
46 return false;
47 }
48 if (bytes < 0) {
49 Logger::logFrom(LogLevel::ERROR, "SocketManager",
50 std::string("recv() failed: ") + std::strerror(errno));
51 cleanupClientConnectionClose(client_fd, index);
52 return false;
53 }
54 buffer[bytes] = '\0';
55
56 std::string single_msg(buffer, bytes);
57 _client_info[client_fd].requestBuffer += single_msg;
58 size_t headerEndPos = _client_info[client_fd].requestBuffer.find("\r\n\r\n");
59
60 if (headerEndPos == std::string::npos) {
61 if (_client_info[client_fd].headerBytesReceived == 0) {
62 _client_info[client_fd].connectionStartTime = getCurrentTime();
63 }
64 _client_info[client_fd].headerBytesReceived += bytes;
65 } else {
66 if (!_client_info[client_fd].headerComplete) {
67 size_t fullHeaderSize = headerEndPos + 4;
68 size_t oldBufferSize = _client_info[client_fd].requestBuffer.size() - bytes;
69 size_t headerBytesThisTime =
70 std::max((ssize_t) 0, (ssize_t) (fullHeaderSize - oldBufferSize));
71 _client_info[client_fd].headerBytesReceived += headerBytesThisTime;
72 _client_info[client_fd].bodyBytesReceived += (bytes - headerBytesThisTime);
73 _client_info[client_fd].headerComplete = true;
74 } else {
75 // Header already counted, this must be body
76 _client_info[client_fd].bodyBytesReceived += bytes;
77 }
78 }
79
80 return true;
81}

83static const Location* findMatchingLocation(const std::string& path, const Server& server) {
84 const Location* best = nullptr;
85 size_t max = 0;
86 for (const Location& loc : server.getLocations()) {
87 if (path.rfind(normalizePath(loc.getPath()), 0) == 0 &&
88 normalizePath(loc.getPath()).size() > max) {
89 best = &loc;
90 max = normalizePath(loc.getPath()).size();
91 }
92 }
93 return best;
94}

96void SocketManager::processPendingRequests(int client_fd) {
97 ClientInfo& client = _client_info[client_fd];
98
99 // As long as there is at least one pending request AND no CGI is currently running:
100 while (!client.pendingRequests.empty() && !client.isCgiProcessRunning) {
101 HttpRequest nextReq = client.pendingRequests.front();
102 const Server& server = client.serversOnPort[nextReq.getMatchedServerIndex()];
103
104 // nextReq.printRequest();
105
106 int code = nextReq.getParseErrorCode();
107 if (code != 0) {
108 if (handleRequestErrorIfAny(client_fd, code, nextReq, server))
109 return;
110 continue;
111 }
112
113 const Location* location = findMatchingLocation(normalizePath(nextReq.getPath()), server);
114 if (!location) {
115 if (handleRequestErrorIfAny(client_fd, 404, nextReq, server))
116 return;
117 continue;
118 }
119
120 if (shouldSpawnCgi(nextReq, *location)) {
121 client.currentCgiRequest = nextReq;
122 client.isCgiProcessRunning = true;
123 bool ok = handleCgiRequest(client_fd, nextReq, server, *location);
124 client.pendingRequests.pop();
125 if (!ok)
126 continue;
127 return;
128 }
129
130 HttpResponse resp = handleRequest(nextReq, server);
131 client.responses.push(resp);
132 client.pendingRequests.pop();
133 if (resp.isConnectionClose())
134 return;
135 }
136}

138bool SocketManager::parseAndQueueRequests(int client_fd) {
139 ClientInfo& client = _client_info[client_fd];
140
141 while (true) {
142 if (checkRequestLimits(client_fd)) {
143 client.requestBuffer.clear();
144 resetRequestState(client_fd);
145 return true;
146 }
147
148 HttpRequest request;
149 int errorCode = 0;
150 std::size_t consumedBytes = 0;
151
152 bool ok = HttpRequestParser::parse(request, client.requestBuffer, client.serversOnPort,
153 errorCode, consumedBytes);
154
155 if (!ok) {
156 if (errorCode == 0)
157 return false; // incomplete
158
159 request.setParseErrorCode(errorCode);
160 request.printRequest();
161
162 if (errorCode == 415 || errorCode == 411 || errorCode == 400 || errorCode == 413)
163 client.requestBuffer.clear();
164 else
165 client.requestBuffer.erase(0, consumedBytes);
166
167 resetRequestState(client_fd);
168 client.pendingRequests.push(request);
169
170 if (client.requestBuffer.find("\r\n\r\n") == std::string::npos)
171 break;
172
173 continue;
174 }
175
176 client.requestBuffer.erase(0, consumedBytes);
177 resetRequestState(client_fd);
178 client.pendingRequests.push(request);
179
180 if (client.requestBuffer.find("\r\n\r\n") == std::string::npos)
181 break;
182 }
183
184 return true;
185}

187void SocketManager::handleCgiPollEvents() {
188 for (auto& [client_fd, client] : _client_info) {
189 if (!client.cgiProcess)
190 continue;
191
192 try {
193 CgiProcess& cgi = *client.cgiProcess;
194 const Server& server =
195 client.serversOnPort[client.currentCgiRequest.getMatchedServerIndex()];
196
197 if (getCurrentTime() - cgi.last_activity > CGI_TIMEOUT_SECONDS) {
198 Logger::logFrom(LogLevel::WARN, "CGI",
199 "Timeout. Killing CGI process for fd: " +
200 std::to_string(client_fd));
201 client.responses.push(ResponseBuilder::generateError(504, server, {}));
202 CGI::errorOnCgi(cgi);
203 client.cgiProcess.reset();
204 client.isCgiProcessRunning = false;
205 client.currentCgiRequest = HttpRequest();
206 for (auto& pfd : _poll_fds) {
207 if (pfd.fd == client_fd) {
208 pfd.events |= POLLOUT;
209 break;
210 }
211 }
212 continue;
213 }
214
215 if (CGI::tryTerminateCgi(cgi)) {
216 HttpResponse resp = CGI::finalizeCgi(cgi, server, client.currentCgiRequest);
217 client.responses.push(resp);
218 CGI::cleanupCgi(cgi);
219 client.cgiProcess.reset();
220 client.isCgiProcessRunning = false;
221 client.currentCgiRequest = HttpRequest();
222
223 for (auto& pfd : _poll_fds) {
224 if (pfd.fd == client_fd) {
225 pfd.events |= POLLOUT;
226 break;
227 }
228 }
229
230 size_t idx = 0;
231 for (; idx < _poll_fds.size(); ++idx) {
232 if (_poll_fds[idx].fd == client_fd)
233 break;
234 }
235 if (idx < _poll_fds.size())
236 processPendingRequests(client_fd);
237 }
238 } catch (const std::exception& e) {
239 Logger::logFrom(LogLevel::ERROR, "SocketManager",
240 "Exception during CGI handling for fd " + std::to_string(client_fd) +
241 ": " + e.what());
242 respondError(client_fd, 500);
243 cleanupCgiForClient(client_fd);
244 for (auto& pfd : _poll_fds) {
245 if (pfd.fd == client_fd) {
246 pfd.events |= POLLOUT;
247 break;
248 }
249 }
250 }
251 }
252}

259bool SocketManager::handleCgiRequest(int client_fd, const HttpRequest& request,
260 const Server& server, const Location& location) {
261 ClientInfo& client = _client_info[client_fd];
262 client.cgiProcess.emplace();
263
264 int errorCode = 500;
265 if (!CGI::initCgiProcess(*client.cgiProcess, request, server, location, _poll_fds, errorCode)) {
266 Logger::logFrom(LogLevel::ERROR, "SocketManager",
267 "[CGI] Failed to initialize CGI process for client_fd " +
268 std::to_string(client_fd) + " with script: " + location.getPath());
269 HttpResponse err = ResponseBuilder::generateError(errorCode, server, request);
270 _client_info[client_fd].responses.push(err);
271 client.cgiProcess.reset();
272 client.isCgiProcessRunning = false;
273 client.currentCgiRequest = HttpRequest(); // clears request
274 return false; // error response queued
275 }
276
277 return true; // handled as CGI
278}
