61/**
62 * @brief Receives raw data from a client socket.
63 *
64 * @details
65 * - Performs a non-blocking `recv()` on the given client FD.
66 * - Handles connection termination cases:
67 * - `bytes == 0`: client closed the connection → cleanup and return `false`.
68 * - `bytes < 0`: read error → cleanup and return `false`.
69 * - Appends received data to the client’s `requestBuffer`.
70 * - Tracks request progress:
71 * - Updates header vs. body byte counters.
72 * - Marks `headerComplete` when the `\r\n\r\n` delimiter is first encountered.
73 *
74 * @param client_fd The client socket file descriptor.
75 * @param index Index of the client FD in the poll list (needed for cleanup).
76 *
77 * @return `true` if data was successfully received and buffered,
78 * `false` if the connection should be closed.
79 *
80 * @throws None. Errors are logged, and cleanup is performed internally.
81 */
82bool SocketManager::receiveFromClient(int client_fd, size_t index) {
83 char buffer[RECV_BUFFER];
84
85 // Update last activity timestamp (used for idle timeout detection)
86 _client_info[client_fd].lastRequestTime = getCurrentTime();
87
88 // Non-blocking read from the client
89 int bytes = recv(client_fd, buffer, sizeof(buffer) - 1, MSG_DONTWAIT);
90
91 if (bytes == 0) {
92 // Client performed an orderly shutdown
93 Logger::logFrom(LogLevel::INFO, "SocketManager",
94 "Client fd " + std::to_string(client_fd) + " disconnected.");
95 cleanupClientConnectionClose(client_fd, index);
96 return false;
97 }
98 if (bytes < 0) {
99 // Fatal read error
100 Logger::logFrom(LogLevel::ERROR, "SocketManager",
101 std::string("recv() failed: ") + std::strerror(errno));
102 cleanupClientConnectionClose(client_fd, index);
103 return false;
104 }
105
106 // Null-terminate buffer for safe string construction
107 buffer[bytes] = '\0';
108
109 // Append the new chunk to the client's request buffer
110 std::string single_msg(buffer, bytes);
111 _client_info[client_fd].requestBuffer += single_msg;
112
113 // Detect end of HTTP headers ("\r\n\r\n")
114 size_t headerEndPos = _client_info[client_fd].requestBuffer.find("\r\n\r\n");
115
116 if (headerEndPos == std::string::npos) {
117 // Still receiving headers
118 if (_client_info[client_fd].headerBytesReceived == 0) {
119 // First time seeing header data → mark connection start
120 _client_info[client_fd].connectionStartTime = getCurrentTime();
121 }
122 _client_info[client_fd].headerBytesReceived += bytes;
123 } else {
124 if (!_client_info[client_fd].headerComplete) {
125 // Header just completed in this recv()
126 size_t fullHeaderSize = headerEndPos + 4;
127 size_t oldBufferSize = _client_info[client_fd].requestBuffer.size() - bytes;
128 size_t headerBytesThisTime =
129 std::max((ssize_t) 0, (ssize_t) (fullHeaderSize - oldBufferSize));
130
131 _client_info[client_fd].headerBytesReceived += headerBytesThisTime;
132 _client_info[client_fd].bodyBytesReceived += (bytes - headerBytesThisTime);
133 _client_info[client_fd].headerComplete = true;
134 } else {
135 // Header already completed → this is pure body data
136 _client_info[client_fd].bodyBytesReceived += bytes;
137 }
138 }
139
140 return true;
141}

143/**
144 * @brief Finds the best-matching Location for a given request path.
145 *
146 * @details
147 * Iterates through all @ref Location objects configured in the given @ref Server
148 * and selects the one whose `path` prefix matches the request path with the
149 * **longest match length** (longest-prefix match).
150 *
151 * Matching logic:
152 * - A location matches if `normalizePath(loc.getPath())` is a prefix of `path`.
153 * - If multiple locations match, the one with the longest normalized path is chosen.
154 * - If no location matches, returns `nullptr`.
155 *
156 * @param path The normalized request target path (e.g. `/images/foo.jpg`).
157 * @param server The server whose configured locations should be searched.
158 *
159 * @return Pointer to the best matching @ref Location, or `nullptr` if none match.
160 */
161static const Location* findMatchingLocation(const std::string& path, const Server& server) {
162 const Location* best = nullptr;
163 size_t max = 0;
164
165 // Iterate over all configured locations for this server
166 for (const Location& loc : server.getLocations()) {
167 // Check if location path is a prefix of the request path
168 if (path.rfind(normalizePath(loc.getPath()), 0) == 0 &&
169 normalizePath(loc.getPath()).size() > max) {
170 // Keep track of the longest prefix match so far
171 best = &loc;
172 max = normalizePath(loc.getPath()).size();
173 }
174 }
175
176 return best; // nullptr if no location matched
177}

179/**
180 * @brief Processes all pending HTTP requests for a client.
181 *
182 * @details
183 * Runs through the client's queued @ref HttpRequest objects and:
184 * 1. Skips processing if a CGI process is currently running
185 * (only one CGI may run per client at a time).
186 * 2. Checks for parse errors in the request:
187 * - If present, generates an error response via @ref handleRequestErrorIfAny.
188 * 3. Resolves the request target to the best-matching @ref Location
189 * using @ref findMatchingLocation.
190 * - If no location matches, responds with 404.
191 * 4. If the request must be served by a CGI:
192 * - Marks client as running CGI, initializes the process with
193 * @ref handleCgiRequest, and defers further processing.
194 * 5. Otherwise, handles the request normally via @ref handleRequest and
195 * queues the @ref HttpResponse.
196 *
197 * The loop continues until:
198 * - No more pending requests, OR
199 * - A CGI request is started, OR
200 * - A response requires connection close.
201 *
202 * @param client_fd File descriptor of the client whose requests are processed.
203 */
204void SocketManager::processPendingRequests(int client_fd) {
205 ClientInfo& client = _client_info[client_fd];
206
207 // Process while there are queued requests and no active CGI process
208 while (!client.pendingRequests.empty() && !client.isCgiProcessRunning) {
209 HttpRequest nextReq = client.pendingRequests.front();
210 const Server& server = client.serversOnPort[nextReq.getMatchedServerIndex()];
211
212 // Handle requests with parsing errors
213 int code = nextReq.getParseErrorCode();
214 if (code != 0) {
215 if (handleRequestErrorIfAny(client_fd, code, nextReq, server))
216 return; // stop if error requires closing connection
217 continue; // otherwise, process next pending request
218 }
219
220 // Match request path against server locations
221 const Location* location = findMatchingLocation(normalizePath(nextReq.getPath()), server);
222 if (!location) {
223 if (handleRequestErrorIfAny(client_fd, 404, nextReq, server))
224 return;
225 continue;
226 }
227
228 // If request matches a CGI location → launch CGI process
229 if (shouldSpawnCgi(nextReq, *location)) {
230 client.currentCgiRequest = nextReq;
231 client.isCgiProcessRunning = true;
232 bool ok = handleCgiRequest(client_fd, nextReq, server, *location);
233 client.pendingRequests.pop();
234 if (!ok)
235 continue; // CGI init failed → continue with next request
236 return; // CGI started → stop until it finishes
237 }
238
239 // Otherwise, handle request normally and enqueue response
240 HttpResponse resp = handleRequest(nextReq, server);
241 client.responses.push(resp);
242 client.pendingRequests.pop();
243
244 // If response signals "Connection: close" → stop processing
245 if (resp.isConnectionClose())
246 return;
247 }
248}

250/**
251 * @brief Parses buffered client data into HTTP requests and queues them.
252 *
253 * @details
254 * This function repeatedly attempts to parse complete HTTP requests
255 * from a client’s `requestBuffer`:
256 *
257 * 1. **Check request limits:**
258 * - If headers exceed configured limits, clears buffer and pushes
259 * a `431 Request Header Fields Too Large` error.
260 *
261 * 2. **Parse next request:**
262 * - Uses @ref HttpRequestParser::parse to decode from the buffer.
263 * - On success: consumes parsed bytes, resets state, and queues the request.
264 * - On incomplete parse (`errorCode == 0`): returns `false` to wait for more data.
265 * - On parse error (`400`, `411`, `413`, `415`, etc.):
266 * - Marks the request with the error code.
267 * - Either clears the whole buffer (fatal errors) or erases consumed bytes.
268 * - Resets state and queues the error request.
269 *
270 * 3. **Loop continuation:**
271 * - If more data remains in `requestBuffer` (and headers are present),
272 * parsing continues until buffer is exhausted or incomplete.
273 *
274 * @param client_fd File descriptor of the client whose buffer is parsed.
275 *
276 * @return `true` if parsing succeeded or error requests were queued,
277 * `false` if parsing is incomplete and more data is required.
278 */
279bool SocketManager::parseAndQueueRequests(int client_fd) {
280 ClientInfo& client = _client_info[client_fd];
281
282 while (true) {
283 // 1) enforce request size/limit rules
284 if (checkRequestLimits(client_fd)) {
285 client.requestBuffer.clear();
286 resetRequestState(client_fd);
287 return true;
288 }
289
290 HttpRequest request;
291 int errorCode = 0;
292 std::size_t consumedBytes = 0;
293
294 // 2) attempt to parse one HTTP request from the buffer
295 bool ok = HttpRequestParser::parse(request, client.requestBuffer, client.serversOnPort,
296 errorCode, consumedBytes);
297
298 if (!ok) {
299 if (errorCode == 0)
300 return false; // Incomplete request → wait for more data
301
302 // Fatal parse error (e.g. invalid headers, bad content length)
303 request.setParseErrorCode(errorCode);
304 request.printRequest();
305
306 // Clear buffer for hard errors, else drop only consumed part
307 if (errorCode == 415 || errorCode == 411 || errorCode == 400 || errorCode == 413)
308 client.requestBuffer.clear();
309 else
310 client.requestBuffer.erase(0, consumedBytes);
311
312 // Reset parsing state and queue error request
313 resetRequestState(client_fd);
314 client.pendingRequests.push(request);
315
316 // Stop if no headers left in buffer
317 if (client.requestBuffer.find("\r\n\r\n") == std::string::npos)
318 break;
319
320 continue;
321 }
322
323 // Success: remove consumed bytes, reset state, queue request
324 client.requestBuffer.erase(0, consumedBytes);
325 resetRequestState(client_fd);
326 client.pendingRequests.push(request);
327
328 // Stop if buffer doesn’t contain another full header
329 if (client.requestBuffer.find("\r\n\r\n") == std::string::npos)
330 break;
331 }
332
333 return true;
334}

336/**
337 * @brief Monitors and manages active CGI processes for all clients.
338 *
339 * @details
340 * Iterates over all connected clients and checks if a CGI process
341 * is associated with them. For each active CGI:
342 *
343 * 1. **Timeout check:**
344 * - If the process has been idle longer than @ref CGI_TIMEOUT_SECONDS,
345 * it is killed and a `504 Gateway Timeout` response is queued.
346 *
347 * 2. **Completion check:**
348 * - If @ref CGI::tryTerminateCgi indicates the process is done,
349 * finalize output with @ref CGI::finalizeCgi, push response,
350 * cleanup process state, and re-enable `POLLOUT` to send data.
351 * - After cleanup, if more requests remain queued for the client,
352 * continue processing them with @ref processPendingRequests.
353 *
354 * 3. **Exception safety:**
355 * - If any error occurs during CGI handling, logs the error,
356 * queues a `500 Internal Server Error`, and cleans up CGI state.
357 *
358 * Integration:
359 * - This method is called once per poll loop iteration before
360 * handling socket-level events.
361 * - It ensures CGI subprocesses do not block the server’s main loop.
362 *
363 * @see processPendingRequests
364 */
365void SocketManager::handleCgiPollEvents() {
366 for (auto& [client_fd, client] : _client_info) {
367 if (!client.cgiProcess)
368 continue; // Skip clients without an active CGI
369
370 try {
371 CgiProcess& cgi = *client.cgiProcess;
372 const Server& server =
373 client.serversOnPort[client.currentCgiRequest.getMatchedServerIndex()];
374
375 // 1. Timeout check
376 if (getCurrentTime() - cgi.last_activity > CGI_TIMEOUT_SECONDS) {
377 Logger::logFrom(LogLevel::WARN, "CGI",
378 "Timeout. Killing CGI process for fd: " +
379 std::to_string(client_fd));
380 client.responses.push(ResponseBuilder::generateError(504, server, {}));
381 CGI::errorOnCgi(cgi); // Kill process with error
382 client.cgiProcess.reset(); // Drop process handle
383 client.isCgiProcessRunning = false;
384 client.currentCgiRequest = HttpRequest();
385
386 // Ensure POLLOUT so timeout response gets sent
387 for (auto& pfd : _poll_fds) {
388 if (pfd.fd == client_fd) {
389 pfd.events |= POLLOUT;
390 break;
391 }
392 }
393 continue;
394 }
395
396 // 2. Completion check
397 if (CGI::tryTerminateCgi(cgi)) {
398 HttpResponse resp = CGI::finalizeCgi(cgi, server, client.currentCgiRequest);
399 client.responses.push(resp);
400 CGI::cleanupCgi(cgi); // Free temp files/resources
401 client.cgiProcess.reset();
402 client.isCgiProcessRunning = false;
403 client.currentCgiRequest = HttpRequest();
404
405 // Mark socket ready for sending CGI response
406 for (auto& pfd : _poll_fds) {
407 if (pfd.fd == client_fd) {
408 pfd.events |= POLLOUT;
409 break;
410 }
411 }
412
413 // Resume processing queued requests (pipelined)
414 size_t idx = 0;
415 for (; idx < _poll_fds.size(); ++idx) {
416 if (_poll_fds[idx].fd == client_fd)
417 break;
418 }
419 if (idx < _poll_fds.size())
420 processPendingRequests(client_fd);
421 }
422 } catch (const std::exception& e) {
423 // 3. Exception handling: fail safe
424 Logger::logFrom(LogLevel::ERROR, "SocketManager",
425 "Exception during CGI handling for fd " + std::to_string(client_fd) +
426 ": " + e.what());
427 respondError(client_fd, 500); // Queue internal error
428 cleanupCgiForClient(client_fd); // Force cleanup
429
430 // Mark socket for POLLOUT to flush error response
431 for (auto& pfd : _poll_fds) {
432 if (pfd.fd == client_fd) {
433 pfd.events |= POLLOUT;
434 break;
435 }
436 }
437 }
438 }
439}

446/**
447 * @brief Initializes and starts a CGI process for a client request.
448 *
449 * @details
450 * - Allocates a new @ref CgiProcess in the client state.
451 * - Attempts to initialize the CGI environment via @ref CGI::initCgiProcess.
452 * - On success: leaves the process active and returns `true`.
453 * - On failure:
454 * - Logs an error with details.
455 * - Queues an error @ref HttpResponse (e.g. `500`, or errorCode returned).
456 * - Cleans up CGI state and clears the current request.
457 * - Returns `false`.
458 *
459 * @param client_fd The client file descriptor.
460 * @param request The HTTP request to be executed by CGI.
461 * @param server The server context for this request.
462 * @param location The location context where CGI execution is configured.
463 *
464 * @return `true` if CGI initialization succeeded,
465 * `false` if an error response was queued instead.
466 *
467 * @see handleCgiPollEvents, processPendingRequests
468 */
469bool SocketManager::handleCgiRequest(int client_fd, const HttpRequest& request,
470 const Server& server, const Location& location) {
471 ClientInfo& client = _client_info[client_fd];
472 client.cgiProcess.emplace(); // Allocate a new CGI process slot
473
474 int errorCode = 500; // Default fallback error code
475 if (!CGI::initCgiProcess(*client.cgiProcess, request, server, location, _poll_fds, errorCode)) {
476 // Failed to spawn CGI → log and queue error response
477 Logger::logFrom(LogLevel::ERROR, "SocketManager",
478 "[CGI] Failed to initialize CGI process for client_fd " +
479 std::to_string(client_fd) + " with script: " + location.getPath());
480
481 HttpResponse err = ResponseBuilder::generateError(errorCode, server, request);
482 _client_info[client_fd].responses.push(err);
483
484 // Reset CGI state so client can recover
485 client.cgiProcess.reset();
486 client.isCgiProcessRunning = false;
487 client.currentCgiRequest = HttpRequest();
488
489 return false; // Error response queued instead of running CGI
490 }
491
492 return true; // CGI successfully started, will be managed by handleCgiPollEvents()
493}

495/**
496 * @brief Handles an HTTP request error by generating and queuing an error response.
497 *
498 * @details
499 * - Builds an error @ref HttpResponse using @ref ResponseBuilder::generateError.
500 * - Pushes the response into the client’s response queue.
501 * - Removes the faulty request from the pending request queue.
502 * - Returns whether the response requires closing the connection (e.g. HTTP/1.0 without
503 * keep-alive).
504 *
505 * @param fd File descriptor of the client connection.
506 * @param code HTTP status code to return (e.g. 400, 404, 413).
507 * @param req The offending HTTP request (may be partially parsed).
508 * @param server The server context used for error page resolution.
509 *
510 * @return `true` if the error response requires closing the connection,
511 * `false` otherwise (keep-alive).
512 */
513bool SocketManager::handleRequestErrorIfAny(int fd, int code, HttpRequest& req,
514 const Server& server) {
515 // Generate error response for this request
516 HttpResponse err = ResponseBuilder::generateError(code, server, req);
517
518 // Queue the response to be sent back to the client
519 _client_info[fd].responses.push(err);
520
521 // Drop the faulty request from the pending queue
522 _client_info[fd].pendingRequests.pop();
523
524 // Decide if the connection must be closed after sending this response
525 return err.isConnectionClose();
526}

528/**
529 * @brief Determines whether a request should be handled by a CGI process.
530 *
531 * @details
532 * A request qualifies for CGI execution if:
533 * - The path can be resolved to a valid absolute file path (`resolveAbsolutePath` not empty).
534 * - The HTTP method is either `GET` or `POST` (other methods are not CGI-eligible).
535 * - The resolved path matches a CGI-enabled location (`isCgiRequest`).
536 *
537 * If all conditions are met, returns `true` so the request is delegated
538 * to @ref handleCgiRequest. Otherwise, it should be served as a normal static/dynamic response.
539 *
540 * @param req The incoming HTTP request to evaluate.
541 * @param location The matched @ref Location context for the request path.
542 *
543 * @return `true` if the request should spawn a CGI process,
544 * `false` if it should be handled normally.
545 */
546bool SocketManager::shouldSpawnCgi(const HttpRequest& req, const Location& location) {
547 // Resolve absolute filesystem path of the requested resource
548 std::string resolved = location.resolveAbsolutePath(req.getPath());
549
550 // Only GET/POST methods are supported for CGI execution,
551 // and only if the path belongs to a CGI-enabled location.
552 return !resolved.empty() && (req.getMethod() == "GET" || req.getMethod() == "POST") &&
553 location.isCgiRequest(normalizePath(req.getPath()));
554}
