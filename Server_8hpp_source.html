34/**
35 * @brief Represents a virtual server block.
36 *
37 * @details Encapsulates configuration for a single virtual host, including
38 * its listening address, server names, error pages, body size limits,
39 * and nested location blocks. Each Server instance corresponds to a
40 * `server` block in the configuration file and participates in request
41 * routing based on host and port matching.

43 * @ingroup config
44 */

105 void addServerName(std::string_view name);

220 */
221 std::vector<Location>& getLocations() noexcept;

228 *

232 */
233 bool hasServerName(std::string_view name) const noexcept;
